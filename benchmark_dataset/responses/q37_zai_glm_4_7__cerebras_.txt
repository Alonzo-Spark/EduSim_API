        // --- 3. Formula Lab Logic (Interactive Simulation) ---

        let labAnimationId;
        let labState = {
            L: 1.0,
            g: 9.8,
            angle: Math.PI / 4, // Initial amplitude
            velocity: 0,
            time: 0
        };

        const canvas = document.getElementById('pendulumCanvas');
        const ctx = canvas.getContext('2d');
        
        // Setup canvas resolution
        function resizeCanvas() {
            const parent = canvas.parentElement;
            canvas.width = parent.clientWidth;
            canvas.height = 300;
        }
        window.addEventListener('resize', resizeCanvas);

        function openFormulaLab() {
            document.getElementById('lab-modal').style.display = 'flex';
            resizeCanvas();
            if(!labAnimationId) {
                labLoop();
            }
        }

        function closeLab() {
            document.getElementById('lab-modal').style.display = 'none';
            // Optional: cancelAnimationFrame(labAnimationId); 
            // We keep it running or pause it depending on preference. Let's keep it running for smoothness.
        }

        // Input Listeners
        const inputL = document.getElementById('input-L');
        const inputG = document.getElementById('input-g');
        const valL = document.getElementById('val-L');
        const valG = document.getElementById('val-g');
        const resultDisplay = document.getElementById('result-display');

        function updateLabValues() {
            labState.L = parseFloat(inputL.value);
            labState.g = parseFloat(inputG.value);
            
            valL.textContent = labState.L.toFixed(1);
            valG.textContent = labState.g.toFixed(1);

            // Calculate T
            const T = 2 * Math.PI * Math.sqrt(labState.L / labState.g);
            resultDisplay.textContent = T.toFixed(2) + ' s';
        }

        inputL.addEventListener('input', updateLabValues);
        inputG.addEventListener('input', updateLabValues);

        // Physics Engine for Lab
        function labLoop() {
            // Physics update (Simple Euler integration for visualization)
            // Equation: angular_acceleration = -(g/L) * sin(theta)
            
            const dt = 0.05; // Time step for simulation speed
            const acceleration = -(labState.g / labState.L) * Math.sin(labState.angle);
            
            labState.velocity += acceleration * dt;
            // Apply slight damping so it doesn't explode if integration errors accumulate
            labState.velocity *= 0.995; 
            labState.angle += labState.velocity * dt;

            drawPendulum();
            labAnimationId = requestAnimationFrame(labLoop);
        }

        function drawPendulum() {
            const w = canvas.width;
            const h = canvas.height;
            const originX = w / 2;
            const originY = 20;
            
            // Scale: Map meters to pixels. Max length 2m should fit in ~250px
            const scale = 120; 
            
            const bobX = originX + (labState.L * scale) * Math.sin(labState.angle);
            const bobY = originY + (labState.L * scale) * Math.cos(labState.angle);

            ctx.clearRect(0, 0, w, h);

            // Draw Pivot
            ctx.beginPath();
            ctx.fillStyle = '#2c3e50';
            ctx.arc(originX, originY, 5, 0, Math.PI * 2);
            ctx.fill();

            // Draw String
            ctx.beginPath();
            ctx.moveTo(originX, originY);
            ctx.lineTo(bobX, bobY);
            ctx.strokeStyle = '#333';
            ctx.lineWidth = 2;
            ctx.stroke();

            // Draw Bob
            ctx.beginPath();
            ctx.arc(bobX, bobY, 15, 0, Math.PI * 2);
            ctx.fillStyle = '#e74c3c';
            ctx.fill();
            ctx.strokeStyle = '#c0392b';
            ctx.lineWidth = 1;
            ctx.stroke();

            // Draw Arc (Amplitude guide)
            ctx.beginPath();
            ctx.arc(originX, originY, labState.L * scale, Math.PI/2 - 0.5, Math.PI/2 + 0.5);
            ctx.strokeStyle = 'rgba(0,0,0,0.1)';
            ctx.setLineDash([5, 5]);
            ctx.stroke();
            ctx.setLineDash([]);
        }

        renderContent();
        updateLabValues(); // Init lab values
