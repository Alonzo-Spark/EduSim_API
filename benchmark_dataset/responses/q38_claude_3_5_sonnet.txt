// Formula Lab component
function FormulaLab({ variables, labValues, setLabValues }) {
  // Simple lab for resonance frequency: f0 = 1/(2π√(LC))
  const inputVars = variables.filter(v => v.input);
  
  const handleChange = (symbol, value) => {
    setLabValues(prev => ({ ...prev, [symbol]: parseFloat(value) }));
  };

  const L = labValues['L'] || 0.1;
  const C = labValues['C'] || 0.0001;
  const result = 1 / (2 * Math.PI * Math.sqrt(L * C));

  return (
    <div className="mt-3 bg-white rounded-lg p-3 border border-purple-100">
      <p className="text-xs font-semibold text-purple-700 mb-2">⚗️ Formula Lab — Adjust values:</p>
      <div className="grid grid-cols-2 gap-2 mb-3">
        {inputVars.map((v, i) => (
          <div key={i} className="flex flex-col gap-1">
            <label className="text-xs text-gray-500">
              <InlineMath math={v.symbol} /> ({v.unit})
            </label>
            <input
              type="number"
              step="any"
              defaultValue={v.default}
              onChange={(e) => handleChange(v.symbol, e.target.value)}
              className="border border-gray-200 rounded px-2 py-1 text-xs w-full"
            />
          </div>
        ))}
      </div>
      <div className="bg-indigo-50 rounded-lg p-2 text-center">
        <span className="text-xs text-gray-500">Resonant Frequency </span>
        <span className="font-bold text-indigo-700 text-sm">{isNaN(result) || !isFinite(result) ? "—" : result.toFixed(2)} Hz</span>
      </div>
    </div>
  );
}

function FormulaCard({ title, formula, variables, context, calculate, resultLabel = "Result", resultUnit = "" }) {
