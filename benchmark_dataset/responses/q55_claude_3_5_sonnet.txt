// ============================================================
// PHYSICS ENGINE
// ============================================================
function computeStoppingDistance(v0: number, mu: number, g = 9.8): number {
  if (mu < 0.001) return Infinity;
  return (v0 * v0) / (2 * mu * g);
}

function computeStoppingTime(v0: number, mu: number, g = 9.8): number {
  if (mu < 0.001) return Infinity;
  return v0 / (mu * g);
}

function PhysicsSandbox() {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const animRef = useRef<number>(0);
  const stateRef = useRef<BlockState>({ x: 60, v: 0, a: 0, t: 0, stopped: false, distanceTraveled: 0 });
  const runningRef = useRef(false);
  const lastTimeRef = useRef<number | null>(null);

  const [mu, setMu] = useState(0.3);
  const [v0, setV0] = useState(8);
  const [isRunning, setIsRunning] = useState(false);
  const [displayState, setDisplayState] = useState<BlockState>({
    x: 60, v: 0, a: 0, t: 0, stopped: false, distanceTraveled: 0,
  });
  const [trailPoints, setTrailPoints] = useState<number[]>([]);

  const g = 9.8;
  const CANVAS_W = 620;
  const CANVAS_H = 180;
  const GROUND_Y = 130;
  const BLOCK_SIZE = 36;
  const MAX_X = CANVAS_W - BLOCK_SIZE - 20;
  const PIXELS_PER_METER = 28;

  const reset = useCallback(() => {
    cancelAnimationFrame(animRef.current);
    runningRef.current = false;
    lastTimeRef.current = null;
    setIsRunning(false);
    const s: BlockState = { x: 60, v: v0, a: -(mu * g), t: 0, stopped: mu < 0.001, distanceTraveled: 0 };
    stateRef.current = s;
    setDisplayState(s);
    setTrailPoints([]);
  }, [mu, v0]);

  useEffect(() => { reset(); }, [mu, v0]);

  const draw = useCallback((state: BlockState, trail: number[]) => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    ctx.clearRect(0, 0, CANVAS_W, CANVAS_H);

    // Background
    const bgGrad = ctx.createLinearGradient(0, 0, 0, CANVAS_H);
    bgGrad.addColorStop(0, "#f8fafc");
    bgGrad.addColorStop(1, "#f1f5f9");
    ctx.fillStyle = bgGrad;
    ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);

    // Ground
    const groundGrad = ctx.createLinearGradient(0, GROUND_Y, 0, CANVAS_H);
    const muColor = mu < 0.05 ? "#e0f2fe" : mu < 0.3 ? "#d1fae5" : mu < 0.6 ? "#fef3c7" : "#fee2e2";
    groundGrad.addColorStop(0, muColor);
    groundGrad.addColorStop(1, "#e2e8f0");
    ctx.fillStyle = groundGrad;
    ctx.fillRect(0, GROUND_Y, CANVAS_W, CANVAS_H - GROUND_Y);

    // Ground line
    ctx.strokeStyle = mu < 0.05 ? "#38bdf8" : mu < 0.3 ? "#34d399" : mu < 0.6 ? "#fbbf24" : "#f87171";
    ctx.lineWidth = 2.5;
    ctx.beginPath();
    ctx.moveTo(0, GROUND_Y);
    ctx.lineTo(CANVAS_W, GROUND_Y);
    ctx.stroke();

    // Surface texture lines
    if (mu > 0.05) {
      ctx.strokeStyle = "rgba(0,0,0,0.06)";
      ctx.lineWidth = 1;
      const spacing = Math.max(8, 40 - mu * 30);
      for (let x = 0; x < CANVAS_W; x += spacing) {
        ctx.beginPath();
        ctx.moveTo(x, GROUND_Y);
        ctx.lineTo(x - 10, CANVAS_H);
        ctx.stroke();
      }
    }

    // Trail dots
    trail.forEach((tx, i) => {
      const alpha = (i / trail.length) * 0.4;
      ctx.fillStyle = `rgba(99, 102, 241, ${alpha})`;
      ctx.beginPath();
      ctx.arc(tx + BLOCK_SIZE / 2, GROUND_Y - 4, 3, 0, Math.PI * 2);
      ctx.fill();
    });

    // Block shadow
    ctx.fillStyle = "rgba(0,0,0,0.08)";
    ctx.beginPath();
    ctx.ellipse(state.x + BLOCK_SIZE / 2, GROUND_Y + 4, BLOCK_SIZE / 2, 5, 0, 0, Math.PI * 2);
    ctx.fill();

    // Block body
    const blockGrad = ctx.createLinearGradient(state.x, GROUND_Y - BLOCK_SIZE, state.x, GROUND_Y);
    blockGrad.addColorStop(0, "#6366f1");
    blockGrad.addColorStop(1, "#4338ca");
    ctx.fillStyle = blockGrad;
    ctx.beginPath();
    ctx.roundRect(state.x, GROUND_Y - BLOCK_SIZE, BLOCK_SIZE, BLOCK_SIZE, 6);
    ctx.fill();

    // Block shine
    ctx.fillStyle = "rgba(255,255,255,0.15)";
    ctx.beginPath();
    ctx.roundRect(state.x + 4, GROUND_Y - BLOCK_SIZE + 4, BLOCK_SIZE - 8, 10, 3);
    ctx.fill();

    // Velocity arrow
    if (state.v > 0.2 && !state.stopped) {
      const arrowLen = Math.min(state.v * 6, 80);
      const arrowX = state.x + BLOCK_SIZE + 4;
      const arrowY = GROUND_Y - BLOCK_SIZE / 2;
      ctx.strokeStyle = "#10b981";
      ctx.lineWidth = 2.5;
      ctx.beginPath();
      ctx.moveTo(arrowX, arrowY);
      ctx.lineTo(arrowX + arrowLen, arrowY);
      ctx.stroke();
      ctx.fillStyle = "#10b981";
      ctx.beginPath();
      ctx.moveTo(arrowX + arrowLen + 8, arrowY);
      ctx.lineTo(arrowX + arrowLen, arrowY - 5);
      ctx.lineTo(arrowX + arrowLen, arrowY + 5);
      ctx.fill();
      ctx.fillStyle = "#10b981";
      ctx.font = "bold 10px system-ui";
      ctx.fillText(`v = ${state.v.toFixed(1)} m/s`, arrowX + arrowLen / 2 - 20, arrowY - 8);
    }

    // Friction arrow (opposing)
    if (mu > 0.01 && state.v > 0.2 && !state.stopped) {
      const fArrowLen = Math.min(mu * 80, 60);
      const fArrowX = state.x - 4;
      const fArrowY = GROUND_Y - 10;
      ctx.strokeStyle = "#ef4444";
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.moveTo(fArrowX, fArrowY);
      ctx.lineTo(fArrowX - fArrowLen, fArrowY);
      ctx.stroke();
      ctx.fillStyle = "#ef4444";
      ctx.beginPath();
      ctx.moveTo(fArrowX - fArrowLen - 8, fArrowY);
      ctx.lineTo(fArrowX - fArrowLen, fArrowY - 4);
      ctx.lineTo(fArrowX - fArrowLen, fArrowY + 4);
      ctx.fill();
      ctx.font = "bold 9px system-ui";
      ctx.fillStyle = "#ef4444";
      ctx.fillText("fₖ", fArrowX - fArrowLen / 2 - 6, fArrowY - 6);
    }

    // Zero friction badge
    if (mu < 0.01) {
      ctx.fillStyle = "#0ea5e9";
      ctx.font = "bold 11px system-ui";
      ctx.fillText("✦ Frictionless", 10, GROUND_Y - 8);
    }

    // Stopped indicator
    if (state.stopped && mu >= 0.01) {
      ctx.fillStyle = "#64748b";
      ctx.font = "bold 11px system-ui";
      ctx.fillText("STOPPED", state.x - 4, GROUND_Y - BLOCK_SIZE - 10);
    }
  }, [mu]);

  const startAnimation = useCallback(() => {
    if (runningRef.current) return;
    runningRef.current = true;
    setIsRunning(true);
    lastTimeRef.current = null;
    const trails: number[] = [];

    const tick = (timestamp: number) => {
      if (!runningRef.current) return;
      if (lastTimeRef.current === null) lastTimeRef.current = timestamp;
      const dt = Math.min((timestamp - lastTimeRef.current) / 1000, 0.05);
      lastTimeRef.current = timestamp;

      const s = stateRef.current;
      if (s.stopped || s.x >= MAX_X) {
        runningRef.current = false;
        setIsRunning(false);
        return;
      }

      const newV = mu < 0.001 ? s.v : Math.max(0, s.v - mu * g * dt);
      const newX = s.x + newV * PIXELS_PER_METER * dt;
      const clampedX = Math.min(newX, MAX_X);
      const newT = s.t + dt;
      const newDist = s.distanceTraveled + (clampedX - s.x) / PIXELS_PER_METER;
      const stopped = newV < 0.01 || clampedX >= MAX_X;

      if (Math.floor(newT * 10) > Math.floor(s.t * 10)) {
        trails.push(clampedX);
        setTrailPoints([...trails]);
      }

      const newState: BlockState = {
        x: clampedX,
        v: newV,
        a: mu < 0.001 ? 0 : -(mu * g),
        t: newT,
        stopped,
        distanceTraveled: newDist,
      };

      stateRef.current = newState;
      setDisplayState({ ...newState });
      draw(newState, trails);

      if (!stopped) {
        animRef.current = requestAnimationFrame(tick);
      } else {
        runningRef.current = false;
        setIsRunning(false);
      }
    };

    animRef.current = requestAnimationFrame(tick);
  }, [mu, draw]);

  useEffect(() => {
    draw(stateRef.current, []);
  }, [draw]);

  const muLabel = mu < 0.01 ? "Frictionless (Ice)" : mu < 0.2 ? "Very Smooth" : mu < 0.4 ? "Wood / Tile" : mu < 0.7 ? "Rough Concrete" : "Very Rough";
  const muColor = mu < 0.05 ? "text-sky-600" : mu < 0.3 ? "text-emerald-600" : mu < 0.6 ? "text-amber-600" : "text-red-600";

  return (
    <div className="rounded-2xl border border-slate-200 bg-white shadow-sm overflow-hidden">
      <div className="bg-gradient-to-r from-violet-700 to-indigo-700 px-5 py-3 flex items-center justify-between">
        <div>
          <p className="text-violet-300 text-[10px] font-semibold uppercase tracking-widest">Live Sandbox</p>
          <h3 className="text-white font-bold text-sm">Sliding Block · Friction Scale</h3>
        </div>
        <div className="flex items-center gap-1.5">
          <div className={`h-2 w-2 rounded-full ${isRunning ? "bg-emerald-400 animate-pulse" : "bg-slate-400"}`} />
          <span className="text-[10px] text-violet-300">{isRunning ? "Running" : "Ready"}</span>
        </div>
      </div>

      {/* Canvas */}
      <div className="relative bg-slate-50 border-b border-slate-100 overflow-hidden">
        <canvas
          ref={canvasRef}
          width={CANVAS_W}
          height={CANVAS_H}
          className="w-full"
          style={{ maxHeight: 180 }}
        />
      </div>

      {/* Live Readouts */}
      <div className="grid grid-cols-4 border-b border-slate-100">
        {[
          { label: "Velocity", value: displayState.stopped ? "0.00" : displayState.v.toFixed(2), unit: "m/s", color: "text-emerald-600" },
          { label: "Acceleration", value: displayState.a.toFixed(2), unit: "m/s²", color: "text-red-500" },
          { label: "Time", value: displayState.t.toFixed(2), unit: "s", color: "text-slate-700" },
          { label: "Distance", value: displayState.distanceTraveled.toFixed(2), unit: "m", color: "text-indigo-600" },
        ].map((item) => (
          <div key={item.label} className="px-3 py-3 text-center border-r border-slate-100 last:border-r-0">
            <p className={`text-lg font-bold font-mono ${item.color}`}>{item.value}</p>
            <p className="text-[9px] text-slate-400">{item.unit}</p>
            <p className="text-[10px] font-medium text-slate-500 mt-0.5">{item.label}</p>
          </div>
        ))}
      </div>

      {/* Controls */}
      <div className="p-4 space-y-3">
        <div className="grid grid-cols-2 gap-4">
          <div>
            <div className="flex justify-between mb-1">
              <label className="text-xs font-semibold text-slate-600">
                <InlineMath math="\mu_k" /> — Friction
              </label>
              <span className={`text-xs font-bold ${muColor}`}>{mu.toFixed(3)}</span>
            </div>
            <input
              type="range" min={0} max={0.9} step={0.001} value={mu}
              onChange={(e) => setMu(parseFloat(e.target.value))}
              className="w-full accent-indigo-600"
              disabled={isRunning}
            />
            <p className={`text-[10px] font-medium mt-0.5 ${muColor}`}>{muLabel}</p>
          </div>
          <div>
            <div className="flex justify-between mb-1">
              <label className="text-xs font-semibold text-slate-600">
                <InlineMath math="v_0" /> — Initial Speed
              </label>
              <span className="text-xs font-bold text-slate-700">{v0} m/s</span>
            </div>
            <input
              type="range" min={2} max={20} step={0.5} value={v0}
              onChange={(e) => setV0(parseFloat(e.target.value))}
              className="w-full accent-indigo-600"
              disabled={isRunning}
            />
          </div>
        </div>

        {/* Predicted values */}
        <div className="rounded-xl bg-slate-50 border border-slate-100 px-4 py-2.5 flex gap-6 text-xs">
          <div>
            <span className="text-slate-400">Predicted stop dist: </span>
            <span className="font-bold text-indigo-700">
              {mu < 0.001 ? "∞" : `${computeStoppingDistance(v0, mu).toFixed(1)} m`}
            </span>
          </div>
          <div>
            <span className="text-slate-400">Predicted stop time: </span>
            <span className="font-bold text-indigo-700">
              {mu < 0.001 ? "∞" : `${computeStoppingTime(v0, mu).toFixed(1)} s`}
            </span>
          </div>
        </div>

        {mu < 0.01 && (
          <div className="rounded-xl bg-sky-50 border border-sky-200 px-4 py-2.5 flex gap-2 items-start">
            <span className="text-sky-500 text-sm mt-0.5">🧊</span>
            <p className="text-xs text-sky-700 leading-relaxed">
              <strong>μ ≈ 0 — Newton's First Law:</strong> No net force acts on the block. It
              slides at <InlineMath math="v_0" /> indefinitely. The block exits the canvas and
              would travel forever in an ideal system.
            </p>
          </div>
        )}

        <div className="flex gap-2">
          <button
            onClick={startAnimation}
            disabled={isRunning}
            className="flex-1 rounded-xl bg-indigo-600 py-2.5 text-sm font-semibold text-white hover:bg-indigo-700 disabled:opacity-40 disabled:cursor-not-allowed transition-colors shadow-sm"
          >
            {isRunning ? "Simulating…" : "▶  Launch Block"}
          </button>
          <button
            onClick={reset}
            className="rounded-xl border border-slate-200 bg-white px-4 py-2.5 text-sm font-medium text-slate-600 hover:bg-slate-50 transition-colors"
          >
            ↺ Reset
          </button>
        </div>
      </div>
    </div>
  );
}
