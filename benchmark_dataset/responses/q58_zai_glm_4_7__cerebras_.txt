        // Physics Calculations
        function calculatePhysics() {
            const thetaRad = state.theta * (Math.PI / 180);
            
            // 1. Acceleration down ramp: a = g(sin(theta) - mu*cos(theta))
            let accel = state.g * (Math.sin(thetaRad) - state.mu * Math.cos(thetaRad));
            if (accel < 0) accel = 0; // Friction too high, won't move

            // 2. Ramp Length L from height h: sin(theta) = h/L => L = h/sin(theta)
            const L = state.h / Math.sin(thetaRad);

            // 3. Velocity at bottom: v = sqrt(2*a*L)
            const v = Math.sqrt(2 * accel * L);

            // 4. Projectile Motion
            // Launch angle is -theta (downwards) relative to horizontal? 
            // Usually inclined plane problems assume it leaves horizontally or tangent to slope.
            // Let's assume it leaves tangent to the slope (downward angle -theta).
            // vx = v * cos(theta)
            // vy = -v * sin(theta)
            
            // Time to hit ground (y = 0). Start y = h.
            // y(t) = h + vy*t - 0.5*g*t^2
            // 0 = h - v*sin(theta)*t - 0.5*g*t^2
            // 0.5*g*t^2 + v*sin(theta)*t - h = 0
            // Quadratic: At^2 + Bt + C = 0
            const A = 0.5 * state.g;
            const B = v * Math.sin(thetaRad);
            const C = -state.h;

            const discriminant = B*B - 4*A*C;
            let t = 0;
            if (discriminant >= 0) {
                t = (-B + Math.sqrt(discriminant)) / (2*A);
            }

            const range = v * Math.cos(thetaRad) * t;

            return { v, range, L, accel, thetaRad };
        }

        // Drawing
        function draw() {
            const width = canvas.width / 2;
            const height = canvas.height / 2;
            const phys = calculatePhysics();

            // Clear
            ctx.clearRect(0, 0, width, height);
            
            // Scale Factor (pixels per meter)
            // We need to fit the ramp and the range in the view.
            const totalWidthNeeded = phys.L * Math.cos(phys.thetaRad) + phys.range;
            const maxDim = Math.max(totalWidthNeeded, state.h * 1.5);
            const scale = (width * 0.8) / maxDim;

            // Coordinate Transform: Origin at bottom-left of ramp
            const originX = 50;
            const originY = height - 50;

            // Draw Ground
            ctx.beginPath();
            ctx.moveTo(0, originY);
            ctx.lineTo(width, originY);
            ctx.strokeStyle = '#64748b';
            ctx.lineWidth = 2;
            ctx.stroke();

            // Draw Ramp Triangle
            const rampBaseX = originX + phys.L * Math.cos(phys.thetaRad) * scale;
            const rampTopY = originY - state.h * scale;

            ctx.beginPath();
            ctx.moveTo(originX, originY); // Bottom left
            ctx.lineTo(rampBaseX, originY); // Bottom right
            ctx.lineTo(originX, rampTopY); // Top
            ctx.closePath();
            ctx.fillStyle = '#e2e8f0';
            ctx.fill();
            ctx.stroke();

            // Draw Block (at top)
            const blockSize = 20;
            ctx.fillStyle = '#2563eb';
            ctx.fillRect(originX - blockSize/2, rampTopY - blockSize/2, blockSize, blockSize);

            // Draw Trajectory Path (Dashed)
            ctx.beginPath();
            ctx.setLineDash([5, 5]);
            ctx.strokeStyle = '#f59e0b';
            ctx.lineWidth = 3;
            
            // Simulate points
            const vx = phys.v * Math.cos(phys.thetaRad);
            const vy = -phys.v * Math.sin(phys.thetaRad); // Negative because down
            
            let curX = originX;
            let curY = rampTopY;
            let curVX = vx;
            let curVY = vy;
            const dt = 0.05; // Time step for drawing

            ctx.moveTo(curX, curY);

            while (curY < originY) {
                curX += curVX * dt * scale;
                curY -= curVY * dt * scale; // Canvas Y is inverted
                curVY -= state.g * dt; // Gravity acts down (positive in math, negative in canvas Y logic relative to velocity)
                
                // Actually simpler: Update Physics Y then map to Canvas
                // Let's stick to standard physics coordinates then map
            }
            
            // Better trajectory drawing: Map physics coords to canvas
            // Start: x=0, y=h. End: x=Range, y=0.
            // Canvas Start: x=originX, y=rampTopY
            
            ctx.beginPath();
            ctx.moveTo(originX, rampTopY);
            
            for(let t=0; t<=phys.t*1.1; t+=0.1) {
                const px = vx * t;
                const py = state.h + vy * t - 0.5 * state.g * t * t;
                
                const canvasX = originX + px * scale;
                const canvasY = originY - py * scale; // Flip Y
                
                if (canvasY > originY) break; // Hit ground
                ctx.lineTo(canvasX, canvasY);
            }
            ctx.stroke();
            ctx.setLineDash([]);

            // Draw Impact Marker
            const impactX = originX + phys.range * scale;
            ctx.beginPath();
            ctx.arc(impactX, originY, 4, 0, Math.PI*2);
            ctx.fillStyle = '#ef4444';
            ctx.fill();

            // Update UI Displays
            displays.vel.textContent = phys.v.toFixed(2) + " m/s";
            displays.range.textContent = phys.range.toFixed(2) + " m";
        }

        // Event Listeners
        function updateState() {
            state.g = parseFloat(inputs.g.value);
            state.theta = parseFloat(inputs.theta.value);
            state.mu = parseFloat(inputs.mu.value);
            state.h = parseFloat(inputs.h.value);

            displays.g.textContent = state.g.toFixed(1) + " m/s²";
            displays.theta.textContent = state.theta + "°";
            displays.mu.textContent = state.mu.toFixed(2);
            displays.h.textContent = state.h + " m";

            draw();
        }

        Object.values(inputs).forEach(input => {
            input.addEventListener('input', updateState);
        });

        // Initial Draw
        setTimeout(() => {
            resizeCanvas();
            updateState();
        }, 100);

    </script>
