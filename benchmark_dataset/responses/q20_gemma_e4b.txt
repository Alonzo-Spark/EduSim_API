```python
import re

def parse_table(markdown_table_string):
    """Parses a Markdown table string and returns a list of lists representing the table."""
    lines = markdown_table_string.strip().split('\n')
    header = [h.strip() for h in re.split(r"\|", lines[0])[1:-1]]
    data = []
    for line in lines[2:]:
        row = [d.strip() for d in re.split(r"\|", line)[1:-1]]
        data.append(row)
    return header, data

def render_content(blocks):
    """Renders the content blocks into a simple HTML structure."""
    html = ""
    for block in blocks:
        if block.type == "heading":
            html += f"<h1>{block.content}</h1>\n"
        elif block.type == "paragraph":
            html += f"<p>{block.content}</p>\n"
        elif block.type == "formula":
            html += f"<h2>{block.content['title']}</h2>\n"
            html += f"<p><strong>{block.content['formula']}</strong></p>\n"
            html += "<p><em>{block.content['surroundingContext']}</em></p>\n"
            html += "<ul>"
            for var in block.content['variables']:
              html += f"<li>{var}</li>"
            html += "</ul>"
            html += "<a href='#'>[ Explain Formula ]</a>\n"
            html += "<a href='#'>[ Open Formula Lab ]</a>\n"
        elif block.type == "table":
            header, data = block.content
            html += "<table>\n"
            html += "  <thead>\n"
            html += "    <tr>\n"
            for h in header:
                html += f"      <th>{h}</th>\n"
            html += "    </tr>\n"
            html += "  </thead>\n"
            html += "  <tbody>\n"
            for row in data:
                html += "    <tr>\n"
                for cell in row:
                    html += f"      <td>{cell}</td>\n"
                html += "    </tr>\n"
            html += "  </tbody>\n"
            html += "</table>\n"
        elif block.type == "section":
            html += f"<div><h2>{block.content}</h2></div>\n"
        else:
            html += f"<p>Unknown block type: {block.type}</p>\n"
    return html
```
