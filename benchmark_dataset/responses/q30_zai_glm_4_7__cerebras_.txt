        // Since we are using the auto-render extension script usually, 
        // let's ensure we have a robust way to render if the extension isn't fully loaded 
        // or if we want to be explicit.
        
        function renderMathInElement(element, options) {
            // Check if renderMathInElement exists (from auto-render extension)
            if (typeof renderMathInElement === 'function') {
                renderMathInElement(element, options);
            } else {
                // Fallback: manual search for $$ blocks
                const texBlocks = element.querySelectorAll('.formula-display');

                    try {

                });
            }
        }

        const rhoSlider = document.getElementById('density-slider');
        const velDisplay = document.getElementById('vel-display');
        const rhoDisplay = document.getElementById('rho-display');
        const keResult = document.getElementById('ke-result');
        const pressureResult = document.getElementById('pressure-result');

            const v = parseFloat(velSlider.value);
            const rho = parseFloat(rhoSlider.value);

            // Update Labels

            rhoDisplay.textContent = rho;
