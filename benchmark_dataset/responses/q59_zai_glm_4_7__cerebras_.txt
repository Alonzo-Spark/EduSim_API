    <script>
        // --- Data for Formula Explanation ---
        const formulaData = {
            momentum: {
                title: "Conservation of Momentum",
                text: `
                    <p>This formula states that the total momentum of an isolated system remains constant. In the context of equal masses colliding:</p>
                    <p>Before collision, the total momentum is the sum of individual momentums ($mv$). After collision, it is the sum of the final momentums.</p>
                    <p>Because mass $m$ is the same for both objects, we can divide the entire equation by $m$, leaving us with a simple relationship between velocities:</p>
                    <div style="text-align:center; padding: 10px;">$$v_{1i} + v_{2i} = v_{1f} + v_{2f}$$</div>
                    <p>This means the <em>sum</em> of the velocities must be the same before and after impact.</p>
                    <ul class="variable-list">
                        <li><span class="variable-symbol">m</span> Mass of the object (kg)</li>
                        <li><span class="variable-symbol">v</span> Velocity (m/s)</li>
                        <li><span class="variable-symbol">i</span> Initial state (before collision)</li>
                        <li><span class="variable-symbol">f</span> Final state (after collision)</li>
                    </ul>
                `
            }
        };

        // --- Modal Handling ---
        function openExplanation(type) {
            const modal = document.getElementById('explanation-modal');
            const content = document.getElementById('explanation-text');
            
            if (formulaData[type]) {
                content.innerHTML = formulaData[type].text;
                // Re-render math in the modal
                renderMathInElement(content);
            }
            
            modal.classList.add('active');
        }

        function openLab() {
            const modal = document.getElementById('lab-modal');
            modal.classList.add('active');
            resetSimulation();
            animate();
        }

        function closeModals() {
            document.querySelectorAll('.modal-overlay').forEach(el => el.classList.remove('active'));
            stopAnimation();
        }

        // Close modal on click outside
        window.onclick = function(event) {
            if (event.target.classList.contains('modal-overlay')) {
                closeModals();
            }
        }

        // --- Collision Simulation Logic (Canvas) ---
        const canvas = document.getElementById('simCanvas');
        const ctx = canvas.getContext('2d');
        let animationId;
        let isRunning = false;
        
        // Simulation State
        let balls = [];
        const restitutionSlider = document.getElementById('restitution-slider');
        const restitutionVal = document.getElementById('restitution-val');
        const observationText = document.getElementById('lab-observation');

        // Setup Canvas Resolution
        function resizeCanvas() {
            canvas.width = canvas.parentElement.clientWidth;
            canvas.height = 200;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        class Ball {
            constructor(x, radius, color, label) {
                this.x = x;
                this.y = canvas.height / 2;
                this.radius = radius;
                this.color = color;
                this.label = label;
                this.vx = 0;
                this.mass = 1; // Equal mass
            }

            draw() {
                // Draw Ball
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();
                ctx.strokeStyle = '#333';
                ctx.lineWidth = 2;
                ctx.stroke();
                ctx.closePath();

                // Draw Label
                ctx.fillStyle = '#fff';
                ctx.font = 'bold 16px Inter';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillText(this.label, this.x, this.y);

                // Draw Velocity Vector
                if (Math.abs(this.vx) > 0.1) {
                    ctx.beginPath();
                    ctx.moveTo(this.x, this.y - this.radius - 10);
                    ctx.lineTo(this.x + this.vx * 20, this.y - this.radius - 10); // Scale vector for visibility
                    ctx.strokeStyle = '#2563eb';
                    ctx.lineWidth = 3;
                    ctx.stroke();
                    
                    // Arrowhead
                    ctx.beginPath();
                    const dir = this.vx > 0 ? 1 : -1;
                    ctx.moveTo(this.x + this.vx * 20, this.y - this.radius - 10);
                    ctx.lineTo(this.x + this.vx * 20 - (5 * dir), this.y - this.radius - 15);
                    ctx.lineTo(this.x + this.vx * 20 - (5 * dir), this.y - this.radius - 5);
                    ctx.lineTo(this.x + this.vx * 20, this.y - this.radius - 10);
                    ctx.fillStyle = '#2563eb';
                    ctx.fill();
                }
            }

            update() {
                this.x += this.vx;

                // Wall Collisions
                if (this.x - this.radius < 0) {
                    this.x = this.radius;
                    this.vx *= -1;
                }
                if (this.x + this.radius > canvas.width) {
                    this.x = canvas.width - this.radius;
                    this.vx *= -1;
                }
            }
        }

        function resetSimulation() {
            balls = [
                new Ball(100, 25, '#ef4444', 'A'), // Red
                new Ball(400, 25, '#3b82f6', 'B')  // Blue
            ];
            
            // Initial Conditions: A moves right at 4, B is stationary
            balls[0].vx = 4; 
            balls[1].vx = 0;

            updateObservation();
        }

        function updateObservation() {
            const e = parseFloat(restitutionSlider.value);
            if (e === 1) {
                observationText.innerText = "Elastic: Mass A stops, Mass B takes off at A's speed.";
            } else if (e === 0) {
                observationText.innerText = "Inelastic: Both masses stick together and move at half speed.";
            } else {
                observationText.innerText = `Partially Elastic (e=${e}): Masses separate but lose some speed.`;
            }
        }

        restitutionSlider.addEventListener('input', (e) => {
            restitutionVal.innerText = parseFloat(e.target.value).toFixed(1);
            updateObservation();
            // Optional: Reset sim when slider changes to avoid chaos
            resetSimulation();
        });

        function checkCollision(b1, b2) {
            const dx = b2.x - b1.x;
            const dist = Math.abs(dx);
            const minDist = b1.radius + b2.radius;

            if (dist < minDist) {
                // Resolve overlap to prevent sticking
                const overlap = minDist - dist;
                b1.x -= overlap / 2;
                b2.x += overlap / 2;

                // 1D Collision Formula with Coefficient of Restitution (e)
                // v1_final = ( (m1 - e*m2)*v1 + (1+e)*m2*v2 ) / (m1+m2)
                // Since m1 = m2 = 1:
                // v1_final = ( (1 - e)*v1 + (1+e)*v2 ) / 2
                // v2_final = ( (1+e)*v1 + (1 - e)*v2 ) / 2

                const e = parseFloat(restitutionSlider.value);
                const v1 = b1.vx;
                const v2 = b2.vx;

                b1.vx = ((1 - e) * v1 + (1 + e) * v2) / 2;
                b2.vx = ((1 + e) * v1 + (1 - e) * v2) / 2;
            }
        }

        function animate() {
            if (!isRunning) return;
            
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Draw Floor Line
            ctx.beginPath();
            ctx.moveTo(0, canvas.height/2 + 25);
            ctx.lineTo(canvas.width, canvas.height/2 + 25);
            ctx.strokeStyle = '#e5e7eb';
            ctx.lineWidth = 2;
            ctx.stroke();

            balls.forEach(ball => {
                ball.update();
                ball.draw();
            });

            checkCollision(balls[0], balls[1]);

            animationId = requestAnimationFrame(animate);
        }

        function startAnimation() {
            if (!isRunning) {
                isRunning = true;
                animate();
            }
        }

        function stopAnimation() {
            isRunning = false;
            cancelAnimationFrame(animationId);
        }

        // Initialize simulation but don't start loop until modal opens
        resizeCanvas();
        resetSimulation();
        
        // Override reset button to ensure animation starts
        const originalReset = resetSimulation;
        resetSimulation = function() {
            originalReset();
            startAnimation();
        };

    </script>
