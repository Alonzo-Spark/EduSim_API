import { useState } from "react";
import katex from "katex";

function M({ t, d = false }) {
  try {
    const html = katex.renderToString(t, { throwOnError: false, displayMode: d });
    return (
      <span
        className={d ? "block my-1" : "inline"}
        dangerouslySetInnerHTML={{ __html: html }}
      />
    );
  } catch {
    return <code className="text-sm">{t}</code>;
  }
}

function FormulaLab() {
  const [v, setV] = useState({ V: "", I: "", R: "" });
  const [res, setRes] = useState(null);

  function calc() {
    const V = parseFloat(v.V), I = parseFloat(v.I), R = parseFloat(v.R);
    const r = {};
    if (!isNaN(V) && !isNaN(I)) r.vi = (V * I).toFixed(4);
    if (!isNaN(I) && !isNaN(R)) r.ir = (I * I * R).toFixed(4);
    if (!isNaN(V) && !isNaN(R) && R !== 0) r.vr = (V * V / R).toFixed(4);
    setRes(Object.keys(r).length ? r : "none");
  }

  return (
    <div className="space-y-3">
      <div className="grid grid-cols-3 gap-2">
        {[["V","Voltage","V"],["I","Current","A"],["R","Resistance","Ω"]].map(([k,lbl,unit]) => (
          <div key={k}>
            <div className="text-xs text-gray-500 mb-1">{lbl}</div>
            <div className="flex border border-gray-200 rounded-lg overflow-hidden bg-white">
              <input
                type="number"
                value={v[k]}
                onChange={e => setV(p => ({ ...p, [k]: e.target.value }))}
                placeholder="0"
                className="flex-1 w-0 min-w-0 px-2 py-2 text-sm focus:outline-none"
              />
              <span className="px-2 bg-gray-50 border-l border-gray-200 text-xs text-gray-400 flex items-center">{unit}</span>
            </div>
          </div>
        ))}
      </div>
      <button
        onClick={calc}
        className="w-full py-2 bg-violet-600 hover:bg-violet-700 text-white text-sm font-semibold rounded-lg transition-colors"
      >
        Calculate →
      </button>
      {res && res !== "none" && (
        <div className="bg-white border border-violet-100 rounded-xl divide-y divide-gray-50 text-sm">
          {res.vi && (
            <div className="flex justify-between px-4 py-2.5">
              <span className="text-gray-500">P = V × I</span>
              <span className="font-bold text-violet-700">{res.vi} W</span>
            </div>
          )}
          {res.ir && (
            <div className="flex justify-between px-4 py-2.5">
              <span className="text-gray-500">P = I²R</span>
              <span className="font-bold text-violet-700">{res.ir} W</span>
            </div>
          )}
          {res.vr && (
            <div className="flex justify-between px-4 py-2.5">
              <span className="text-gray-500">P = V²/R</span>
              <span className="font-bold text-violet-700">{res.vr} W</span>
            </div>
          )}
        </div>
      )}
      {res === "none" && (
        <p className="text-xs text-red-500 text-center">Enter at least two values to calculate.</p>
      )}
    </div>
  );
}
