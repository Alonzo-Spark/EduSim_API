```python
import re  # For regular expressions (formula extraction)

class Section:
    def __init__(self, title, content):
        self.title = title
        self.content = content

class FormulaCard:
    def __init__(self, formula, title, surrounding_context, variables):
        self.formula = formula
        self.title = title
        self.surrounding_context = surrounding_context
        self.variables = variables

class TableCard:
    def __init__(self, data, headers):
        self.data = data  # List of lists representing the table
        self.headers = headers  # List of strings for the table headers

class AI_Tutor_Output:
    def __init__(self, sections):
        self.sections = sections

def parse_markdown(markdown_text):
    """
    Parses markdown text and generates the structured AI Tutor output.
    """

    sections = []
    current_section = None
    formula_data = []  # Store formula extraction results

    lines = markdown_text.splitlines()

    for line in lines:
        line = line.strip()

        if line.startswith("# "):
            if current_section: #save previous section
                sections.append(current_section)
            current_section = Section(line[2:], "") # Extract title

        elif line.startswith("## "):
            if current_section:
                current_section.content += "\n" + line[3:]
            else:
                #Handle headings outside of sections (unlikely) -- maybe flag as error
                print("Warning: Heading outside of section detected.")
                pass

        elif line.startswith("$$"):
            # Extract formula
            formula_match = re.search(r'\$\$([\s\S]*?)\$\$', line)
            if formula_match:
                formula_content = formula_match.group(1).strip()
                title = "Main Formula" # Default title
                surrounding_context =  "..."  # Placeholder, will be populated later
                variables = [] # Placeholder
                formula_data.append({
                    "formula": formula_content,
                    "title": title,
                    "surrounding_context": surrounding_context,
                    "variables": variables
                })
                # Remove the formula line from the input text
                lines.remove(line)
                continue  # Skip further processing for this line

        elif line.startswith("Table"):
            # Detect tables
            table_data = []
            headers = []
            table_lines = []
            i = lines.index(line) + 1
            while i < len(lines) and not lines[i].startswith("#"):
                table_lines.append(lines[i])
                i += 1
            # Extract headers
            if table_lines:
                headers = table_lines[0].split()
                table_data = table_lines[1:] # remaining rows
            
            if table_data:
                table_card = TableCard(table_data, headers)
                sections.append(table_card)
                lines = lines[i:]  # Remove the table from the input
                continue

        else:
            if current_section:
                current_section.content += "\n" + line
            else:
                #Handle text outside sections -- maybe flag as error
                print("Warning: Text outside of section detected.")
                pass # ignore

    # Process any remaining section
    if current_section:
        sections.append(current_section)

    # Process the extracted formulas
    for formula_info in formula_data:
        formula_card = FormulaCard(
            formula_info["formula"],
            formula_info["title"],
            formula_info["surrounding_context"],
            formula_info["variables"]
        )
        #Add the FormulaCard to the sections in a defined location
        sections.append(formula_card)

    return AI_Tutor_Output(sections)
```

*   **FormulaCard:**  A dedicated `FormulaCard` class is created to hold the formula and its associated information.
*   **TableCard:** A `TableCard` class to hold the table data and headers.
*   **Structured Output:** The `AI_Tutor_Output` class holds a list of `Section` objects, which can then be iterated over to generate the UI.
*   **Data Storage:**  The `formula_data` list stores the extracted formulas in a well-defined format, ready for use in the "Formula Lab."
