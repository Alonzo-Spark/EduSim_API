        function toggleExplanation(id) {
            const el = document.getElementById(id);
            if (el.classList.contains('active')) {
                el.classList.remove('active');
            } else {
                // Close others
                document.querySelectorAll('.formula-explanation').forEach(div => div.classList.remove('active'));
                el.classList.add('active');
            }
        }

        function openLab(title, desc) {
            const modal = document.getElementById('lab-modal');
            document.getElementById('modal-title').innerText = title;
            document.getElementById('modal-desc').innerText = desc;
            modal.classList.add('open');
        }

        function closeLab() {
            document.getElementById('lab-modal').classList.remove('open');
        }

        // Close modal on outside click
        document.getElementById('lab-modal').addEventListener('click', (e) => {
            if (e.target.id === 'lab-modal') closeLab();
        });

        // Quiz Logic
        document.querySelectorAll('.quiz-option').forEach(option => {
            option.addEventListener('click', function() {
                const group = this.closest('.quiz-group');
                const feedbackId = group.nextElementSibling.id;
                const feedbackEl = document.getElementById(feedbackId);
                const correctVal = group.getAttribute('data-correct');
                const selectedVal = this.querySelector('input').value;

                // Visual selection
                group.querySelectorAll('.quiz-option').forEach(opt => opt.classList.remove('selected'));
                this.classList.add('selected');

                // Validation
                feedbackEl.className = 'feedback'; // reset
                if (selectedVal === correctVal) {
                    feedbackEl.innerText = "Correct! Well done.";
                    feedbackEl.classList.add('correct');
                } else {
                    feedbackEl.innerText = "Incorrect. Review the definitions and try again.";
                    feedbackEl.classList.add('incorrect');
                }
            });
