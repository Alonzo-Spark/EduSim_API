        // ==========================================
        // 1. KaTeX Initialization
        // ==========================================
        document.addEventListener("DOMContentLoaded", function() {
            renderMathInElement(document.body, {
                delimiters: [
                    {left: "$$", right: "$$", display: true},

            });

            // Initialize Formula Extraction
            extractFormulas();

        // ==========================================
        // 2. Formula Extraction System
        // ==========================================
        let extractedFormulaData = {};

        function extractFormulas() {
            // In a real app, this would parse complex markdown. 
            // Here, we extract from our structured UI block.
            const formulaCard = document.getElementById('mainFormulaCard');
            
            // We retrieve the raw LaTeX from a data attribute or by parsing the text content
            // For this robust demo, we define the context based on the DOM structure.
            const formulaContent = "\\Delta L = \\alpha L_0 \\Delta T";
            
            extractedFormulaData = {
                formula: formulaContent,
                title: "Linear Expansion Formula",
                variables: {
                    "ΔL": "Change in Length",
                    "α": "Coefficient of Linear Expansion",
                    "L₀": "Initial Length",
                    "ΔT": "Change in Temperature"
                },
                surroundingContext: "The change in length is proportional to original length and temperature change."
            };

            console.log("Formula Extracted:", extractedFormulaData);
        }

        // ==========================================
        // 3. Interactive Modal System
        // ==========================================
        function openExplanation() {
            document.getElementById('explainModal').classList.add('active');
        }

        function openFormulaLab() {
            document.getElementById('labModal').classList.add('active');
            calculateLab(); // Initial calculation
        }

        function closeModals() {
            document.querySelectorAll('.modal-overlay').forEach(el => el.classList.remove('active'));
        }

        // Close modal on click outside
        window.onclick = function(event) {
            if (event.target.classList.contains('modal-overlay')) {
                closeModals();
            }
        }

        // ==========================================
        // 4. Formula Lab Logic
        // ==========================================
        function calculateLab() {
            const alpha = parseFloat(document.getElementById('labMaterial').value);
            const L0 = parseFloat(document.getElementById('labL0').value);
            const DT = parseFloat(document.getElementById('labDT').value);

            // Calculate Delta L: ΔL = α * L0 * ΔT
            const deltaL = alpha * L0 * DT;
            
            // Update Text Result
            const resultDiv = document.getElementById('labResult');
            resultDiv.innerHTML = `Expansion ($\\Delta L$): ${deltaL.toFixed(4)} m (${(deltaL*100).toFixed(2)} cm)`;
            
            // Re-render math inside the result if needed (simple text here is fine, but if we used latex...)
            renderMathInElement(resultDiv);

            // Visual Update (Exaggerated for visibility)
            // We assume 100px in the visualizer represents L0
            // We want to visualize the *expansion* part exaggeratedly so the user sees it grow.
            const visualScale = 2000; // Exaggeration factor for the animation
            const baseWidth = 100; 
            const expansionPx = deltaL * visualScale; 
            
            const rod = document.getElementById('labRod');
            const newWidth = baseWidth + expansionPx;
            
            // Limit max width to prevent breaking layout
            rod.style.width = `${Math.min(newWidth, 350)}px`;
            
            // Update label
            document.getElementById('rodLabel').innerText = `${L0}m + ${(deltaL*100).toFixed(2)}cm`;
        }

        // ==========================================
        // 5. Microscopic Atom Simulation
        // ==========================================
        function initAtomSimulation() {
            const container = document.getElementById('atomBox');
            const slider = document.getElementById('heatSlider');
            const label = document.getElementById('heatLabel');
            
            // Create atoms
            const numAtoms = 8;
            const atoms = [];
            
            for(let i=0; i<numAtoms; i++) {
                const atom = document.createElement('div');
                atom.className = 'atom';
                // Distribute horizontally centered
                const leftPos = 20 + (i * (60 / (numAtoms-1))); 
                atom.style.left = `${leftPos}%`;
                atom.style.top = '50%';
                
                // Store initial position
                atom.dataset.baseLeft = leftPos;
                atom.dataset.baseTop = 50;
                
                container.appendChild(atom);
                atoms.push(atom);
            }

            slider.addEventListener('input', (e) => {
                const heat = parseInt(e.target.value);
                
                // Update Label
                if(heat < 4) label.innerText = "Low";
                else if(heat < 8) label.innerText = "Medium";
                else label.innerText = "High";

                // Animate Atoms
                atoms.forEach((atom, index) => {
                    // Vibration amplitude increases with heat
                    const amplitude = heat * 1.5; 
                    
                    // Random jitter
                    const randomX = (Math.random() - 0.5) * amplitude;
                    const randomY = (Math.random() - 0.5) * amplitude;
                    
                    // Expansion: Atoms push slightly apart
                    const expansion = (heat - 1) * 0.5; // Slight shift right
                    
                    atom.style.transform = `translate(${randomX + expansion}px, ${randomY}px)`;
                    
                    // Color shift from green (cool) to red (hot)
                    const r = Math.min(255, 16 + (heat * 25));
                    const g = Math.max(0, 185 - (heat * 20));
                    const b = Math.max(0, 129 - (heat * 15));
                    atom.style.backgroundColor = `rgb(${r},${g},${b})`;
                    atom.style.boxShadow = `0 0 ${5 + heat}px rgb(${r},${g},${b})`;
                });
            });
        }
    </script>
