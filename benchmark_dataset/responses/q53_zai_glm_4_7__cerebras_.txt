    <script>
        // ==========================================
        // 1. RENDERING SYSTEM (KaTeX)
        // ==========================================
        document.addEventListener("DOMContentLoaded", function() {
            // Render all KaTeX formulas
            if (typeof katex !== "undefined") {
                const formulas = document.querySelectorAll(".formula-display, [style*='overflow-x: auto']");
                formulas.forEach(el => {
                    katex.render(el.textContent.trim(), el, {
                        throwOnError: false,
                        displayMode: true
                    });
                });
            }
        });

        // ==========================================
        // 2. INTERACTION LOGIC (Toast)
        // ==========================================
        function showExplanation() {
            showToast(
                "Formula Explained", 
                "The period $T$ depends on mass and stiffness. Since $k$ is in the denominator under the square root, increasing $k$ decreases $T$. The mass acts as inertia (resisting change), while stiffness acts as the driving force."
            );
        }

        function openFormulaLab() {
            showToast(
                "Formula Lab", 
                "In a full lab environment, you would be able to rearrange this formula to solve for Mass ($m$) or Stiffness ($k$) given a specific Frequency. For now, use the Live Sandbox to experiment."
            );
        }

        function showToast(title, msg) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-title').innerText = title;
            document.getElementById('toast-msg').innerHTML = msg; // innerHTML to allow simple latex if needed, though text is safer
            toast.classList.add('active');
            
            // Auto hide after 4 seconds
            setTimeout(() => {
                toast.classList.remove('active');
            }, 4000);
        }

        // ==========================================
        // 3. PHYSICS SIMULATION (The Sandbox)
        // ==========================================
        
        // Physics State
        const state = {
            k: 50,      // Stiffness
            m: 2.0,     // Mass
            c: 0.1,     // Damping
            y: 100,     // Position (displacement from rest)
            v: 0,       // Velocity
            restY: 150, // Pixel position of equilibrium
            t: 0,       // Time accumulator
            history: [] // For graph
        };

        // DOM Elements
        const canvas = document.getElementById('simCanvas');
        const ctx = canvas.getContext('2d');
        const graphCanvas = document.getElementById('graphCanvas');
        const graphCtx = graphCanvas.getContext('2d');
        
        const kSlider = document.getElementById('k-slider');
        const mSlider = document.getElementById('m-slider');
        const cSlider = document.getElementById('c-slider');
        const kVal = document.getElementById('k-val');
        const mVal = document.getElementById('m-val');
        const cVal = document.getElementById('c-val');
        const resetBtn = document.getElementById('reset-btn');

        // Resize Canvas
        function resize() {
            canvas.width = canvas.parentElement.offsetWidth;
            canvas.height = canvas.parentElement.offsetHeight;
            graphCanvas.width = graphCanvas.parentElement.offsetWidth;
            graphCanvas.height = graphCanvas.parentElement.offsetHeight;
            state.restY = canvas.height / 2;
        }
        window.addEventListener('resize', resize);
        resize();

        // Update Physics Parameters
        kSlider.addEventListener('input', (e) => {
            state.k = parseFloat(e.target.value);
            kVal.innerText = state.k + " N/m";
        });

        mSlider.addEventListener('input', (e) => {
            state.m = parseFloat(e.target.value);
            mVal.innerText = state.m.toFixed(1) + " kg";
        });

        cSlider.addEventListener('input', (e) => {
            state.c = parseFloat(e.target.value);
            cVal.innerText = state.c.toFixed(2);
        });

        resetBtn.addEventListener('click', () => {
            state.y = 100; // Pull down
            state.v = 0;
            state.history = [];
        });

        // Physics Loop
        function updatePhysics() {
            // F_spring = -k * y
            // F_damping = -c * v
            // a = F / m
            
            const F_spring = -state.k * state.y;
            const F_damping = -state.c * state.v * 5; // Multiplier to make damping slider feel right
            
            const a = (F_spring + F_damping) / state.m;
            
            state.v += a * 0.16; // dt approximation
            state.y += state.v * 0.16;
            
            state.t += 0.016;

            // Store history for graph
            state.history.push(state.y);
            if (state.history.length > graphCanvas.width / 2) {
                state.history.shift();
            }
        }

        // Drawing Loop
        function draw() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            const centerX = canvas.width / 2;
            const anchorY = 20;
            const massY = state.restY + state.y;
            const massRadius = 20 + (state.m * 2); // Visual size based on mass

            // Draw Ceiling
            ctx.fillStyle = "#334155";
            ctx.fillRect(centerX - 40, 0, 80, 10);

            // Draw Spring
            ctx.beginPath();
            ctx.moveTo(centerX, anchorY);
            
            const segments = 12;
            const springLength = massY - anchorY - massRadius;
            const segmentHeight = springLength / segments;
            
            for (let i = 1; i <= segments; i++) {
                const x = (i % 2 === 0) ? centerX - 15 : centerX + 15;
                const y = anchorY + (i * segmentHeight);
                // Taper the ends
                const currentX = (i === segments) ? centerX : x;
                ctx.lineTo(currentX, y);
            }
            ctx.lineTo(centerX, massY);
            
            ctx.strokeStyle = "#94a3b8";
            ctx.lineWidth = 3;
            ctx.lineJoin = "round";
            ctx.stroke();

            // Draw Mass
            ctx.beginPath();
            ctx.arc(centerX, massY, massRadius, 0, Math.PI * 2);
            ctx.fillStyle = "#2563eb";
            ctx.fill();
            ctx.strokeStyle = "#1e40af";
            ctx.lineWidth = 2;
            ctx.stroke();

            // Draw Equilibrium Line (Ghost)
            ctx.beginPath();
            ctx.moveTo(centerX - 60, state.restY);
            ctx.lineTo(centerX + 60, state.restY);
            ctx.strokeStyle = "rgba(255, 255, 255, 0.3)";
            ctx.setLineDash([5, 5]);
            ctx.stroke();
            ctx.setLineDash([]);
        }

        function drawGraph() {
            graphCtx.clearRect(0, 0, graphCanvas.width, graphCanvas.height);
            
            const centerY = graphCanvas.height / 2;
            
            graphCtx.beginPath();
            graphCtx.moveTo(0, centerY + state.history[0]);
            
            for (let i = 1; i < state.history.length; i++) {
                // Scale Y for visibility in graph
                const graphY = centerY + state.history[i];
                graphCtx.lineTo(i * 2, graphY);
            }
            
            graphCtx.strokeStyle = "#f59e0b";
            graphCtx.lineWidth = 2;
            graphCtx.stroke();

            // Center line
            graphCtx.beginPath();
            graphCtx.moveTo(0, centerY);
            graphCtx.lineTo(graphCanvas.width, centerY);
            graphCtx.strokeStyle = "#e2e8f0";
            graphCtx.stroke();
        }

        function loop() {
            updatePhysics();
            draw();
            drawGraph();
            requestAnimationFrame(loop);
        }

        // Start Simulation
        loop();

    </script>
