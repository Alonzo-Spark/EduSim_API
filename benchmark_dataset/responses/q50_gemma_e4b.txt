        # --- Section Detection ---
        if line.startswith("#"):
            section_title = line[1:].strip()
            if not section_stack:
                #Start a new section
                section_stack.append({"type": "section", "title": section_title, "content": ""})
            else:
                section_stack[-1]["content"] += line + "\n"

        # --- Regular Text ---
        elif line:

        elif line.startswith("|"):  #Basic table detection

            html_output += table_html + "\n"

    # --- Final Section Rendering ---
    for section in section_stack:
        if section["type"] == "section":
            html_output += generate_section_card(section)


    return html_output
