    <script>
        // Canvas setup
        const canvas = document.getElementById('orbitCanvas');
        const ctx = canvas.getContext('2d');
        const width = canvas.width;
        const height = canvas.height;
        
        // Simulation parameters
        let animationId = null;
        let simulationRunning = false;
        let planet = {
            x: width/2 + 200,
            y: height/2,
            vx: 0,
            vy: 0,
            radius: 12,
            mass: 1.0,
            trail: []
        };
        
        const star = {
            x: width/2,
            y: height/2,
            radius: 30,
            mass: 1000
        };
        
        // Physics constants
        const G = 6.67430e-11; // Gravitational constant
        const scale = 1e9; // Scale for visualization
        const timeStep = 0.05;
        
        // Control elements
        const velocitySlider = document.getElementById('velocitySlider');
        const velocityValue = document.getElementById('velocityValue');
        const massSlider = document.getElementById('massSlider');
        const massValue = document.getElementById('massValue');
        const distanceSlider = document.getElementById('distanceSlider');
        const distanceValue = document.getElementById('distanceValue');
        const startBtn = document.getElementById('startBtn');
        const resetBtn = document.getElementById('resetBtn');
        const stableBtn = document.getElementById('stableBtn');
        const decayBtn = document.getElementById('decayBtn');
        
        // Info displays
        const orbitType = document.getElementById('orbitType');
        const orbitPeriod = document.getElementById('orbitPeriod');
        const currentVel = document.getElementById('currentVel');
        const stableVel = document.getElementById('stableVel');
        const kinetic = document.getElementById('kinetic');
        const potential = document.getElementById('potential');
        const totalEnergy = document.getElementById('totalEnergy');
        const energyState = document.getElementById('energyState');
        
        // Initialize simulation
        function initSimulation() {
            // Set initial distance
            const distance = parseInt(distanceSlider.value) / 100;
            planet.x = star.x + distance * 100;
            planet.y = star.y;
            
            // Set initial velocity
            const velocityPercent = parseInt(velocitySlider.value);
            updateVelocity(velocityPercent);
            
            // Reset trail
            planet.trail = [];
            for (let i = 0; i < 5; i++) {
                planet.trail.push({x: planet.x, y: planet.y});
            }
            
            updateDisplays();
        }
        
        // Calculate stable circular velocity
        function calculateStableVelocity() {
            const distance = Math.sqrt((planet.x - star.x)**2 + (planet.y - star.y)**2);
            return Math.sqrt(G * star.mass * planet.mass / (distance / scale)) * scale;
        }
        
        // Update planet velocity based on slider percentage
        function updateVelocity(percent) {
            const stableSpeed = calculateStableVelocity();
            const speed = stableSpeed * percent / 100;
            
            // Direction perpendicular to radius vector
            const dx = planet.x - star.x;
            const dy = planet.y - star.y;
            const distance = Math.sqrt(dx*dx + dy*dy);
            
            // Normalize and rotate 90 degrees for tangential direction
            const tx = -dy / distance;
            const ty = dx / distance;
            
            planet.vx = tx * speed;
            planet.vy = ty * speed;
        }
        
        // Update simulation
        function update() {
            // Calculate distance to star
            const dx = planet.x - star.x;
            const dy = planet.y - star.y;
            const distance = Math.sqrt(dx*dx + dy*dy);
            
            // Calculate gravitational force
            const F = G * star.mass * planet.mass / (distance/scale)**2;
            const Fx = -F * dx / distance * scale;
            const Fy = -F * dy / distance * scale;
            
            // Update velocity (F = ma => a = F/m)
            planet.vx += Fx / planet.mass * timeStep;
            planet.vy += Fy / planet.mass * timeStep;
            
            // Update position
            planet.x += planet.vx * timeStep;
            planet.y += planet.vy * timeStep;
            
            // Add current position to trail
            planet.trail.push({x: planet.x, y: planet.y});
            if (planet.trail.length > 500) {
                planet.trail.shift();
            }
            
            // Update displays
            updateDisplays();
        }
        
        // Update information displays
        function updateDisplays() {
            const distance = Math.sqrt((planet.x - star.x)**2 + (planet.y - star.y)**2) / 100;
            const velocityPercent = parseInt(velocitySlider.value);
            const stableSpeed = calculateStableVelocity();
            const currentSpeed = Math.sqrt(planet.vx**2 + planet.vy**2);
            
            // Update control displays
            velocityValue.textContent = velocityPercent + '%';
            massValue.textContent = (planet.mass).toFixed(1);
            distanceValue.textContent = (parseInt(distanceSlider.value)/100).toFixed(1);
            
            // Update info displays
            currentVel.textContent = velocityPercent + '%';
            stableVel.textContent = (stableSpeed / 100).toFixed(2) + ' AU/year';
            
            // Determine orbit type
            const eccentricity = Math.abs(velocityPercent - 100) / 100;
            if (eccentricity < 0.05) {
                orbitType.textContent = "Nearly Circular";
            } else if (eccentricity < 0.3) {
                orbitType.textContent = "Slightly Elliptical";
            } else if (velocityPercent > 100) {
                orbitType.textContent = "Elliptical (High)";
            } else if (velocityPercent < 70) {
                orbitType.textContent = "Decaying Orbit";
            } else {
                orbitType.textContent = "Elliptical (Low)";
            }
            
            // Update energy displays
            const kineticVal = 0.5 * planet.mass * currentSpeed**2;
            const potentialVal = -G * star.mass * planet.mass / (distance/scale) * scale;
            const total = kineticVal + potentialVal;
            
            kinetic.textContent = kineticVal.toFixed(0);
            potential.textContent = potentialVal.toFixed(0);
            totalEnergy.textContent = total.toFixed(0);
            energyState.textContent = total < 0 ? "Bound" : "Unbound";
        }
        
        // Draw simulation
        function draw() {
            // Clear canvas
            ctx.fillStyle = '#001220';
            ctx.fillRect(0, 0, width, height);
            
            // Draw star
            const starGradient = ctx.createRadialGradient(
                star.x, star.y, 0, 
                star.x, star.y, star.radius
            );
            starGradient.addColorStop(0, '#ffff00');
            starGradient.addColorStop(1, '#ff9800');
            
            ctx.beginPath();
            ctx.arc(star.x, star.y, star.radius, 0, Math.PI * 2);
            ctx.fillStyle = starGradient;
            ctx.fill();
            
            // Draw glow effect
            const glowGradient = ctx.createRadialGradient(
                star.x, star.y, star.radius,
                star.x, star.y, star.radius * 4
            );
            glowGradient.addColorStop(0, 'rgba(255, 152, 0, 0.6)');
            glowGradient.addColorStop(1, 'rgba(255, 152, 0, 0)');
            
            ctx.beginPath();
            ctx.arc(star.x, star.y, star.radius * 4, 0, Math.PI * 2);
            ctx.fillStyle = glowGradient;
            ctx.fill();
            
            // Draw orbit trail
            if (planet.trail.length > 2) {
                ctx.beginPath();
                ctx.moveTo(planet.trail[0].x, planet.trail[0].y);
                
                for (let i = 1; i < planet.trail.length; i++) {
                    ctx.lineTo(planet.trail[i].x, planet.trail[i].y);
                }
                
                const velocityPercent = parseInt(velocitySlider.value);
                if (velocityPercent < 70) {
                    ctx.strokeStyle = '#ff3d00'; // Red for decaying orbits
                } else if (Math.abs(velocityPercent - 100) < 5) {
                    ctx.strokeStyle = '#00c853'; // Green for stable orbits
                } else {
                    ctx.strokeStyle = '#40e0d0'; // Turquoise for elliptical orbits
                }
                
                ctx.lineWidth = 2;
                ctx.stroke();
            }
            
            // Draw planet
            const planetGradient = ctx.createRadialGradient(
                planet.x, planet.y, 0,
                planet.x, planet.y, planet.radius
            );
            planetGradient.addColorStop(0, '#4fc3f7');
            planetGradient.addColorStop(1, '#01579b');
            
            ctx.beginPath();
            ctx.arc(planet.x, planet.y, planet.radius, 0, Math.PI * 2);
            ctx.fillStyle = planetGradient;
            ctx.fill();
            
            // Draw velocity vector
            const speed = Math.sqrt(planet.vx**2 + planet.vy**2);
            if (speed > 0) {
                const vx = planet.vx / speed * 30;
                const vy = planet.vy / speed * 30;
                
                ctx.beginPath();
                ctx.moveTo(planet.x, planet.y);
                ctx.lineTo(planet.x + vx, planet.y + vy);
                ctx.strokeStyle = '#00e5ff';
                ctx.lineWidth = 2;
                ctx.stroke();
                
                // Draw arrowhead
                ctx.beginPath();
                ctx.moveTo(planet.x + vx, planet.y + vy);
                ctx.lineTo(planet.x + vx*0.8 - vy*0.2, planet.y + vy*0.8 + vx*0.2);
                ctx.lineTo(planet.x + vx*0.8 + vy*0.2, planet.y + vy*0.8 - vx*0.2);
                ctx.closePath();
                ctx.fillStyle = '#00e5ff';
                ctx.fill();
            }
            
            // Draw distance indicator
            const dist = Math.sqrt((planet.x - star.x)**2 + (planet.y - star.y)**2);
            ctx.beginPath();
            ctx.moveTo(star.x, star.y);
            ctx.lineTo(planet.x, planet.y);
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.3)';
            ctx.lineWidth = 1;
            ctx.stroke();
            
            // Draw distance text
            ctx.font = '14px Arial';
            ctx.fillStyle = 'rgba(255, 255, 255, 0.7)';
            ctx.fillText(`Distance: ${(dist/100).toFixed(2)} AU`, (star.x + planet.x)/2, (star.y + planet.y)/2 - 10);
        }
        
        // Animation loop
        function animate() {
            update();
            draw();
            
            if (simulationRunning) {
                animationId = requestAnimationFrame(animate);
            }
        }
        
        // Event listeners
        velocitySlider.addEventListener('input', () => {
            updateVelocity(parseInt(velocitySlider.value));
            updateDisplays();
        });
        
        massSlider.addEventListener('input', () => {
            planet.mass = parseInt(massSlider.value) / 100;
            updateVelocity(parseInt(velocitySlider.value));
            updateDisplays();
        });
        
        distanceSlider.addEventListener('input', () => {
            const distance = parseInt(distanceSlider.value) / 100;
            const angle = Math.atan2(planet.y - star.y, planet.x - star.x);
            planet.x = star.x + Math.cos(angle) * distance * 100;
            planet.y = star.y + Math.sin(angle) * distance * 100;
            updateVelocity(parseInt(velocitySlider.value));
            updateDisplays();
        });
        
        startBtn.addEventListener('click', () => {
            if (simulationRunning) {
                simulationRunning = false;
                startBtn.innerHTML = '<i class="fas fa-play"></i> Start Simulation';
                cancelAnimationFrame(animationId);
            } else {
                simulationRunning = true;
                startBtn.innerHTML = '<i class="fas fa-pause"></i> Pause Simulation';
                animate();
            }
        });
        
        resetBtn.addEventListener('click', () => {
            simulationRunning = false;
            startBtn.innerHTML = '<i class="fas fa-play"></i> Start Simulation';
            cancelAnimationFrame(animationId);
            initSimulation();
            draw();
        });
        
        stableBtn.addEventListener('click', () => {
            velocitySlider.value = 100;
            updateVelocity(100);
            updateDisplays();
        });
        
        decayBtn.addEventListener('click', () => {
            velocitySlider.value = 65;
            updateVelocity(65);
            updateDisplays();
        });
        
        // Initialize
        initSimulation();

## Features of this Orbital Simulation

This interactive sandbox demonstrates orbital mechanics with a focus on what happens when tangential velocity is reduced below the stable circular speed:

1. **Real-time Physics Simulation**: Visualizes gravitational interactions between a planet and star
2. **Adjustable Parameters**:
   - Tangential velocity (% of stable orbit)
   - Planet mass
   - Orbital distance
3. **Visual Feedback**:
   - Orbit path color-coded based on stability
   - Velocity vector showing direction and magnitude
   - Distance indicator with measurement
4. **Dynamic Information Display**:
   - Current orbit classification
   - Energy calculations (kinetic, potential, total)
   - Stable velocity reference value
5. **Pre-set Scenarios**:
   - Set stable circular orbit
   - Simulate orbital decay
   - Start/pause/reset controls

When you reduce the tangential velocity below 100% (especially below 70%), you'll see the orbit become elliptical and eventually decay into a collision course with the star, demonstrating the consequences of insufficient orbital velocity.

The simulation uses real physics equations to model gravitational interactions, providing an accurate representation of orbital mechanics principles.
