    <script>
        // --- Interaction Logic ---

        // Toggle Variable Explanation
        function toggleExplanation() {
            const el = document.getElementById('variable-explanation');
            el.classList.toggle('visible');
        }

        // Modal Logic
        const modal = document.getElementById('labModal');
        
        function openLab() {
            modal.showModal();
            updateLab(); // Initialize values
        }

        function closeLab() {
            modal.close();
        }

        // Close modal if clicking outside
        modal.addEventListener('click', (event) => {
            if (event.target === modal) {
                modal.close();
            }
        });

        // Formula Lab Simulation Logic
        const rangeM1 = document.getElementById('range-m1');
        const rangeM2 = document.getElementById('range-m2');
        const rangeR = document.getElementById('range-r');
        
        const valM1 = document.getElementById('val-m1');
        const valM2 = document.getElementById('val-m2');
        const valR = document.getElementById('val-r');
        
        const resForce = document.getElementById('result-force');
        
        const obj1 = document.getElementById('obj1');
        const obj2 = document.getElementById('obj2');

        function updateLab() {
            // Get values
            const m1 = parseFloat(rangeM1.value);
            const m2 = parseFloat(rangeM2.value);
            const r = parseFloat(rangeR.value);

            // Update Text Labels
            valM1.textContent = m1;
            valM2.textContent = m2;
            valR.textContent = r;

            // Calculate Force (Using G=1 for simplicity in visualization)
            // F = (G * m1 * m2) / r^2
            const G = 1; 
            const F = (G * m1 * m2) / (r * r);
            
            // Update Result Display
            resForce.textContent = F.toFixed(2) + " N";

            // Update Visuals
            // 1. Size based on mass
            const baseSize = 30;
            const sizeScale = 1.5;
            obj1.style.width = (baseSize + m1 * sizeScale) + 'px';
            obj1.style.height = (baseSize + m1 * sizeScale) + 'px';
            obj1.textContent = m1 + 'kg';

            obj2.style.width = (baseSize + m2 * sizeScale) + 'px';
            obj2.style.height = (baseSize + m2 * sizeScale) + 'px';
            obj2.textContent = m2 + 'kg';

            // 2. Distance based on r
            // Container width is roughly 100% (let's say 500px max in modal)
            // We need to position them absolutely.
            // Center is 50%. Left object at 50% - r/2 * scale, Right object at 50% + r/2 * scale
            
            const containerWidth = document.querySelector('.lab-visual').offsetWidth;
            const visualScale = 30; // pixels per meter for visual
            
            const center = containerWidth / 2;
            const offset = (r * visualScale) / 2;

            // Clamp logic to prevent them from leaving the box or overlapping too much visually
            // We limit the max offset so they stay inside
            const maxOffset = (containerWidth / 2) - 40; 
            const safeOffset = Math.min(offset, maxOffset);

            obj1.style.left = (center - safeOffset - (obj1.offsetWidth/2)) + 'px';
            obj2.style.left = (center + safeOffset - (obj2.offsetWidth/2)) + 'px';
        }

        // Event Listeners for Sliders
        rangeM1.addEventListener('input', updateLab);
        rangeM2.addEventListener('input', updateLab);
        rangeR.addEventListener('input', updateLab);

        // Handle window resize for the visualizer
        window.addEventListener('resize', updateLab);

    </script>
