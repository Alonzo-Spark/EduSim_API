              <span style={{ color: colours.textMuted, fontSize: 13, lineHeight: 1.6 }}>{desc}</span>
            </div>
          ))}
        </div>
      </Card>

      {/* ── SUGGESTED QUESTIONS ── */}
      <SuggestedQuestions onAsk={(q) => setAsked(q)} />

      {/* Asked question toast */}
      {asked && (
        <div
          style={{
            position: "fixed",
            bottom: 24,
            left: "50%",
            transform: "translateX(-50%)",
            background: colours.surface,
            border: `1px solid ${colours.purple}60`,
            borderRadius: 12,
            padding: "14px 20px",
            maxWidth: 500,
            width: "90%",
            boxShadow: `0 8px 40px ${colours.purple}30`,
            display: "flex",
            gap: 12,
            alignItems: "center",
          }}
        >
          <span style={{ fontSize: 18 }}>💬</span>
          <div style={{ flex: 1 }}>
            <div style={{ color: colours.purple, fontSize: 11, fontWeight: 700, letterSpacing: "0.08em", textTransform: "uppercase", marginBottom: 2 }}>
              Question Selected
            </div>
            <div style={{ color: colours.text, fontSize: 13 }}>{asked}</div>
          </div>
          <button
            onClick={() => setAsked(null)}
            style={{ background: "none", border: "none", color: colours.textDim, cursor: "pointer", fontSize: 16 }}
          >
            ✕
          </button>
        </div>
      )}
    </div>
  );
}
