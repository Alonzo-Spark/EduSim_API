// Interactive functionality
document.querySelectorAll('.lab-btn').forEach(button => {

    const panel = button.nextElementSibling;
    panel.style.display = panel.style.display === 'block' ? 'none' : 'block';
  });
});

document.querySelector('.solution-btn').addEventListener('click', function() {
  const solution = this.nextElementSibling;
  solution.style.display = solution.style.display === 'block' ? 'none' : 'block';
  this.textContent = solution.style.display === 'block' ? 'Hide Solution' : 'Reveal Analysis';
});
