    <script src="https://cdn.jsdelivr.net/npm/katex@0.16.8/dist/katex.min.js"></script>

    <script>
        // Formula Database extracted during "rendering"
        const formulaDatabase = {
            1: {
                latex: "C = \\frac{Q}{\\Delta T}",
                title: "Heat Capacity",
                context: "Relates total heat to total temperature change of a specific object.",
                variables: [
                    { symbol: "C", meaning: "Heat Capacity (J/K)" },
                    { symbol: "Q", meaning: "Heat Energy Supplied (J)" },
                    { symbol: "ΔT", meaning: "Change in Temperature (K or °C)" }
                ]
            },
            2: {
                latex: "Q = mc\\Delta T",
                title: "Specific Heat Equation",
                context: "Used to calculate the heat energy absorbed or released by a substance of known mass.",
                variables: [
                    { symbol: "Q", meaning: "Heat Energy (Joules)" },
                    { symbol: "m", meaning: "Mass of substance (kg)" },
                    { symbol: "c", meaning: "Specific Heat Capacity (J/kg°C)" },
                    { symbol: "ΔT", meaning: "Temperature Change (°C)" }
                ]
            }
        };

        // Initialize KaTeX rendering
        document.addEventListener("DOMContentLoaded", () => {
            renderMath();
        });

        function renderMath() {
            const mathElements = document.querySelectorAll('.formula-display');
            mathElements.forEach(el => {
                katex.render(el.innerText.trim(), el, {
                    throwOnError: false,
                    displayMode: true
                });
            });
        }

        // Modal Logic
        function closeModal(modalId) {
            document.getElementById(modalId).classList.remove('active');
        }

        // Explanation Logic
        function explainFormula(id) {
            const data = formulaDatabase[id];
            const contentDiv = document.getElementById('explanationContent');
            
            let html = `
                <div class="formula-display" style="margin-bottom: 20px;">$$${data.latex}$$</div>
                <p style="margin-bottom: 15px;"><strong>Context:</strong> ${data.context}</p>
                <p><strong>Variables involved:</strong></p>
                <ul class="variable-list">
            `;

            data.variables.forEach(v => {
                html += `<li><span class="var-symbol">${v.symbol}</span> : ${v.meaning}</li>`;
            });

            html += `</ul>`;
            
            contentDiv.innerHTML = html;
            
            // Re-render KaTeX inside the modal
            katex.render(data.latex, contentDiv.querySelector('.formula-display'), {
                throwOnError: false,
                displayMode: true
            });

            document.getElementById('explanationModal').classList.add('active');
        }

        // Formula Lab Logic
        function openFormulaLab(id) {
            if(id !== 2) return; // Only implemented for the main formula in this demo
            // Reset inputs
            document.getElementById('lab-mass').value = '';
            document.getElementById('lab-shc').value = '';
            document.getElementById('lab-dt').value = '';
            document.getElementById('lab-result').innerText = '0 Joules';
            document.getElementById('labModal').classList.add('active');
        }

        function calculateLab() {
            const m = parseFloat(document.getElementById('lab-mass').value);
            const c = parseFloat(document.getElementById('lab-shc').value);
            const dt = parseFloat(document.getElementById('lab-dt').value);

            const resultEl = document.getElementById('lab-result');

            if (isNaN(m) || isNaN(c) || isNaN(dt)) {
                resultEl.innerText = '...';
                return;
            }

            const Q = m * c * dt;
            
            // Format number nicely
            let formattedQ = Q.toLocaleString(undefined, { maximumFractionDigits: 2 });
            resultEl.innerText = `${formattedQ} Joules`;
        }

        // Close modal when clicking outside
        window.onclick = function(event) {
            if (event.target.classList.contains('modal-overlay')) {
                event.target.classList.remove('active');
            }
        }
    </script>
</body>
