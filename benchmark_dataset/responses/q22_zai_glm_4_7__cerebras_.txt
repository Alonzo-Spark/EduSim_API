                <span class="card-title">Derivation Steps</span>

            <div class="card-body">
                <p>To derive the expression for $g$, we equate Newton's Law of Universal Gravitation with Newton's Second Law of Motion.</p>
                
                <ul>
                    <li>
                        <strong>Step 1: Gravitational Force</strong><br>
                        Consider a planet of mass $M$ and radius $R$. An object of mass $m$ is placed on the surface of the planet. According to Newton's Law of Gravitation, the force of attraction is:
                        $$ F = \frac{G M m}{R^2} $$
                    </li>
                    <li>
                        <strong>Step 2: Force acting on the object</strong><br>
                        This gravitational force is the weight of the object. According to Newton's Second Law ($F=ma$), the force acting on the object is the product of its mass and the acceleration due to gravity ($g$):
                        $$ F = mg $$
                    </li>
                    <li>
                        <strong>Step 3: Equating the Forces</strong><br>
                        Since both expressions represent the same force:
                        $$ mg = \frac{G M m}{R^2} $$
                    </li>
                    <li>
                        <strong>Step 4: Simplification</strong><br>
                        The mass of the object ($m$) appears on both sides and cancels out (demonstrating that $g$ is independent of the mass of the falling object):
                        $$ g = \frac{G M}{R^2} $$
                    </li>
                </ul>

        // 2. Modal Logic
        function openModal(modalId) {
            document.getElementById(modalId).classList.add('active');
        }

        function closeModal(modalId) {
            document.getElementById(modalId).classList.remove('active');
        }

        // Close modal when clicking outside content
        window.onclick = function(event) {
            if (event.target.classList.contains('modal-overlay')) {
                event.target.classList.remove('active');
            }
        }

        // 3. Formula Lab Logic
        const G = 6.67430e-11; // Gravitational Constant
        const EARTH_MASS = 5.972e24; // kg
        const EARTH_RADIUS = 6.371e6; // m

        const massRange = document.getElementById('massRange');
        const radiusRange = document.getElementById('radiusRange');
        const gValueDisplay = document.getElementById('gValue');
        const massDisplay = document.getElementById('massDisplay');
        const radiusDisplay = document.getElementById('radiusDisplay');
        const planetVisual = document.getElementById('planetVisual');

        function updateLab() {
            // Get multipliers
            const mMult = parseFloat(massRange.value);
            const rMult = parseFloat(radiusRange.value);

            // Calculate actual values
            const M = mMult * EARTH_MASS;
            const R = rMult * EARTH_RADIUS;

            // Calculate g = GM / R^2
            const g = (G * M) / (R * R);

            // Update UI
            gValueDisplay.textContent = g.toFixed(2);
            massDisplay.textContent = mMult.toFixed(1) + " Earths";
            radiusDisplay.textContent = rMult.toFixed(1) + " Earths";

            // Update Visual (Size changes with radius)
            // Base size 80px, scale by radius multiplier
            const visualSize = 60 * rMult; 
            planetVisual.style.width = `${visualSize}px`;
            planetVisual.style.height = `${visualSize}px`;
            
            // Change color intensity based on gravity (Redder = heavy, Bluer = light)
            // Base hue is roughly 220 (Blue). We'll shift towards red (0) as g increases.
            // Max g approx 10x Earth (M=10, R=0.5 -> 40g). Min g approx 0.01g.
            // Let's map 0g - 20g to a hue shift.
            const maxG = 20;
            const hue = 220 - Math.min((g / maxG) * 220, 220);
            planetVisual.style.background = `hsl(${hue}, 70%, 30%)`;
        }

        // Event Listeners
        massRange.addEventListener('input', updateLab);
        radiusRange.addEventListener('input', updateLab);

        // Initialize
        updateLab();

    </script>
