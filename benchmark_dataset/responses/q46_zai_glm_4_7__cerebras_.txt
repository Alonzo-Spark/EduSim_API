    <script>
        // 1. Render LaTeX using KaTeX
        document.addEventListener("DOMContentLoaded", function() {
            renderMathInElement(document.body, {
                delimiters: [
                    {left: "$$", right: "$$", display: true},
                    {left: "$", right: "$", display: false}
                ]
            });
        });

        // 2. Formula Card Interactivity
        
        // Toggle Explanation Panel
        function toggleExplanation() {
            const panel = document.getElementById('explanation-panel');
            const btn = event.target;
            
            if (panel.style.display === 'none') {
                panel.style.display = 'block';
                btn.textContent = 'Hide Explanation';
                btn.style.borderColor = 'var(--primary)';
                btn.style.color = 'var(--primary)';
            } else {
                panel.style.display = 'none';
                btn.textContent = 'Explain Formula';
                btn.style.borderColor = 'var(--border)';
                btn.style.color = 'var(--text-main)';
            }
        }

        // Toggle Lab Container
        function toggleLab() {
            const lab = document.getElementById('lab-container');
            const btn = event.target;
            
            if (lab.style.display === 'none') {
                lab.style.display = 'block';
                btn.textContent = 'Close Formula Lab';
                btn.classList.replace('btn-primary', 'btn-outline');
            } else {
                lab.style.display = 'none';
                btn.textContent = 'Open Formula Lab';
                btn.classList.replace('btn-outline', 'btn-primary');
            }
        }

        // 3. Formula Lab Logic (Ohm's Law Calculator)
        function calculateOhm() {
            // Get inputs
            const voltage = parseFloat(document.getElementById('input-voltage').value);
            const resistance = parseFloat(document.getElementById('input-resistance').value);

            // Update Labels
            document.getElementById('val-v').textContent = voltage;
            document.getElementById('val-r').textContent = resistance;

            // Calculate I = V / R
            let current = voltage / resistance;
            
            // Calculate Power P = VI
            let power = voltage * current;

            // Update Display
            document.getElementById('result-current').textContent = current.toFixed(2) + " A";
            document.getElementById('result-power').textContent = power.toFixed(2) + " W";
        }

        // Initialize calculator on load just in case
        calculateOhm();
    </script>
