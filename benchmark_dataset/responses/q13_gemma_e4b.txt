```python
import re
import pandas as pd
from typing import List, Dict, Union

class SectionCard:
    def __init__(self, title: str, content: str):
        self.title = title
        self.content = content

    def to_html(self) -> str:
        return f"""
        <div class="section-card">
            <h2>{self.title}</h2>
            <div class="content">{self.content}</div>
        </div>
        """

class FormulaCard:
    def __init__(self, formula: str, title: str, surrounding_context: str, variables: List[str] = []):
        self.formula = formula
        self.title = title
        self.surrounding_context = surrounding_context
        self.variables = variables

    def to_html(self) -> str:
        return f"""
        <div class="formula-card">
            <h3>{self.title}</h3>
            <div class="formula-content">
                {self.formula}
            </div>
            <div class="actions">
                <a href="#">Explain Formula</a> | <a href="#">Open Formula Lab</a>
            </div>
        </div>
        """

class TableCard:
    def __init__(self, data: List[List[str]], caption: str = ""):
        self.data = data
        self.caption = caption

    def to_html(self) -> str:
       df = pd.DataFrame(self.data)
       return f"""
       <div class="table-card">
           <table>
               {df.to_html(index=False)}
           </table>
           <p class="caption">{self.caption}</p>
       </div>
       """


class ContentProcessor:
    def __init__(self):
        pass

    def process_markdown(self, markdown_text: str) -> Union[str, List[SectionCard]]: #Return either HTML or list of SectionCards
        # 1. Section Detection (Simple Keyword Based)
        sections = self._detect_sections(markdown_text)

        # 2. Formula Extraction
        formulas = self._extract_formulas(markdown_text)

        # 3. Table Extraction
        tables = self._extract_tables(markdown_text)

        # 4. Construct UI Blocks
        cards = []
        for section in sections:
            cards.append(section)

        for formula in formulas:
            cards.append(FormulaCard(**formula))

        for table in tables:
            cards.append(TableCard(table))



        return cards # Return a list of section cards

    def _detect_sections(self, markdown_text: str) -> List[str]:
        #Simple detection based on headings
        sections = re.findall(r"^#+\s+(.+)", markdown_text, re.MULTILINE)
        return sections

    def _extract_formulas(self, markdown_text: str) -> List[Dict]:
        formulas = []
        for match in re.finditer(r"$$((?:[^\n]+)\n)+$$", markdown_text, re.DOTALL):
            formula_string = match.group(1).strip()
            title = "Formula"  # Default title, can be improved
            context = match.group(0) #The whole formula block
            variables = []
            #Simple variable extraction
            variables = re.findall(r"[a-zA-Z]+", formula_string)
            formulas.append({
                "formula": formula_string,
                "title": title,
                "surroundingContext": context,
                "variables": variables
            })
        return formulas

    def _extract_tables(self, markdown_text: str) -> List[List[str]]:
        #Find table markdown
        table_matches = re.findall(r"\|(.*)\|", markdown_text, re.DOTALL)
        tables = []
        for match in table_matches:
            #Split the table into rows
            rows = match.strip().split('\n')
            #Split each row into cells
            table_data = [cell.strip() for cell in rows[0].split('|')[1:-1]] #remove the first and last "|".
            tables.append(table_data)
        return tables
```
