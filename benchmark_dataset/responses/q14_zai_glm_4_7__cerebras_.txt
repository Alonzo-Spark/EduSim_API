        const rawMarkdown = `
## Introduction
Momentum is a fundamental concept in physics that describes the quantity of motion of a moving body. It is a vector quantity, meaning it has both magnitude and direction. In simple terms, momentum tells us how hard it is to stop a moving object.

## Quick Definition
Linear momentum ($$p$$) is defined as the product of an object's mass ($$m$$) and its velocity ($$v$$).

$$ p = m \\cdot v $$

This relationship shows that an object with a large mass or a high velocity has a large momentum.

## Key Concepts
*   **Vector Quantity:** Since velocity is a vector, momentum is also a vector. The direction of momentum is the same as the direction of the velocity.
*   **SI Unit:** The unit of momentum is kilogram-meter per second (kg·m/s).
*   **Newton's Second Law:** Momentum is closely tied to force. Force is the rate of change of momentum.

$$ \\vec{F} = \\frac{d\\vec{p}}{dt} $$

## Derivation of Conservation Law
To derive the Law of Conservation of Momentum, let us consider a collision between two isolated objects, Object A and Object B. There are no external forces acting on the system.

### Step 1: Newton's Third Law
According to Newton's Third Law of Motion, when two objects interact, they exert equal and opposite forces on each other. Let the force exerted by A on B be $$F_{AB}$$ and the force exerted by B on A be $$F_{BA}$$.

$$ F_{AB} = -F_{BA} $$

### Step 2: Impulse and Momentum Change
Force is equal to the rate of change of momentum. Over a time interval $$\\Delta t$$, the impulse equals the change in momentum.

For Object A:
$$ F_{BA} \\Delta t = m_A \\Delta v_A $$

For Object B:
$$ F_{AB} \\Delta t = m_B \\Delta v_B $$

### Step 3: Equating the Changes
Substitute the force relationship from Step 1 into the equations in Step 2.

$$ m_A \\Delta v_A = - m_B \\Delta v_B $$

Rearranging this, we get:

$$ m_A (v_{Af} - v_{Ai}) + m_B (v_{Bf} - v_{Bi}) = 0 $$

This simplifies to the total momentum before the collision equals the total momentum after the collision.

$$ m_A v_{Ai} + m_B v_{Bi} = m_A v_{Af} + m_B v_{Bf} $$

## Comparison of Collisions
The conservation of momentum applies to all collisions, but kinetic energy behaves differently.

| Type | Description | Momentum Conserved? | Kinetic Energy Conserved? |
| :--- | :--- | :---: | :---: |
| **Elastic** | Objects bounce off perfectly. No energy lost to heat/deformation. | Yes | Yes |
| **Inelastic** | Objects stick together or deform. Energy is lost. | Yes | No |
| **Perfectly Inelastic** | Objects stick together and move as one. | Yes | No |

## Summary
*   Momentum is mass times velocity ($$p=mv$$).
*   It is a vector quantity.
*   The Law of Conservation of Momentum states that the total momentum of an isolated system remains constant if no external force acts on it.
*   This law is derived directly from Newton's Third Law.

## Suggested Questions
1.  A 1000 kg car moving at 20 m/s hits a stationary 2000 kg truck. If they stick together, what is their final velocity?
2.  Why is a heavy truck harder to stop than a small car moving at the same speed?
3.  Does momentum conserve if a strong wind blows during the collision? Why or why not?
`;

        // ==========================================
        // STATE MANAGEMENT
        // ==========================================
        const formulaRegistry = [];
        const tocRegistry = [];

        // ==========================================
        // PARSING ENGINE
        // ==========================================
        function parseContent(markdown) {
            const container = document.createElement('div');
            
            // Normalize line breaks
            const lines = markdown.split('\n');
            let currentSection = null;
            let currentCard = null;
            let buffer = [];
            
            // Helper to flush text buffer
            const flushBuffer = () => {
                if (buffer.length > 0) {
                    const p = document.createElement('div');
                    p.innerHTML = buffer.join('<br>').replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
                    if (currentCard) currentCard.appendChild(p);
                    buffer = [];
                }
            };

            // Helper to create a new section card
            const createSection = (titleText) => {
                const id = titleText.toLowerCase().replace(/[^a-z0-9]+/g, '-');
                
                // Add to TOC
                tocRegistry.push({ title: titleText, id: id });

                // Create Card DOM
                const card = document.createElement('div');
                card.className = 'card';
                card.id = id;
                
                const header = document.createElement('div');
                header.className = 'section-header';
                
                // Determine icon based on keywords
                let iconLetter = titleText.charAt(0);
                if(titleText.includes('Formula')) iconLetter = 'ƒ';
                if(titleText.includes('Summary')) iconLetter = 'S';
                if(titleText.includes('Question')) iconLetter = '?';
                if(titleText.includes('Table') || titleText.includes('Comparison')) iconLetter = '▤';

                header.innerHTML = `
                    <div class="section-icon">${iconLetter}</div>
                    <div class="section-title">${titleText}</div>
                `;
                
                card.appendChild(header);
                container.appendChild(card);
                
                currentCard = card;
                currentSection = titleText;
            };

            lines.forEach(line => {
                const trimmed = line.trim();

                // 1. Detect Headers (##)
                if (trimmed.startsWith('## ')) {
                    flushBuffer();
                    createSection(trimmed.replace('## ', ''));
                    return;
                }

                // 2. Detect Display Math ($$ ... $$)
                if (trimmed.startsWith('$$') && trimmed.endsWith('$$')) {
                    flushBuffer();
                    
                    const latex = trimmed.slice(2, -2).trim();
                    
                    // Extract context (previous text or section title)
                    const context = currentSection || "Introduction";
                    
                    // Generate ID
                    const fid = `formula-${formulaRegistry.length}`;

                    // Store in Registry
                    formulaRegistry.push({
                        id: fid,
                        formula: latex,
                        title: `Formula in ${context}`,
                        context: context,
                        variables: extractVariables(latex) // Mock extraction
                    });

                    // Create Formula Card Component
                    const fCard = document.createElement('div');
                    fCard.className = 'formula-card';
                    fCard.id = fid;
                    
                    fCard.innerHTML = `
                        <div class="formula-render" data-latex="${latex}"></div>
                        <div class="formula-actions">
                            <button class="btn-text" onclick="toggleExplanation('${fid}')">Explain Formula</button>
                            <button class="btn-text" onclick="openLabToFormula('${fid}')">Open in Lab</button>
                        </div>
                        <div class="formula-explanation" id="explain-${fid}">
                            <strong>Context:</strong> This formula is found in the <em>${context}</em> section.<br>
                            It describes the relationship between the physical quantities involved.
                        </div>
                    `;
                    
                    if (currentCard) currentCard.appendChild(fCard);
                    return;
                }

                // 3. Detect Table
                if (trimmed.includes('|---')) {
                    // Start of table detected, we need to grab previous lines too
                    // But for simplicity in this parser, we assume table is contiguous block
                    // We will handle table parsing specifically.
                    // Let's collect table lines until empty line
                    let tableLines = [];
                    // We actually need to look back. 
                    // Simplified: If buffer has pipe lines, treat buffer as table.
                    if (buffer.length > 0 && buffer[0].includes('|')) {
                        tableLines = [...buffer];
                        buffer = []; // Clear buffer
                        
                        // Read ahead until empty line
                        // (This loop is tricky inside forEach, skipping complex look-ahead for this demo 
                        // and assuming table ends with empty line or end of section)
                        
                        // Construct Table HTML
                        const tableHTML = renderSimpleMarkdownTable(tableLines);
                        const tableWrapper = document.createElement('div');
                        tableWrapper.className = 'card table-card';
                        tableWrapper.innerHTML = `<div class="table-responsive">${tableHTML}</div>`;
                        
                        if (currentCard) currentCard.appendChild(tableWrapper);
                        return;
                    }
                }

                // Regular Text
                if (trimmed.length > 0) {
                    buffer.push(trimmed);
                } else {
                    flushBuffer();
                }
            });

            flushBuffer();
            return container;
        }

        // Helper to render a markdown table from array of strings
        function renderSimpleMarkdownTable(lines) {
            let html = '<table>';
            lines.forEach((line, index) => {
                if (line.includes('---')) return; // Skip separator line
                const cells = line.split('|').filter(c => c.trim() !== ''); // Filter empty edges
                html += '<tr>';
                cells.forEach(cell => {
                    const tag = index === 0 ? 'th' : 'td';
                    html += `<${tag}>${cell.trim()}</${tag}>`;
                });
                html += '</tr>';
            });
            html += '</table>';
            return html;
        }

        // Mock variable extractor
        function extractVariables(latex) {
            const matches = latex.match(/[a-z]+/gi);
            return matches ? [...new Set(matches)] : [];
        }

        // ==========================================
        // RENDERING LOGIC
        // ==========================================
        function renderApp() {
            const renderer = document.getElementById('content-renderer');
            const tocList = document.getElementById('toc-list');
            
            // 1. Parse Markdown to DOM
            const dom = parseContent(rawMarkdown);
            renderer.appendChild(dom);

            // 2. Render Math (KaTeX)
            document.querySelectorAll('.formula-render').forEach(el => {
                katex.render(el.getAttribute('data-latex'), el, {
                    throwOnError: false,
                    displayMode: true
                });
            });

            // 3. Render TOC
            tocRegistry.forEach(item => {
                const li = document.createElement('li');
                li.className = 'toc-item';
                li.innerHTML = `<a href="#${item.id}" class="toc-link">${item.title}</a>`;
                tocList.appendChild(li);
            });

            // 4. Render Formula Lab
            const labList = document.getElementById('lab-list');
            formulaRegistry.forEach(item => {
                const el = document.createElement('div');
                el.className = 'lab-item';
                el.onclick = () => {
                    document.getElementById(item.id).scrollIntoView({ behavior: 'smooth', block: 'center' });
                    toggleFormulaLab();
                    // Highlight effect

                    card.style.transform = 'scale(1.02)';
                    setTimeout(() => card.style.transform = 'scale(1)', 300);
                };
                
                el.innerHTML = `
                    <div class="lab-item-formula" id="lab-render-${item.id}"></div>
                    <div class="lab-item-title">${item.title}</div>
                    <div class="lab-item-context">Context: ${item.context}</div>
                `;
                labList.appendChild(el);
                
                // Render mini math in lab
                katex.render(item.formula, document.getElementById(`lab-render-${item.id}`), {
                    throwOnError: false,
                    displayMode: true
                });
            });
        }

        // ==========================================
        // INTERACTIONS
        // ==========================================
        
        function toggleExplanation(id) {
            const el = document.getElementById(`explain-${id}`);
            if (el) {
                el.classList.toggle('visible');
            }
        }

        function toggleFormulaLab() {
            const overlay = document.getElementById('lab-overlay');
            const panel = document.getElementById('lab-panel');
            overlay.classList.toggle('open');
            panel.classList.toggle('open');
        }

        function openLabToFormula(id) {
            toggleFormulaLab();
            // Optional: Highlight specific item in lab
        }

        // Initialize
        document.addEventListener('DOMContentLoaded', renderApp);
