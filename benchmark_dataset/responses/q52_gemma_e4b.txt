```python
import re

def process_markdown(markdown_text):
  """
  Processes markdown text to extract sections, formulas, and tables.

  Args:
    markdown_text: The markdown text to process.

  Returns:
    A structured data representation of the content.
  """

  sections = []
  current_section = None

  for line in markdown_text.splitlines():
    line = line.strip()
    if line.startswith("##"):  # Example: Section Header
      if current_section:
        sections.append(current_section)
      current_section = {"title": line[2:].strip(), "content": []}
    elif line.startswith("$$"): #Formula
        match = re.search(r"$$([\s\S]*?)$$", line)
        if match:
            formula = match.group(1).strip()
            current_section["content"].append({"type": "formula", "value": formula})
    else:
      if current_section:
        current_section["content"].append({"type": "paragraph", "value": line})


  if current_section:
    sections.append(current_section)

  return sections


def generate_html(data):
  """
  Generates HTML from the structured data.
  """
  html = ""
  for section in data:
    html += f"<Card>\n"
    html += f"  <h2>{section['title']}</h2>\n"
    for item in section['content']:
      if item["type"] == "paragraph":
        html += f"  <p>{item['value']}</p>\n"
      elif item["type"] == "formula":
          html += f"  <FormulaCard>\n"
          html += f"    <h3>Main Formula</h3>\n"
          html += f"    {item['value']}\n"
          html += "    <a href='#'>Explain Formula</a>\n"
          html += "    <a href='#'>Open Formula Lab</a>\n"
          html += "  </FormulaCard>\n"

    html += f"</Card>\n"
  return html


# Example usage:
markdown = """
Live Sandbox: Explain the visual effect of adding air resistance to a projectile trajectory.
Air resistance is a force that opposes the motion of an object through a fluid (like air). When a projectile is launched, air resistance acts in the opposite direction to its velocity, causing several visual changes to its trajectory.
Live Sandbox: Explain the visual effect of adding air resistance to a projectile trajectory.
F<sub>d</sub> = 0.5 * ρ * v<sup>2</sup> * C<sub>d</sub> * A
Where:
ρ = Density of the air
v = Velocity of the projectile
C<sub>d</sub> = Drag coefficient (a dimensionless number)
A = Cross-sectional area of the projectile
"""

data = process_markdown(markdown)
html = generate_html(data)
print(html)
```
