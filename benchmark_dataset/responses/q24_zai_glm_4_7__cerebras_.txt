    <script>
        // Constants
        const G = 6.67430e-11; // m^3 kg^-1 s^-2

        // DOM Elements
        const resultSection = document.getElementById('result-section');
        const resultValueEl = document.getElementById('result-value');
        const resultSciEl = document.getElementById('result-sci');
        const formulaExplanation = document.getElementById('formula-explanation');
        const labModal = document.getElementById('lab-modal');
        const gValueDisplay = document.getElementById('g-value');

        // Initialize KaTeX
        document.addEventListener("DOMContentLoaded", function() {
            renderMathInElement(document.body, {
                delimiters: [
                    {left: "$$", right: "$$", display: true},
                    {left: "$", right: "$", display: false}
                ]
            });
        });

        // Toggle Formula Explanation
        function toggleExplanation() {
            if (formulaExplanation.style.display === 'none') {
                formulaExplanation.style.display = 'block';
                // Re-render math inside the newly visible section
                renderMathInElement(formulaExplanation, {
                    delimiters: [
                        {left: "$", right: "$", display: false}
                    ]
                });
            } else {
                formulaExplanation.style.display = 'none';
            }
        }

        // Calculation Logic
        function calculateForce(event) {
            event.preventDefault();

            // Get inputs
            const m1 = parseFloat(document.getElementById('mass1').value);
            const m2 = parseFloat(document.getElementById('mass2').value);
            const r = parseFloat(document.getElementById('distance').value);

            // Validation
            if (m1 <= 0 || m2 <= 0 || r <= 0) {
                alert("Mass and Distance must be positive numbers.");
                return;
            }

            // Calculate Force: F = G * (m1 * m2) / r^2
            const force = G * (m1 * m2) / (r * r);

            // Display Result
            resultSection.style.display = 'block';
            
            // Format decimal
            resultValueEl.textContent = force.toExponential(4); // e.g., 4.1719e-10
            
            // Format Scientific Notation text
            resultSciEl.textContent = `${force.toExponential(2)} N`;

            // Smooth scroll to result
            resultSection.scrollIntoView({ behavior: 'smooth', block: 'center' });
        }

        // Modal Logic
        function openLab() {
            labModal.classList.add('active');
        }

        function closeLab() {
            labModal.classList.remove('active');
        }

        // Close modal when clicking outside content
        labModal.addEventListener('click', (e) => {
            if (e.target === labModal) {
                closeLab();
            }
        });

        // Lab Interaction
        function updateLabDisplay(val) {
            gValueDisplay.textContent = val;
            // In a full app, this might update a graph or dynamic preview
        }
    </script>
