    <script>
        // --- Physics Engine & State ---
        const state = {
            mass: 1.0,      // kg
            length: 1.0,    // m
            gravity: 9.8,   // m/s^2
            angle: Math.PI / 4, // Initial angle (45 degrees)
            velocity: 0,
            acceleration: 0,
            damping: 0.998, // Air resistance approximation
            running: true,
            timeScale: 1.0
        };

        const canvas = document.getElementById('pendulumCanvas');
        const ctx = canvas.getContext('2d');
        let animationId;

        // --- DOM Elements ---
        const els = {
            mass: document.getElementById('input-mass'),
            length: document.getElementById('input-length'),
            gravity: document.getElementById('input-gravity'),
            valMass: document.getElementById('val-mass'),
            valLength: document.getElementById('val-length'),
            valGravity: document.getElementById('val-gravity'),
            periodDisplay: document.getElementById('display-period'),
            toggleBtn: document.getElementById('btn-toggle'),
            resetBtn: document.getElementById('btn-reset')
        };

        // --- Initialization ---
        function resizeCanvas() {
            // Handle high-DPI displays
            const dpr = window.devicePixelRatio || 1;
            const rect = canvas.getBoundingClientRect();
            
            canvas.width = rect.width * dpr;
            canvas.height = rect.height * dpr;
            
            ctx.scale(dpr, dpr);
            
            // Store logical size for calculations
            state.width = rect.width;
            state.height = rect.height;
        }

        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        // --- Physics Loop ---
        function updatePhysics() {
            if (!state.running) return;

            // Simple Pendulum Equation of Motion:
            // angular_acceleration = -(g / L) * sin(theta)
            
            // We perform Euler integration (simple but sufficient for this visual)
            // Note: We use a small time step approximation
            const dt = 0.016 * state.timeScale; // approx 60fps

            state.acceleration = (-1 * state.gravity / state.length) * Math.sin(state.angle);
            state.velocity += state.acceleration * dt;
            state.velocity *= state.damping; // Apply air resistance
            state.angle += state.velocity * dt;
        }

        function calculatePeriod() {
            // T = 2 * pi * sqrt(L / g)
            return 2 * Math.PI * Math.sqrt(state.length / state.gravity);
        }

        // --- Rendering ---
        function draw() {
            ctx.clearRect(0, 0, state.width, state.height);

            const originX = state.width / 2;
            const originY = 50;
            
            // Scale pixels per meter (dynamic based on canvas height)
            // Ensure the pendulum fits even at max length (3.0m)
            const maxLenPixels = state.height - 100;
            const pixelsPerMeter = maxLenPixels / 3.0; 

            const bobX = originX + Math.sin(state.angle) * (state.length * pixelsPerMeter);
            const bobY = originY + Math.cos(state.angle) * (state.length * pixelsPerMeter);

            // Draw Pivot
            ctx.beginPath();
            ctx.fillStyle = '#64748b';
            ctx.arc(originX, originY, 5, 0, Math.PI * 2);
            ctx.fill();

            // Draw String
            ctx.beginPath();
            ctx.moveTo(originX, originY);
            ctx.lineTo(bobX, bobY);
            ctx.strokeStyle = '#334155';
            ctx.lineWidth = 2;
            ctx.stroke();

            // Draw Bob (Size scales slightly with mass for visual feedback)
            // Base radius 15px, increases with mass
            const bobRadius = 15 + (state.mass * 5); 
            
            // Bob Gradient
            const gradient = ctx.createRadialGradient(bobX - 5, bobY - 5, 2, bobX, bobY, bobRadius);
            gradient.addColorStop(0, '#60a5fa');
            gradient.addColorStop(1, '#1d4ed8');

            ctx.beginPath();
            ctx.arc(bobX, bobY, bobRadius, 0, Math.PI * 2);
            ctx.fillStyle = gradient;
            ctx.fill();
            
            // Bob Border
            ctx.strokeStyle = '#1e3a8a';
            ctx.lineWidth = 1;
            ctx.stroke();

            // Draw Shadow (Fake depth)
            ctx.beginPath();
            ctx.ellipse(originX, originY + (state.length * pixelsPerMeter) + 40, bobRadius * 0.8, bobRadius * 0.2, 0, 0, Math.PI * 2);
            ctx.fillStyle = 'rgba(0,0,0,0.1)';
            ctx.fill();

            // Update UI Text
            els.periodDisplay.textContent = calculatePeriod().toFixed(2) + ' s';
        }

        function loop() {
            updatePhysics();
            draw();
            animationId = requestAnimationFrame(loop);
        }

        // --- Interaction Handlers ---
        function updateInputs() {
            state.mass = parseFloat(els.mass.value);
            state.length = parseFloat(els.length.value);
            state.gravity = parseFloat(els.gravity.value);

            els.valMass.textContent = state.mass.toFixed(1) + ' kg';
            els.valLength.textContent = state.length.toFixed(1) + ' m';
            els.valGravity.textContent = state.gravity.toFixed(1) + ' m/s²';

            // Recalculate and update display immediately even if paused
            els.periodDisplay.textContent = calculatePeriod().toFixed(2) + ' s';
        }

        els.mass.addEventListener('input', updateInputs);
        els.length.addEventListener('input', updateInputs);
        els.gravity.addEventListener('input', updateInputs);

        els.toggleBtn.addEventListener('click', () => {
            state.running = !state.running;
            els.toggleBtn.textContent = state.running ? 'Pause' : 'Resume';
            els.toggleBtn.classList.toggle('btn-outline');
        });

        els.resetBtn.addEventListener('click', () => {
            state.angle = Math.PI / 4;
            state.velocity = 0;
            state.acceleration = 0;
            state.running = true;
            els.toggleBtn.textContent = 'Pause';
            els.toggleBtn.classList.remove('btn-outline');
            
            // Reset inputs to default
            els.mass.value = 1.0;
            els.length.value = 1.0;
            els.gravity.value = 9.8;
            updateInputs();
            
            showToast("Simulation reset to defaults.");
        });

        // --- Utilities ---
        function showToast(message) {
            const toast = document.getElementById("toast");
            toast.textContent = message;
            toast.className = "show";
            setTimeout(function(){ toast.className = toast.className.replace("show", ""); }, 3000);
        }

        function scrollToSim() {
            document.querySelector('.simulation-grid').scrollIntoView({ behavior: 'smooth' });
        }

        // --- Render KaTeX ---
        document.addEventListener("DOMContentLoaded", function() {
            renderMathInElement(document.body, {
                delimiters: [
                    {left: "$$", right: "$$", display: true},
                    {left: "$", right: "$", display: false}
                ]
            });
        });

        // Start
        loop();

    </script>
