// ── tiny KaTeX helpers ──────────────────────────────────
function KT({ tex, block = false }) {
  try {
    const html = katex.renderToString(tex, {
      throwOnError: false,
      displayMode: block,
    });
    return (
      <span
        dangerouslySetInnerHTML={{ __html: html }}
        style={{ fontFamily: "inherit" }}
      />
    );
  } catch {
    return <code>{tex}</code>;
  }
}

function OrbitalAnimation({ velocityFactor }) {
  // velocityFactor: 1 = circular, <1 = elliptical/falling
  const [tick, setTick] = useState(0);
  const [running, setRunning] = useState(false);
  const [animId, setAnimId] = useState(null);

  const cx = 160, cy = 160, r = 0;
  const circR = 90;

  // Ellipse parameters based on velocity factor
  // At v_factor=1: circle. As factor drops, semi-minor shrinks
  const vf = Math.max(0.3, Math.min(1.0, velocityFactor));
  // Semi-major axis stays ~circR, semi-minor shrinks
  const semiA = circR;
  const semiB = circR * vf * vf; // shrinks quadratically
  // Ellipse centre offset so periapsis stays on right
  const c_dist = Math.sqrt(semiA * semiA - semiB * semiB);
  const ellipseCX = cx - c_dist; // shift left so right focus = cx,cy

  function startAnim() {
    if (running) return;
    setRunning(true);
    let t = tick;
    const step = () => {
      t += 0.018;
      setTick(t);
      const id = requestAnimationFrame(step);
      setAnimId(id);
    };
    const id = requestAnimationFrame(step);
    setAnimId(id);
  }

  function stopAnim() {
    setRunning(false);
    if (animId) cancelAnimationFrame(animId);
  }

  function resetAnim() {
    stopAnim();
    setTick(0);
    setRunning(false);
  }

  // Planet position on ellipse (parametric)
  const px = ellipseCX + semiA * Math.cos(tick);
  const py = cy + semiB * Math.sin(tick);

  // Circular orbit ghost position
  const gpx = cx + circR * Math.cos(tick);
  const gpy = cy + circR * Math.sin(tick);

  // Velocity vector direction (tangent to ellipse)
  const dvx = -semiA * Math.sin(tick);
  const dvy = semiB * Math.cos(tick);
  const dvMag = Math.sqrt(dvx * dvx + dvy * dvy);
  const vLen = 28 * vf;
  const vx2 = px + (dvx / dvMag) * vLen;
  const vy2 = py + (dvy / dvMag) * vLen;

  // Gravity vector toward star
  const gvx = cx - px;
  const gvy = cy - py;
  const gMag = Math.sqrt(gvx * gvx + gvy * gvy);
  const gLen = 22;
  const gx2 = px + (gvx / gMag) * gLen;
  const gy2 = py + (gvy / gMag) * gLen;

  const isCircular = vf >= 0.98;

  return (
    <div style={{ display: "flex", flexDirection: "column", alignItems: "center", gap: 12 }}>
      <svg width={320} height={320} style={{ borderRadius: 12, background: "#090b12", display: "block" }}>
        {/* Star glow */}
        <defs>
          <radialGradient id="starGlow" cx="50%" cy="50%" r="50%">
            <stop offset="0%" stopColor="#fff7a0" stopOpacity="1" />
            <stop offset="40%" stopColor="#f59e0b" stopOpacity="0.8" />
            <stop offset="100%" stopColor="#f59e0b" stopOpacity="0" />
          </radialGradient>
          <radialGradient id="planetGlow" cx="50%" cy="50%" r="50%">
            <stop offset="0%" stopColor="#93c5fd" stopOpacity="1" />
            <stop offset="100%" stopColor="#3b82f6" stopOpacity="0" />
          </radialGradient>
          <marker id="arrowV" markerWidth="6" markerHeight="6" refX="3" refY="3" orient="auto">
            <path d="M0,0 L6,3 L0,6 Z" fill={C.teal} />
          </marker>
          <marker id="arrowG" markerWidth="6" markerHeight="6" refX="3" refY="3" orient="auto">
            <path d="M0,0 L6,3 L0,6 Z" fill={C.amber} />
          </marker>
        </defs>

        {/* Circular orbit ghost */}
        <circle
          cx={cx} cy={cy} r={circR}
          fill="none"
          stroke={isCircular ? C.accent : "#2a2d3e"}
          strokeWidth={isCircular ? 1.5 : 1}
          strokeDasharray={isCircular ? "none" : "4 4"}
          opacity={0.5}
        />

        {/* Actual elliptical orbit */}
        {!isCircular && (
          <ellipse
            cx={ellipseCX} cy={cy}
            rx={semiA} ry={semiB}
            fill="none"
            stroke={C.accent}
            strokeWidth={1.5}
            opacity={0.7}
          />
        )}

        {/* Periapsis & Apoapsis labels */}
        {!isCircular && (
          <>
            <circle cx={ellipseCX + semiA} cy={cy} r={3} fill={C.red} opacity={0.8} />
            <text x={ellipseCX + semiA + 6} y={cy - 6} fill={C.red} fontSize={9} opacity={0.8}>periapsis</text>
            <circle cx={ellipseCX - semiA} cy={cy} r={3} fill={C.teal} opacity={0.8} />

      </svg>

function EnergyCard() {

  const G = 6.674e-11, M = 2e30, m = 6e24, r = 1.5e11;
  const vc = Math.sqrt(G * M / r);
  const v = vc * vFactor;
  const KE = 0.5 * m * v * v;
  const PE = -G * M * m / r;
  const E = KE + PE;
  const KEn = KE / Math.abs(PE);
  const En = E / Math.abs(PE);

  const barW = (val, max) => `${Math.min(100, Math.abs(val / max) * 100).toFixed(1)}%`;
