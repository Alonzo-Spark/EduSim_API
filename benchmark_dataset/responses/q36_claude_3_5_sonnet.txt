const renderMath = (tex: string, display = false) => {
  try {
    return katex.renderToString(tex, {
      displayMode: display,
      throwOnError: false,
    });
  } catch {
    return tex;
  }
};

const LabResult = ({
  labValues,
  tex,
}: {
  labValues: Record<string, string>;
  tex: string;
}) => {
  /* Simple result for F = -kx and a = -ω²x */
  let result: string | null = null;

  if (tex.includes("-kx")) {
    const k = parseFloat(labValues["k"] ?? "");
    const x = parseFloat(labValues["x"] ?? "");
    if (!isNaN(k) && !isNaN(x)) {
      result = `F = ${(-k * x).toFixed(4)} N`;
    }
  } else if (tex.includes("\\omega^2 x")) {
    const omega = parseFloat(labValues["\\omega"] ?? "");
    const x = parseFloat(labValues["x"] ?? "");
    if (!isNaN(omega) && !isNaN(x)) {
      result = `a = ${(-(omega * omega) * x).toFixed(4)} m/s²`;
    }
  } else if (tex.includes("2\\pi")) {
    const m = parseFloat(labValues["m"] ?? "");
    const k = parseFloat(labValues["k"] ?? "");
    if (!isNaN(m) && !isNaN(k) && k > 0) {
      result = `T = ${(2 * Math.PI * Math.sqrt(m / k)).toFixed(4)} s`;
    }
  }

  if (!result)
    return (
      <p className="text-xs text-gray-400 italic">
        Fill in all values to see the result.
      </p>
    );

  return (
    <div
      className="rounded-xl px-4 py-3 text-center font-bold text-lg"
      style={{ background: "#eef2ff", color: accentHex }}
    >
      {result}
    </div>
  );
};
