**1. Markdown Parsing & Formula Extraction**

```python
import markdown
import re

def extract_formulas(markdown_text):
    """
    Extracts LaTeX formulas and associated context from Markdown.

    Args:
        markdown_text: The input Markdown string.

    Returns:
        A list of formula dictionaries.
    """
    formulas = []
    # Regex to find LaTeX formulas ($$...$$)
    latex_pattern = r"$$([^$]+)$$"  #Improved regex to avoid overmatching
    matches = re.findall(latex_pattern, markdown_text)

    for formula_text in matches:
        # Basic context extraction (surrounding lines)
        lines = markdown_text.splitlines()
        formula_index = lines.index(f"$${formula_text}$$")
        surrounding_context = "\n".join(lines[max(0, formula_index - 2):min(len(lines), formula_index + 3)])  # Context around formula
        variables = re.findall(r"[a-zA-Z]+", formula_text) #Simple variable extraction for now

        formulas.append({
            "formula": formula_text,
            "title": "Main Formula", # Default title, can improve
            "surroundingContext": surrounding_context,
            "variables": variables
        })
    return formulas

def parse_markdown_with_formula_extraction(markdown_text):
    """
    Parses Markdown and extracts formulas.
    """
    html = markdown.markdown(markdown_text, extensions=['fenced_code']) # Enable fenced code blocks
    formulas = extract_formulas(markdown_text)

    #Replace formula instances with FormulaCard components
    for formula in formulas:
        #Create a placeholder where formula is
        placeholder = f"""
        <FormulaCard>
            <MainFormula>{formula["formula"]}</MainFormula>
            <Explanation>[Explain Formula]</Explanation>
            <FormulaLab>[Open Formula Lab]</FormulaLab>
        </FormulaCard>
        """
        html = html.replace(f"$${formula['formula']}$$", placeholder)  #Replace with placeholder
    return html
```

**2. Section Detection**

```python
def detect_sections(html_content):
    """
    Detects sections based on headings.

    Args:
        html_content: The HTML content of the document.

    Returns:
        A list of section dictionaries.
    """
    sections = []
    # A simple regex to find headings (e.g., <h1>, <h2>)
    heading_pattern = r"<h[1-6]>.*?</h[1-6]>"
    headings = re.findall(heading_pattern, html_content)

    for heading in headings:
        level = int(heading[0][1]) #Extract heading level
        title = heading[0].strip()
        sections.append({"title": title, "level": level})
    return sections

```

**4. Table Rendering**

```python
def render_table_card(table_data):
    """
    Renders a TableCard component.

    Args:
        table_data: A dictionary containing the table data (headers, rows).

    Returns:
        An HTML string representing the TableCard.
    """
    headers = table_data["headers"]
    rows = table_data["rows"]

    table_html = "<table>"
    table_html += "<thead><tr>"
    for header in headers:
        table_html += f"<th>{header}</th>"
    table_html += "</tr></thead><tbody>"

    for row in rows:
        table_html += "<tr>"
        for cell in row:
            table_html += f"<td>{cell}</td>"
        table_html += "</tr>"

    table_html += "</tbody></table>"

    return f"""
    <TableCard>
        <h2>Table: {table_data['title']}</h2>
        {table_html}
    </TableCard>
    """
```
