        document.addEventListener("DOMContentLoaded", function() {
            // Render initial formulas
            katex.render("v = u + at", document.getElementById("main-formula"), { throwOnError: false, displayMode: true });

            katex.render("v = 0 + (2 \\times 10)", document.getElementById("step2-math"), { throwOnError: false });
            katex.render("v = 20 \\text{ m/s}", document.getElementById("step3-math"), { throwOnError: false });
            
            updateLab();
        });

        // Modal Logic
        function openExplainModal() {
            document.getElementById('explainModal').classList.add('active');
        }

        function openLabModal() {
            document.getElementById('labModal').classList.add('active');
            updateLab(); // Ensure state is fresh
        }

        function closeModals() {
            document.querySelectorAll('.modal-overlay').forEach(el => el.classList.remove('active'));
        }

        // Close on clicking outside
        document.querySelectorAll('.modal-overlay').forEach(overlay => {
            overlay.addEventListener('click', function(e) {
                if (e.target === overlay) {
                    closeModals();
                }
            });
        });

        // Formula Lab Logic
        function updateLab() {
            const u = parseFloat(document.getElementById('input-u').value);
            const a = parseFloat(document.getElementById('input-a').value);
            const t = parseFloat(document.getElementById('input-t').value);

            // Update Labels
            document.getElementById('val-u').textContent = u;
            document.getElementById('val-a').textContent = a;
            document.getElementById('val-t').textContent = t;

            // Calculate Result
            const v = u + (a * t);
            
            // Render Dynamic Formula
            const formulaString = `v = ${u} + (${a} \\times ${t})`;
            katex.render(formulaString, document.getElementById('lab-formula-render'), { throwOnError: false, displayMode: true });
            
            // Update Result Text
            document.getElementById('lab-result').textContent = v.toFixed(2);
        }
