const BADGE: React.CSSProperties = {

// ── Spring Canvas ──────────────────────────────────────────────────────────────
function SpringCanvas({
  k,
  label,
  color,
  phaseOffset = 0,
  running,
}: {
  k: number;
  label: string;
  color: string;
  phaseOffset?: number;
  running: boolean;
}) {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const animRef = useRef<number>(0);
  const startRef = useRef<number | null>(null);
  const pausedAtRef = useRef<number>(0);

  const W = 320, H = 200;
  const m = 1;
  const A = 55;
  const equilibriumX = W / 2;
  const wallX = 30;
  const massSize = 28;

  const omega = Math.sqrt(k / m);
  const T = (2 * Math.PI) / omega;

  const drawSpring = useCallback(
    (ctx: CanvasRenderingContext2D, massX: number) => {
      const coils = 10;
      const coilH = 8;
      const springStart = wallX + 8;
      const springEnd = massX - massSize / 2;
      const segLen = (springEnd - springStart) / (coils * 2);

      ctx.beginPath();
      ctx.moveTo(springStart, H / 2);
      for (let i = 0; i <= coils * 2; i++) {
        const x = springStart + i * segLen;
        const y = i % 2 === 0 ? H / 2 : H / 2 + (i % 4 === 1 ? coilH : -coilH);
        ctx.lineTo(x, y);
      }
      ctx.lineTo(springEnd, H / 2);
      ctx.strokeStyle = color;
      ctx.lineWidth = 2.5;
      ctx.stroke();
    },
    [color]
  );

  const draw = useCallback(
    (t: number) => {
      const canvas = canvasRef.current;
      if (!canvas) return;
      const ctx = canvas.getContext("2d")!;
      ctx.clearRect(0, 0, W, H);

      // Background grid
      ctx.strokeStyle = "rgba(255,255,255,0.04)";
      ctx.lineWidth = 1;
      for (let x = 0; x < W; x += 30) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
      }
      for (let y = 0; y < H; y += 30) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
      }

      // Wall
      ctx.fillStyle = "rgba(99,102,241,0.6)";
      ctx.fillRect(wallX - 8, 20, 8, H - 40);
      ctx.fillStyle = "rgba(99,102,241,0.2)";
      for (let y = 20; y < H - 40; y += 12) {
        ctx.beginPath();
        ctx.moveTo(wallX - 8, y);
        ctx.lineTo(wallX - 18, y + 8);
        ctx.strokeStyle = "rgba(99,102,241,0.4)";
        ctx.lineWidth = 1.5;
        ctx.stroke();
      }

      // Equilibrium line
      ctx.setLineDash([4, 4]);
      ctx.strokeStyle = "rgba(255,255,255,0.15)";
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(equilibriumX, 20);
      ctx.lineTo(equilibriumX, H - 20);
      ctx.stroke();
      ctx.setLineDash([]);

      const displacement = A * Math.cos(omega * t + phaseOffset);
      const massX = equilibriumX + displacement;

      drawSpring(ctx, massX);

      // Mass shadow
      ctx.fillStyle = "rgba(0,0,0,0.3)";
      ctx.beginPath();
      ctx.ellipse(massX, H / 2 + massSize / 2 + 6, massSize / 2, 5, 0, 0, Math.PI * 2);
      ctx.fill();

      // Mass
      const grad = ctx.createRadialGradient(
        massX - 4, H / 2 - 4, 2,
        massX, H / 2, massSize / 2
      );
      grad.addColorStop(0, color + "ff");
      grad.addColorStop(1, color + "88");
      ctx.fillStyle = grad;
      ctx.beginPath();
      ctx.roundRect(
        massX - massSize / 2,
        H / 2 - massSize / 2,
        massSize,
        massSize,
        6
      );
      ctx.fill();
      ctx.strokeStyle = color;
      ctx.lineWidth = 1.5;
      ctx.stroke();

      // Velocity arrow
      const vel = -A * omega * Math.sin(omega * t + phaseOffset);
      const arrowLen = vel * 0.5;
      if (Math.abs(arrowLen) > 3) {
        ctx.strokeStyle = "#fbbf24";
        ctx.fillStyle = "#fbbf24";
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(massX, H / 2 - massSize);
        ctx.lineTo(massX + arrowLen, H / 2 - massSize);
        ctx.stroke();
        const dir = arrowLen > 0 ? 1 : -1;
        ctx.beginPath();
        ctx.moveTo(massX + arrowLen, H / 2 - massSize);
        ctx.lineTo(massX + arrowLen - dir * 8, H / 2 - massSize - 4);
        ctx.lineTo(massX + arrowLen - dir * 8, H / 2 - massSize + 4);
        ctx.closePath();
        ctx.fill();
      }

      // Info overlay
      ctx.fillStyle = "rgba(0,0,0,0.5)";
      ctx.beginPath();
      ctx.roundRect(8, 8, 160, 52, 8);
      ctx.fill();
      ctx.fillStyle = color;
      ctx.font = "bold 13px monospace";
      ctx.fillText(`k = ${k} N/m`, 16, 26);
      ctx.fillStyle = "#94a3b8";
      ctx.font = "11px monospace";
      ctx.fillText(`ω = ${omega.toFixed(2)} rad/s`, 16, 42);
      ctx.fillText(`T = ${T.toFixed(2)} s`, 90, 42);

      // Displacement indicator
      ctx.fillStyle = "rgba(0,0,0,0.5)";
      ctx.beginPath();
      ctx.roundRect(W - 110, 8, 102, 30, 8);
      ctx.fill();
      ctx.fillStyle = displacement >= 0 ? "#34d399" : "#f87171";
      ctx.font = "11px monospace";
      ctx.fillText(`x = ${displacement.toFixed(1)} cm`, W - 104, 27);
    },
    [k, omega, T, color, phaseOffset, drawSpring]
  );

  useEffect(() => {
    if (!running) {
      cancelAnimationFrame(animRef.current);
      draw(pausedAtRef.current);
      return;
    }

    const loop = (timestamp: number) => {
      if (startRef.current === null) startRef.current = timestamp - pausedAtRef.current * 1000;
      const elapsed = (timestamp - startRef.current) / 1000;
      pausedAtRef.current = elapsed;
      draw(elapsed);
      animRef.current = requestAnimationFrame(loop);
    };

    startRef.current = null;
    animRef.current = requestAnimationFrame(loop);
    return () => cancelAnimationFrame(animRef.current);
  }, [running, draw]);

  return (
    <div style={{ textAlign: "center" }}>
      <div style={{ ...BADGE, borderColor: color + "80", color }}>{label}</div>
      <canvas
        ref={canvasRef}
        width={W}
        height={H}
        style={{
          borderRadius: "12px",
          border: `1px solid ${color}40`,
          background: "#0a0a1a",
          display: "block",
          margin: "0 auto",
        }}
      />
    </div>
  );
}

// ── Phase Space Plot ───────────────────────────────────────────────────────────
function PhaseSpacePlot({ k1, k2 }: { k1: number; k2: number }) {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d")!;
    const W = 300, H = 200;
    ctx.clearRect(0, 0, W, H);

    ctx.fillStyle = "#0a0a1a";
    ctx.fillRect(0, 0, W, H);

    // Axes
    ctx.strokeStyle = "rgba(255,255,255,0.2)";
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.moveTo(W / 2, 10); ctx.lineTo(W / 2, H - 10); ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(10, H / 2); ctx.lineTo(W - 10, H / 2); ctx.stroke();

    ctx.fillStyle = "rgba(255,255,255,0.3)";
    ctx.font = "10px monospace";
    ctx.fillText("x →", W - 24, H / 2 - 4);
    ctx.fillText("v ↑", W / 2 + 4, 18);

    const drawEllipse = (k: number, color: string) => {
      const A = 55, m = 1;
      const omega = Math.sqrt(k / m);
      const scaleX = (W / 2 - 20) / A;
      const scaleY = (H / 2 - 20) / (A * omega);

      ctx.beginPath();
      for (let i = 0; i <= 360; i++) {
        const angle = (i * Math.PI) / 180;
        const x = W / 2 + A * Math.cos(angle) * scaleX;
        const y = H / 2 - A * omega * Math.sin(angle) * scaleY;
        i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
      }
      ctx.strokeStyle = color;
      ctx.lineWidth = 2;
      ctx.stroke();
    };

    drawEllipse(k1, "#6366f1");
    drawEllipse(k2, "#f472b6");

    ctx.font = "10px monospace";
    ctx.fillStyle = "#6366f1";
    ctx.fillText(`k=${k1}`, 14, 18);
    ctx.fillStyle = "#f472b6";
    ctx.fillText(`k=${k2}`, 14, 32);
  }, [k1, k2]);

  return (
    <div>
      <div style={BADGE}>Phase Space Portrait</div>
      <canvas
        ref={canvasRef}
        width={300}
        height={200}
        style={{
          borderRadius: "12px",
          border: "1px solid rgba(255,255,255,0.1)",
          display: "block",
        }}
      />
    </div>
  );
}

// ── Period Bar Chart ───────────────────────────────────────────────────────────
function PeriodChart({ k1, k2 }: { k1: number; k2: number }) {
  const T1 = (2 * Math.PI) / Math.sqrt(k1);
  const T2 = (2 * Math.PI) / Math.sqrt(k2);
  const maxT = Math.max(T1, T2);

  const Bar = ({ value, max, color, label }: { value: number; max: number; color: string; label: string }) => (
    <div style={{ marginBottom: "12px" }}>
      <div style={{ display: "flex", justifyContent: "space-between", marginBottom: "4px" }}>
        <span style={{ color: "#94a3b8", fontSize: "12px" }}>{label}</span>
        <span style={{ color, fontSize: "12px", fontFamily: "monospace" }}>{value.toFixed(3)} s</span>
      </div>
      <div style={{ background: "rgba(255,255,255,0.08)", borderRadius: "6px", height: "20px", overflow: "hidden" }}>
        <div
          style={{
            width: `${(value / max) * 100}%`,
            height: "100%",
            background: `linear-gradient(90deg, ${color}cc, ${color}44)`,
            borderRadius: "6px",
            transition: "width 0.5s ease",
          }}
        />
      </div>
    </div>
  );

  return (
    <div>
      <div style={BADGE}>Period Comparison</div>
      <Bar value={T1} max={maxT * 1.1} color="#6366f1" label={`k = ${k1} N/m`} />
      <Bar value={T2} max={maxT * 1.1} color="#f472b6" label={`k = ${k2} N/m`} />
      <div
        style={{
          marginTop: "12px",
          padding: "10px 14px",
          background: "rgba(251,191,36,0.1)",
          border: "1px solid rgba(251,191,36,0.3)",
          borderRadius: "8px",
          fontSize: "12px",
          color: "#fbbf24",
          fontFamily: "monospace",
        }}
      >
        T₂ / T₁ = {(T2 / T1).toFixed(4)} ≈ 1/√2 ≈ {(1 / Math.sqrt(2)).toFixed(4)}
      </div>
    </div>
  );
}
