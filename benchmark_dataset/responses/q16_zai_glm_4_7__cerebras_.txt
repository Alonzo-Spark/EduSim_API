        function renderSections() {
            sections.forEach(section => {
                const card = document.createElement('article');
                card.className = `card ${section.type === 'formula' ? 'formula-card' : ''}`;

                // Header
                const header = document.createElement('div');
                header.className = 'card-header';
                header.innerHTML = `${icons[section.type]} <div class="card-title">${section.title}</div>`;
                card.appendChild(header);

                // Body
                const body = document.createElement('div');
                body.className = 'card-body';

                if (section.type === 'formula') {
                    // Formula Card Logic
                    body.innerHTML = `
                        <div class="formula-display">$$${section.latex}$$</div>
                        <div class="formula-explanation" id="explain-${section.title.replace(/\s/g, '')}">
                            <strong>Variables:</strong><br>
                            ${section.explanation}
                        </div>
                        <div class="formula-actions">
                            <button class="btn btn-outline" onclick="toggleExplanation('${section.title.replace(/\s/g, '')}')">
                                Explain Formula
                            </button>
                            <button class="btn btn-primary" onclick="openLab('${section.latex}', ${JSON.stringify(section.variables).replace(/"/g, '&quot;')})">
                                Open Formula Lab
                            </button>
                        </div>
                    `;
                } else if (section.type === 'table') {
                    // Table Card Logic
                    let tableHTML = '<div class="table-container"><table><thead><tr>';
                    section.headers.forEach(h => tableHTML += `<th>${h}</th>`);
                    tableHTML += '</tr></thead><tbody>';
                    section.rows.forEach(row => {
                        tableHTML += '<tr>';
                        row.forEach(cell => tableHTML += `<td>${cell}</td>`);
                        tableHTML += '</tr>';
                    });
                    tableHTML += '</tbody></table></div>';
                    body.innerHTML = tableHTML;
                } else {
                    // Standard Text Card
                    body.innerHTML = section.content;
                }

                card.appendChild(body);
                root.appendChild(card);
            });

            // Trigger Math Rendering
            if (typeof renderMathInElement === 'function') {
                renderMathInElement(root);
            }
        }

        // ==========================================
        // INTERACTION LOGIC
        // ==========================================

        function toggleExplanation(id) {
            const el = document.getElementById('explain-' + id);
            el.classList.toggle('visible');
        }

        // Formula Lab State
        let currentFormula = null;
        let currentVariables = [];

        function openLab(formula, variables) {
            currentFormula = formula;
            currentVariables = variables;
            
            const labContent = document.getElementById('lab-content');
            labContent.innerHTML = '';
            
            // Generate Inputs based on extracted variables
            variables.forEach(v => {
                const group = document.createElement('div');
                group.className = 'lab-input-group';
                group.innerHTML = `
                    <label>${v.label} (${v.name})</label>
                    <input type="number" id="input-${v.name}" placeholder="0">
                `;
                labContent.appendChild(group);
            });

            // Reset Result
            document.getElementById('lab-result').innerText = 'Enter values to calculate Work.';
            
            // Show Modal
            document.getElementById('lab-modal').classList.add('active');
        }

        function closeLab() {
            document.getElementById('lab-modal').classList.remove('active');
        }

        function calculateWork() {
            // Extract values
            const F = parseFloat(document.getElementById('input-F').value) || 0;
            const s = parseFloat(document.getElementById('input-s').value) || 0;
            const theta = parseFloat(document.getElementById('input-θ').value) || 0;

            // Physics Calculation
            // Convert degrees to radians for JS Math.cos
            const radians = theta * (Math.PI / 180);
            const work = F * s * Math.cos(radians);

            // Format Output
            const resultEl = document.getElementById('lab-result');
            
            // Simple check for significant figures logic
            const formattedWork = Math.abs(work) < 0.001 && work !== 0 ? work.toExponential(2) : work.toFixed(2);
            
            resultEl.innerHTML = `
                Work Done (W) = <span style="font-size: 1.2em; margin-left: 8px;">${formattedWork} J</span>
            `;
        }

        // Initialize
        document.addEventListener('DOMContentLoaded', renderSections);
