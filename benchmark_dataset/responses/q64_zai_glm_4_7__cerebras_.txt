    <script>
        // --- 1. KaTeX Rendering ---
        document.addEventListener("DOMContentLoaded", function() {
            katex.render("F = -kx", document.getElementById("formula-render"), {
                throwOnError: false,
                displayMode: true
            });
        });

        // --- 2. Three.js Scene Setup ---
        const container = document.getElementById('canvas-container');
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0x0f172a); // Matches Tailwind bg-slate-900
        scene.fog = new THREE.Fog(0x0f172a, 10, 50);

        const camera = new THREE.PerspectiveCamera(45, container.clientWidth / container.clientHeight, 0.1, 1000);
        camera.position.set(0, 5, 12);
        camera.lookAt(0, 0, 0);

        const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
        renderer.setSize(container.clientWidth, container.clientHeight);
        renderer.shadowMap.enabled = true;
        container.appendChild(renderer.domElement);

        // Lighting
        const ambientLight = new THREE.AmbientLight(0x404040, 2);
        scene.add(ambientLight);

        const dirLight = new THREE.DirectionalLight(0xffffff, 1);
        dirLight.position.set(5, 10, 7);
        dirLight.castShadow = true;
        scene.add(dirLight);

        const pointLight = new THREE.PointLight(0x38bdf8, 1, 20); // Sci-fi blue glow
        pointLight.position.set(0, 0, 0);
        scene.add(pointLight);

        // Floor Grid
        const gridHelper = new THREE.GridHelper(30, 30, 0x1e293b, 0x1e293b);
        scene.add(gridHelper);

        // --- 3. Spring Object ---
        // We create a custom spring using TubeGeometry along a spiral path
        let springMesh;
        const springRadius = 1.5;
        const springHeight = 6;
        const springTurns = 8;
        
        // Function to generate spring geometry
        function createSpringGeometry(height) {
            const points = [];
            for (let i = 0; i <= 100; i++) {
                const t = i / 100;
                const angle = t * Math.PI * 2 * springTurns;
                const x = springRadius * Math.cos(angle);
                const y = springRadius * Math.sin(angle);
                const z = (t - 0.5) * height;
                points.push(new THREE.Vector3(x, y, z));
            }
            const curve = new THREE.CatmullRomCurve3(points);
            return new THREE.TubeGeometry(curve, 100, 0.2, 12, false);
        }

        const springMaterial = new THREE.MeshStandardMaterial({
            color: 0x38bdf8,
            metalness: 0.8,
            roughness: 0.2,
            emissive: 0x0f172a
        });

        springMesh = new THREE.Mesh(createSpringGeometry(springHeight), springMaterial);
        springMesh.castShadow = true;
        scene.add(springMesh);

        // Anchor points (Top and Bottom plates)
        const plateGeo = new THREE.CylinderGeometry(2, 2, 0.2, 32);
        const plateMat = new THREE.MeshStandardMaterial({ color: 0x475569 });
        
        const topPlate = new THREE.Mesh(plateGeo, plateMat);
        topPlate.position.y = 2; // Initial top position
        scene.add(topPlate);

        const bottomPlate = new THREE.Mesh(plateGeo, plateMat);
        bottomPlate.position.y = -2; // Initial bottom position
        scene.add(bottomPlate);

        // --- 4. Logic & Physics State ---
        const state = {
            force: 20,
            k: 50,
            maxForce: 80, // Elastic Limit
            isBroken: false
        };

        // UI Elements
        const forceSlider = document.getElementById('force-slider');
        const kSlider = document.getElementById('k-slider');
        const forceDisplay = document.getElementById('force-display');
        const kDisplay = document.getElementById('k-display');
        const valX = document.getElementById('val-x');
        const valU = document.getElementById('val-u');
        const valStatus = document.getElementById('val-status');
        const runtimeOverlay = document.getElementById('runtime-overlay');
        const resetBtn = document.getElementById('reset-btn');

        function updatePhysics() {
            // Hooke's Law: F = k * x  =>  x = F / k
            // Scale factor for visual representation (arbitrary units to meters)
            const scaleFactor = 0.02; 
            let displacement = (state.force / state.k) * scaleFactor;

            // Check Limit
            if (state.force > state.maxForce) {
                if (!state.isBroken) {
                    state.isBroken = true;
                    triggerOverlay();
                }
                // Non-linear stretch or break simulation
                displacement += (state.force - state.maxForce) * 0.0005; 
            } else {
                state.isBroken = false;
                hideOverlay();
            }

            // Update UI Values
            forceDisplay.innerText = `${state.force}N`;
            kDisplay.innerText = state.k;
            valX.innerText = displacement.toFixed(2) + " m";
            
            // Potential Energy: U = 0.5 * k * x^2
            const energy = 0.5 * state.k * Math.pow(displacement, 2);
            valU.innerText = energy.toFixed(1) + " J";

            // Update Status Color
            if (state.isBroken) {
                valStatus.innerText = "CRITICAL";
                valStatus.className = "text-xl font-bold text-red-500 animate-pulse";
                springMaterial.color.setHex(0xef4444); // Red
                springMaterial.emissive.setHex(0x7f1d1d);
                pointLight.color.setHex(0xef4444);
            } else if (state.force > state.maxForce * 0.8) {
                valStatus.innerText = "WARNING";
                valStatus.className = "text-xl font-bold text-yellow-500";
                springMaterial.color.setHex(0xf59e0b); // Orange/Yellow
                springMaterial.emissive.setHex(0x451a03);
                pointLight.color.setHex(0xf59e0b);
            } else {
                valStatus.innerText = "ELASTIC";
                valStatus.className = "text-xl font-bold text-green-400";
                springMaterial.color.setHex(0x38bdf8); // Blue
                springMaterial.emissive.setHex(0x0f172a);
                pointLight.color.setHex(0x38bdf8);
            }

            // Update 3D Geometry
            // We remove old geometry and add new one to simulate stretching cleanly
            // (For high performance, vertex shader is better, but for this scale, geometry regen is fine)
            const visualStretch = 6 + (displacement * 10); // Base height + stretch
            
            // Update plates
            topPlate.position.y = visualStretch / 2;
            bottomPlate.position.y = -visualStretch / 2;

            springMesh.geometry.dispose();
            springMesh.geometry = createSpringGeometry(visualStretch);
        }

        function triggerOverlay() {
            runtimeOverlay.classList.add('active');
        }

        function hideOverlay() {
            runtimeOverlay.classList.remove('active');
        }

        // Event Listeners
        forceSlider.addEventListener('input', (e) => {
            state.force = parseInt(e.target.value);
            updatePhysics();
        });

        kSlider.addEventListener('input', (e) => {
            state.k = parseInt(e.target.value);
            updatePhysics();
        });

        resetBtn.addEventListener('click', () => {
            state.force = 20;
            state.k = 50;
            forceSlider.value = 20;
            kSlider.value = 50;
            state.isBroken = false;
            hideOverlay();
            updatePhysics();
        });

        // --- 5. Animation Loop ---
        function animate() {
            requestAnimationFrame(animate);
            
            // Subtle rotation of the spring for 3D effect
            if (springMesh) {
                springMesh.rotation.z += 0.002;
                topPlate.rotation.z += 0.002;
                bottomPlate.rotation.z += 0.002;
            }

            // Particle effect simulation (simple floating particles)
            // (Skipped complex particle system to keep code within single-file constraints effectively)

            renderer.render(scene, camera);
        }

        // Handle Resize
        window.addEventListener('resize', () => {
            const width = container.clientWidth;
            const height = container.clientHeight;
            renderer.setSize(width, height);
            camera.aspect = width / height;
            camera.updateProjectionMatrix();
        });

        // Initialize
        updatePhysics();
        animate();

    </script>
</body>
