function KaTeX({ latex, display = false }: { latex: string; display?: boolean }) {
  const ref = useRef<HTMLSpanElement>(null);
  useEffect(() => {
    const render = () => {
      if (ref.current && window.katex) {
        try {
          ref.current.innerHTML = window.katex.renderToString(latex, {
            displayMode: display,
            throwOnError: false,
          });
        } catch {}
      }
    };
    if (window.katex) {
      render();
    } else {
      const interval = setInterval(() => {
        if (window.katex) {
          clearInterval(interval);
          render();
        }
      }, 100);
      return () => clearInterval(interval);
    }
  }, [latex, display]);
  return <span ref={ref} />;
}

function PostCollisionLab() {

  const [m2, setM2] = useState(2);
  const [v1, setV1] = useState(8);
  const [v2, setV2] = useState(-3);

  const totalM = m1 + m2;
  const pBefore = m1 * v1 + m2 * v2;
  const v1f = (m1 * v1 + m2 * v2 - m2 * e * (v1 - v2)) / totalM;

function MomentumLab() {
  const [m1, setM1] = useState(3);
  const [v1, setV1] = useState(6);
  const [m2, setM2] = useState(2);
  const [v2, setV2] = useState(-2);

  const p1 = m1 * v1;
  const p2 = m2 * v2;
  const total = p1 + p2;
  const maxBar = Math.max(Math.abs(p1), Math.abs(p2), Math.abs(total), 1);

  const Bar = ({ value, color }: { value: number; color: string }) => (
    <div className="flex items-center gap-2">
      <div className="flex-1 h-4 bg-slate-100 rounded-full overflow-hidden relative">
        <div
          className={`absolute top-0 h-full rounded-full transition-all ${color}`}
          style={{
            width: `${(Math.abs(value) / maxBar) * 100}%`,
            left: value >= 0 ? "50%" : `${50 - (Math.abs(value) / maxBar) * 50}%`,
          }}
        />
        <div className="absolute left-1/2 top-0 h-full w-px bg-slate-300" />
      </div>
      <span className="font-mono text-xs w-16 text-right">{value.toFixed(1)}</span>

function PhysicsSandbox() {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const animRef = useRef<number>(0);
  const stateRef = useRef<{
    balls: Ball[];
    collided: boolean;
    collisionData: { v1Before: number; v2Before: number; v1After: number; v2After: number } | null;
  }>({
    balls: [],
    collided: false,
    collisionData: null,
  });

  const [e, setE] = useState(0.8);
  const [mass1, setMass1] = useState(2);
  const [mass2, setMass2] = useState(2);
  const [speed1, setSpeed1] = useState(120);
  const [isRunning, setIsRunning] = useState(false);
  const [collisionData, setCollisionData] = useState<{
    v1Before: number;
    v2Before: number;
    v1After: number;
    v2After: number;
  } | null>(null);
  const [showTrails, setShowTrails] = useState(true);

  const CANVAS_W = 560;
  const CANVAS_H = 200;
  const GRAVITY = 0;

  const initBalls = useCallback(() => {
    const r1 = 10 + mass1 * 4;
    const r2 = 10 + mass2 * 4;
    stateRef.current = {
      balls: [
        {
          x: 80,
          y: CANVAS_H / 2,
          vx: speed1,
          vy: 0,
          radius: r1,
          mass: mass1,
          color: "#6366f1",
          trail: [],
        },
        {
          x: CANVAS_W - 80,
          y: CANVAS_H / 2,
          vx: -speed1 * 0.3,
          vy: 0,
          radius: r2,
          mass: mass2,
          color: "#8b5cf6",
          trail: [],
        },
      ],
      collided: false,
      collisionData: null,
    };
    setCollisionData(null);
  }, [mass1, mass2, speed1]);

  const drawFrame = useCallback(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;
    const { balls } = stateRef.current;

    ctx.clearRect(0, 0, CANVAS_W, CANVAS_H);

    // Background
    ctx.fillStyle = "#f8fafc";
    ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);

    // Grid lines
    ctx.strokeStyle = "#e2e8f0";
    ctx.lineWidth = 1;
    for (let x = 0; x < CANVAS_W; x += 40) {
      ctx.beginPath();
      ctx.moveTo(x, 0);
      ctx.lineTo(x, CANVAS_H);
      ctx.stroke();
    }
    for (let y = 0; y < CANVAS_H; y += 40) {
      ctx.beginPath();
      ctx.moveTo(0, y);
      ctx.lineTo(CANVAS_W, y);
      ctx.stroke();
    }

    // Trails
    if (showTrails) {
      balls.forEach((ball) => {
        if (ball.trail.length < 2) return;
        ctx.beginPath();
        ctx.moveTo(ball.trail[0].x, ball.trail[0].y);
        ball.trail.forEach((pt, i) => {
          if (i > 0) ctx.lineTo(pt.x, pt.y);
        });
        ctx.strokeStyle = ball.color + "44";
        ctx.lineWidth = ball.radius * 0.5;
        ctx.lineCap = "round";
        ctx.stroke();
      });
    }

    // Balls
    balls.forEach((ball) => {
      // Shadow
      ctx.beginPath();
      ctx.arc(ball.x + 2, ball.y + 2, ball.radius, 0, Math.PI * 2);
      ctx.fillStyle = "rgba(0,0,0,0.08)";
      ctx.fill();

      // Ball body
      const grad = ctx.createRadialGradient(
        ball.x - ball.radius * 0.3,
        ball.y - ball.radius * 0.3,
        ball.radius * 0.1,
        ball.x,
        ball.y,
        ball.radius
      );
      grad.addColorStop(0, ball.color + "ff");
      grad.addColorStop(1, ball.color + "99");
      ctx.beginPath();
      ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
      ctx.fillStyle = grad;
      ctx.fill();
      ctx.strokeStyle = ball.color;
      ctx.lineWidth = 1.5;
      ctx.stroke();

      // Velocity arrow
      const arrowLen = Math.abs(ball.vx) * 0.25;
      if (arrowLen > 2) {
        ctx.beginPath();
        ctx.moveTo(ball.x, ball.y);
        ctx.lineTo(ball.x + Math.sign(ball.vx) * (ball.radius + arrowLen), ball.y);
        ctx.strokeStyle = ball.color;
        ctx.lineWidth = 2;
        ctx.stroke();
        // Arrowhead
        const ax = ball.x + Math.sign(ball.vx) * (ball.radius + arrowLen);
        ctx.beginPath();
        ctx.moveTo(ax, ball.y);
        ctx.lineTo(ax - Math.sign(ball.vx) * 8, ball.y - 4);
        ctx.lineTo(ax - Math.sign(ball.vx) * 8, ball.y + 4);
        ctx.fillStyle = ball.color;
        ctx.fill();
      }
    });

    // Labels
    balls.forEach((ball, i) => {
      ctx.fillStyle = "#1e293b";
      ctx.font = "bold 11px monospace";
      ctx.textAlign = "center";
      ctx.fillText(`m${i + 1}=${ball.mass}kg`, ball.x, ball.y - ball.radius - 6);
      ctx.fillStyle = "#64748b";
      ctx.font = "10px monospace";
      ctx.fillText(`${ball.vx.toFixed(0)} m/s`, ball.x, ball.y + ball.radius + 14);
    });
  }, [showTrails]);

  const step = useCallback(() => {
    const state = stateRef.current;
    const { balls } = state;
    const dt = 1 / 60;

    balls.forEach((ball) => {
      ball.x += ball.vx * dt * 1;
      ball.y += ball.vy * dt * 1;

      // Trails
      ball.trail.push({ x: ball.x, y: ball.y });
      if (ball.trail.length > 40) ball.trail.shift();

      // Wall bounce
      if (ball.x - ball.radius < 0) {
        ball.x = ball.radius;
        ball.vx = Math.abs(ball.vx) * e;
      }
      if (ball.x + ball.radius > CANVAS_W) {
        ball.x = CANVAS_W - ball.radius;
        ball.vx = -Math.abs(ball.vx) * e;
      }
    });

    // Ball-ball collision
    const [b1, b2] = balls;
    const dist = Math.abs(b2.x - b1.x);
    const minDist = b1.radius + b2.radius;

    if (dist <= minDist && !state.collided) {
      state.collided = true;
      const v1Before = b1.vx;
      const v2Before = b2.vx;
      const m1 = b1.mass;
      const m2 = b2.mass;
      const totalM = m1 + m2;

      const v1After = (m1 * v1Before + m2 * v2Before - m2 * e * (v1Before - v2Before)) / totalM;
      const v2After = (m1 * v1Before + m2 * v2Before + m1 * e * (v1Before - v2Before)) / totalM;

      b1.vx = v1After;
      b2.vx = v2After;

      // Separate overlapping balls
      const overlap = minDist - dist;
      b1.x -= overlap / 2;
      b2.x += overlap / 2;

      const cd = { v1Before, v2Before, v1After, v2After };
      state.collisionData = cd;
      setCollisionData(cd);
    }

    if (state.collided) {
      const dist2 = Math.abs(b2.x - b1.x);
      if (dist2 > minDist + 10) {
        state.collided = false;
      }
    }
  }, [e]);

  useEffect(() => {
    initBalls();
  }, [initBalls]);

  useEffect(() => {
    drawFrame();
  }, [drawFrame]);

  useEffect(() => {
    if (!isRunning) {
      cancelAnimationFrame(animRef.current);
      return;
    }
    const loop = () => {
      step();
      drawFrame();
      animRef.current = requestAnimationFrame(loop);
    };
    animRef.current = requestAnimationFrame(loop);
    return () => cancelAnimationFrame(animRef.current);
  }, [isRunning, step, drawFrame]);

  const handleReset = () => {
    setIsRunning(false);
    setTimeout(() => {
      initBalls();
      setTimeout(drawFrame, 50);
    }, 50);
  };

  const eLabel =
    e === 0
      ? "Perfectly Inelastic"
      : e < 0.4
      ? "Highly Inelastic"
      : e < 0.8
      ? "Partially Elastic"
      : e < 1
      ? "Nearly Elastic"
      : "Perfectly Elastic";

  const eColor =
    e === 0
      ? "text-rose-600"
      : e < 0.4
      ? "text-orange-500"
      : e < 0.8
      ? "text-amber-500"
      : "text-emerald-500";

  return (
    <div className="bg-white rounded-xl border border-slate-200 shadow-sm overflow-hidden">
      {/* Canvas */}
      <div className="bg-slate-50 border-b border-slate-200 p-3">
        <canvas
          ref={canvasRef}
          width={CANVAS_W}
          height={CANVAS_H}
          className="w-full rounded-lg"
          style={{ maxHeight: "200px" }}
        />
      </div>

      {/* Controls */}
      <div className="p-4 space-y-4">
        {/* Restitution slider - prominent */}
        <div className="bg-gradient-to-r from-indigo-50 to-violet-50 rounded-xl p-4 border border-indigo-100">
          <div className="flex items-center justify-between mb-2">
            <span className="text-sm font-bold text-slate-700">
              Coefficient of Restitution <KaTeX latex="(e)" />
            </span>
            <span className={`text-sm font-bold ${eColor}`}>{eLabel}</span>
          </div>
          <input
            type="range"
            min={0}
            max={1}
            step={0.01}
            value={e}
            onChange={(ev) => {
              setE(Number(ev.target.value));
              handleReset();
            }}
            className="w-full accent-indigo-500 h-2"
          />
          <div className="flex justify-between mt-1">
            <span className="text-xs text-slate-400">e = 0 (Inelastic)</span>
            <span className="text-2xl font-black text-indigo-600 font-mono">{e.toFixed(2)}</span>
            <span className="text-xs text-slate-400">e = 1 (Elastic)</span>
          </div>
        </div>

        {/* Other controls */}
        <div className="grid grid-cols-3 gap-3">
          {[
            { label: "Mass 1 (kg)", val: mass1, set: setMass1, min: 1, max: 8, step: 0.5 },
            { label: "Mass 2 (kg)", val: mass2, set: setMass2, min: 1, max: 8, step: 0.5 },
            { label: "Initial Speed", val: speed1, set: setSpeed1, min: 40, max: 200, step: 10 },
          ].map((ctrl) => (
            <div key={ctrl.label} className="flex flex-col gap-1">
              <label className="text-xs font-medium text-slate-500">{ctrl.label}</label>
              <input
                type="range"
                min={ctrl.min}
                max={ctrl.max}
                step={ctrl.step}
                value={ctrl.val}
                onChange={(ev) => {
                  ctrl.set(Number(ev.target.value));
                  handleReset();
                }}
                className="accent-violet-500"
              />
              <span className="text-xs text-center font-mono text-slate-600">{ctrl.val}</span>
            </div>
          ))}
        </div>

        {/* Buttons */}
        <div className="flex gap-2 flex-wrap">
          <button
            onClick={() => setIsRunning(!isRunning)}
            className={`flex-1 py-2 rounded-lg text-sm font-semibold transition-all ${
              isRunning
                ? "bg-rose-100 text-rose-600 hover:bg-rose-200"
                : "bg-indigo-600 text-white hover:bg-indigo-700"
            }`}
          >
            {isRunning ? "⏸ Pause" : "▷ Launch"}
          </button>
          <button
            onClick={handleReset}
            className="px-4 py-2 rounded-lg text-sm font-semibold bg-slate-100 text-slate-600 hover:bg-slate-200"
          >
            ↺ Reset
          </button>
          <button
            onClick={() => setShowTrails(!showTrails)}
            className={`px-4 py-2 rounded-lg text-sm font-semibold transition-all ${
              showTrails ? "bg-violet-100 text-violet-600" : "bg-slate-100 text-slate-600"
            }`}
          >
            {showTrails ? "◌ Trails On" : "◌ Trails Off"}
          </button>
        </div>
      </div>

      {/* Collision readout */}
      <AnimatePresence>
        {collisionData && (
          <motion.div
            initial={{ opacity: 0, y: 10 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0 }}
            className="border-t border-indigo-100 bg-indigo-50 p-4"
          >
            <p className="text-xs font-bold text-indigo-500 uppercase tracking-wider mb-3">
              ⚡ Collision Detected — e = {e.toFixed(2)}
            </p>
            <div className="grid grid-cols-2 gap-3">
              {[
                { label: "Body 1: Before", val: collisionData.v1Before, color: "text-slate-600" },
                { label: "Body 2: Before", val: collisionData.v2Before, color: "text-slate-600" },
                { label: "Body 1: After", val: collisionData.v1After, color: "text-indigo-700" },
                { label: "Body 2: After", val: collisionData.v2After, color: "text-indigo-700" },
              ].map((item) => (
                <div key={item.label} className="bg-white rounded-lg p-2.5 border border-indigo-100">
                  <p className="text-xs text-slate-400">{item.label}</p>
                  <p className={`text-base font-bold font-mono ${item.color}`}>
                    {item.val.toFixed(2)} m/s
                  </p>
                </div>
              ))}
            </div>
            <div className="mt-3 bg-white rounded-lg p-2.5 border border-indigo-100">
              <p className="text-xs text-slate-400 mb-1">Verification</p>
              <p className="text-xs font-mono text-slate-600">
                e_actual ={" "}
                <span className="text-indigo-600 font-bold">
                  {Math.abs(
                    (collisionData.v2After - collisionData.v1After) /
                      (collisionData.v1Before - collisionData.v2Before)
                  ).toFixed(3)}
                </span>{" "}
                (set: {e.toFixed(2)})
              </p>
            </div>
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
}

// ENERGY LOSS VISUALIZER

function EnergyLossVisualizer() {
  const [e, setE] = useState(0.7);
  const v1 = 10;
  const v2 = 0;
  const m1 = 2;
  const m2 = 2;

  const v1f = (m1 * v1 + m2 * v2 - m2 * e * (v1 - v2)) / (m1 + m2);
  const v2f = (m1 * v1 + m2 * v2 + m1 * e * (v1 - v2)) / (m1 + m2);
  const keBefore = 0.5 * m1 * v1 ** 2 + 0.5 * m2 * v2 ** 2;
  const keAfter = 0.5 * m1 * v1f ** 2 + 0.5 * m2 * v2f ** 2;
  const keLoss = keBefore - keAfter;
  const lossPercent = (keLoss / keBefore) * 100;

  return (
    <div className="bg-white rounded-xl border border-amber-200 overflow-hidden">
      <div className="bg-gradient-to-r from-amber-50 to-orange-50 px-4 py-3 border-b border-amber-100">
        <p className="text-xs font-bold text-amber-600 uppercase tracking-wider">
          ⚡ Energy Loss Visualizer
        </p>
      </div>
      <div className="p-4">
        <div className="flex items-center gap-3 mb-4">
          <span className="text-xs text-slate-500 font-medium w-8">e =</span>
          <input
            type="range"
            min={0}
            max={1}
            step={0.01}
            value={e}
            onChange={(ev) => setE(Number(ev.target.value))}
            className="flex-1 accent-amber-500"
          />
          <span className="font-mono font-bold text-amber-600 w-10">{e.toFixed(2)}</span>
        </div>

        <div className="space-y-3">
          <div>
            <div className="flex justify-between text-xs text-slate-500 mb-1">
              <span>KE Before</span>
              <span className="font-mono">{keBefore.toFixed(1)} J</span>
            </div>
            <div className="h-5 bg-emerald-100 rounded-full overflow-hidden">
              <div className="h-full bg-emerald-400 rounded-full w-full" />
            </div>
          </div>
          <div>
            <div className="flex justify-between text-xs text-slate-500 mb-1">
              <span>KE After</span>
              <span className="font-mono">{keAfter.toFixed(1)} J</span>
            </div>
            <div className="h-5 bg-emerald-100 rounded-full overflow-hidden">
              <motion.div
                className="h-full bg-emerald-400 rounded-full"
                animate={{ width: `${(keAfter / keBefore) * 100}%` }}
                transition={{ type: "spring", stiffness: 200 }}
              />
            </div>
          </div>
          <div>
            <div className="flex justify-between text-xs text-slate-500 mb-1">
              <span>KE Lost ({lossPercent.toFixed(1)}%)</span>
              <span className="font-mono text-orange-500">{keLoss.toFixed(1)} J</span>
            </div>
            <div className="h-5 bg-orange-100 rounded-full overflow-hidden">
              <motion.div
                className="h-full bg-orange-400 rounded-full"
                animate={{ width: `${lossPercent}%` }}
                transition={{ type: "spring", stiffness: 200 }}
              />
            </div>
          </div>
        </div>

        <p className="text-xs text-slate-400 mt-3 text-center">
          {e === 1
            ? "✓ Perfectly elastic — zero energy lost"
            : e === 0
            ? "✗ Perfectly inelastic — maximum energy lost"
            : `${lossPercent.toFixed(1)}% of kinetic energy converted to heat/sound`}
        </p>
      </div>
    </div>
  );
}
