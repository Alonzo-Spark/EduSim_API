// ─── KaTeX Renderer ───────────────────────────────────────────────────────────
function KaTeX({ expression, display = false }) {
  const ref = useRef(null);
  useEffect(() => {
    if (ref.current) {
      try {
        katex.render(expression, ref.current, {
          displayMode: display,
          throwOnError: false,
          strict: false,
        });
      } catch (e) {
        ref.current.textContent = expression;
      }
    }
  }, [expression, display]);
  return (
    <span
      ref={ref}
      style={{ display: display ? "block" : "inline" }}
    />
  );
}

// ─── Spring Visualizer ────────────────────────────────────────────────────────
function SpringVisualizer() {
  const [k, setK] = useState(50);
  const [x, setX] = useState(0.3);
  const canvasRef = useRef(null);
  const animRef = useRef(null);
  const tRef = useRef(0);

  const F = (k * x).toFixed(2);
  const PE = (0.5 * k * x * x).toFixed(3);
  const omega = Math.sqrt(k / 1.0).toFixed(2);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");

    const draw = () => {
      tRef.current += 0.03;
      const t = tRef.current;
      const W = canvas.width;
      const H = canvas.height;

      ctx.clearRect(0, 0, W, H);

      // Background
      ctx.fillStyle = "#0a1628";
      ctx.fillRect(0, 0, W, H);

      const anchorX = 60;
      const centerY = H / 2;
      const naturalLength = 100;
      const displacement = x * 120;
      const massX = anchorX + naturalLength + displacement;

      // Animated oscillation ghost
      const ghostDisp = Math.sin(t) * 40;
      const ghostX = anchorX + naturalLength + ghostDisp;

      // Draw ghost spring (dim)
      drawSpring(ctx, anchorX, centerY, ghostX, centerY, "#1e3a5f", 10);

      // Ghost mass
      ctx.globalAlpha = 0.25;
      ctx.fillStyle = "#6366f1";
      ctx.beginPath();
      ctx.roundRect(ghostX, centerY - 18, 36, 36, 6);
      ctx.fill();
      ctx.globalAlpha = 1;

      // Anchor wall
      ctx.fillStyle = "#334155";
      ctx.fillRect(anchorX - 20, centerY - 50, 20, 100);
      for (let i = 0; i < 6; i++) {
        ctx.strokeStyle = "#1e293b";
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(anchorX - 20, centerY - 50 + i * 17);
        ctx.lineTo(anchorX - 35, centerY - 50 + i * 17 + 10);
        ctx.stroke();
      }

      // Equilibrium dashed line
      ctx.setLineDash([4, 4]);
      ctx.strokeStyle = "#334155";
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(anchorX + naturalLength, centerY - 50);
      ctx.lineTo(anchorX + naturalLength, centerY + 50);
      ctx.stroke();
      ctx.setLineDash([]);

      // Displacement arrow
      if (Math.abs(displacement) > 2) {
        ctx.strokeStyle = "#f59e0b";
        ctx.lineWidth = 1.5;
        ctx.beginPath();
        ctx.moveTo(anchorX + naturalLength, centerY + 38);
        ctx.lineTo(massX + 18, centerY + 38);
        ctx.stroke();
        const dir = displacement > 0 ? 1 : -1;
        ctx.fillStyle = "#f59e0b";
        ctx.beginPath();
        ctx.moveTo(massX + 18 * dir, centerY + 38);
        ctx.lineTo(massX + 18 * dir - 8 * dir, centerY + 34);
        ctx.lineTo(massX + 18 * dir - 8 * dir, centerY + 42);
        ctx.closePath();
        ctx.fill();
        ctx.fillStyle = "#f59e0b";
        ctx.font = "11px monospace";
        ctx.fillText(`x = ${x.toFixed(2)} m`, anchorX + naturalLength + 4, centerY + 52);
      }

      // Main spring
      drawSpring(ctx, anchorX, centerY, massX, centerY,
        x > 0 ? "#ef4444" : x < 0 ? "#3b82f6" : "#22d3ee", 14);

      // Force arrow
      const forceDir = -Math.sign(x);
      if (Math.abs(x) > 0.05) {
        const arrowLen = Math.min(Math.abs(x) * 80, 60);
        const arrowX = massX + 18 + (forceDir > 0 ? 8 : -8);
        ctx.strokeStyle = "#4ade80";
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(arrowX, centerY);
        ctx.lineTo(arrowX + forceDir * arrowLen, centerY);
        ctx.stroke();
        ctx.fillStyle = "#4ade80";
        ctx.beginPath();
        ctx.moveTo(arrowX + forceDir * arrowLen, centerY);
        ctx.lineTo(arrowX + forceDir * arrowLen - forceDir * 10, centerY - 5);
        ctx.lineTo(arrowX + forceDir * arrowLen - forceDir * 10, centerY + 5);
        ctx.closePath();
        ctx.fill();
        ctx.font = "11px monospace";
        ctx.fillStyle = "#4ade80";
        ctx.fillText(`F = ${F} N`, arrowX + forceDir * (arrowLen / 2) - 20, centerY - 10);
      }

      // Mass block
      const grad = ctx.createLinearGradient(massX, centerY - 18, massX + 36, centerY + 18);
      grad.addColorStop(0, "#6366f1");
      grad.addColorStop(1, "#4f46e5");
      ctx.fillStyle = grad;
      ctx.beginPath();
      ctx.roundRect(massX, centerY - 18, 36, 36, 6);
      ctx.fill();
      ctx.strokeStyle = "#818cf8";
      ctx.lineWidth = 1.5;
      ctx.stroke();
      ctx.fillStyle = "#fff";
      ctx.font = "bold 12px sans-serif";
      ctx.textAlign = "center";
      ctx.fillText("m", massX + 18, centerY + 5);
      ctx.textAlign = "left";

      // Anchor point dot
      ctx.fillStyle = "#38bdf8";
      ctx.beginPath();
      ctx.arc(anchorX, centerY, 5, 0, Math.PI * 2);
      ctx.fill();

      animRef.current = requestAnimationFrame(draw);
    };

    animRef.current = requestAnimationFrame(draw);
    return () => cancelAnimationFrame(animRef.current);
  }, [k, x]);

  return (
    <div style={{
      background: "#0a1628", border: "1px solid #1e293b",
      borderRadius: 14, overflow: "hidden"
    }}>
      <div style={{
        padding: "10px 18px", background: "rgba(99,102,241,0.1)",
        borderBottom: "1px solid #1e293b",
        display: "flex", alignItems: "center", gap: 8
      }}>
        <span>🎮</span>
        <span style={{ color: "#818cf8", fontWeight: 700, fontSize: 14 }}>
          Live Spring Simulator
        </span>
      </div>

      <canvas
        ref={canvasRef}
        width={460}
        height={160}
        style={{ width: "100%", display: "block" }}
      />

      <div style={{ padding: "16px 20px" }}>
        <div style={{ display: "flex", gap: 20, marginBottom: 16 }}>
          <div style={{ flex: 1 }}>
            <label style={{ color: "#94a3b8", fontSize: 12, display: "block", marginBottom: 6 }}>
              Spring Constant <KaTeX expression="k" /> = <strong style={{ color: "#38bdf8" }}>{k} N/m</strong>
            </label>
            <input type="range" min={5} max={200} value={k}
              onChange={(e) => setK(Number(e.target.value))}
              style={{ width: "100%", accentColor: "#38bdf8" }} />
          </div>
          <div style={{ flex: 1 }}>
            <label style={{ color: "#94a3b8", fontSize: 12, display: "block", marginBottom: 6 }}>

          </div>
        </div>

        <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr 1fr", gap: 10 }}>
          {[
            { label: "Restoring Force", val: `${F} N`, color: "#4ade80" },
            { label: "Potential Energy", val: `${PE} J`, color: "#f59e0b" },
            { label: "Angular Frequency", val: `${omega} rad/s`, color: "#a78bfa" },
          ].map((item) => (

            </div>

function drawSpring(ctx, x1, y1, x2, y2, color, coils) {
  const dx = x2 - x1;
  const len = Math.sqrt(dx * dx);
  const coilWidth = 10;
  const segLen = len / (coils * 2 + 2);

  ctx.strokeStyle = color;
  ctx.lineWidth = 2;
  ctx.beginPath();
  ctx.moveTo(x1, y1);
  ctx.lineTo(x1 + segLen, y1);

  for (let i = 0; i < coils; i++) {
    ctx.lineTo(x1 + segLen + i * 2 * segLen + segLen / 2, y1 - coilWidth);
    ctx.lineTo(x1 + segLen + i * 2 * segLen + segLen * 1.5, y1 + coilWidth);
  }

  ctx.lineTo(x2, y2);
  ctx.stroke();
}
