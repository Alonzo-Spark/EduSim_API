    <script>
        // 1. Initialize KaTeX Rendering
        function initKatex() {
            renderMathInElement(document.body, {
                delimiters: [
                    {left: '$$', right: '$$', display: true},
                    {left: '$', right: '$', display: false}
                ],
                throwOnError: false
            });
        }

        // 2. Formula Data Extraction (Simulated as per request structure)
        const formulaData = {
            formula: "PE = m \\cdot g \\cdot h",
            title: "Gravitational Potential Energy",
            variables: [
                { name: "Mass", symbol: "m", unit: "kg" },
                { name: "Gravity", symbol: "g", unit: "m/s²" },
                { name: "Height", symbol: "h", unit: "m" }
            ]
        };

        // 3. UI Interaction Logic
        
        // Toggle Explanation
        function toggleExplanation() {
            const el = document.getElementById('formula-explanation');
            el.classList.toggle('active');
        }

        // Formula Lab Logic
        const modal = document.getElementById('lab-modal');
        const inputs = {
            m: document.getElementById('lab-mass'),
            h: document.getElementById('lab-height'),
            g: document.getElementById('lab-gravity')
        };
        const displays = {
            m: document.getElementById('val-mass'),
            h: document.getElementById('val-height'),
            g: document.getElementById('val-gravity')
        };
        const resultDisplay = document.getElementById('lab-result');

        function calculateLab() {
            const m = parseFloat(inputs.m.value);
            const h = parseFloat(inputs.h.value);
            const g = parseFloat(inputs.g.value);

            // Update Text Displays
            displays.m.textContent = m;
            displays.h.textContent = h;
            displays.g.textContent = g;

            // Calculate PE
            const pe = m * g * h;
            
            // Format Result
            resultDisplay.textContent = pe.toLocaleString(undefined, { maximumFractionDigits: 1 }) + " J";
        }

        function openFormulaLab() {
            modal.classList.add('open');
            // Reset to problem defaults for better UX
            inputs.m.value = 5;
            inputs.h.value = 10;
            inputs.g.value = 9.8;
            calculateLab();
        }

        function closeFormulaLab() {
            modal.classList.remove('open');
        }

        // Close modal on outside click
        modal.addEventListener('click', (e) => {
            if (e.target === modal) closeFormulaLab();
        });

        // Attach listeners
        Object.values(inputs).forEach(input => {
            input.addEventListener('input', calculateLab);
        });

    </script>
