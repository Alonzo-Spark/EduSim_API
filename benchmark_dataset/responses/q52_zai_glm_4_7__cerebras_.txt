<script>
  const canvas = document.getElementById('simCanvas');
  const ctx = canvas.getContext('2d');
  const vInput = document.getElementById('velocity');
  const aInput = document.getElementById('angle');
  const dInput = document.getElementById('drag');
  const fireBtn = document.getElementById('fire-btn');

  // Display values next to sliders
  vInput.oninput = () => document.getElementById('val-velocity').textContent = vInput.value;
  aInput.oninput = () => document.getElementById('val-angle').textContent = aInput.value;
  dInput.oninput = () => {
    const val = parseFloat(dInput.value).toFixed(3);
    document.getElementById('val-drag').textContent = val;
  };

  // Simulation parameters
  const g = 9.81; // Gravity (m/s^2)
  const dt = 0.05; // Time step (s)
  const scale = 5; // Pixels per meter

  function draw() {
    // Clear canvas
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // Draw ground
    ctx.fillStyle = '#4CAF50';
    ctx.fillRect(0, canvas.height - 20, canvas.width, 20);

    // Get parameters from inputs
    const v0 = parseFloat(vInput.value);
    const theta = parseFloat(aInput.value) * Math.PI / 180;
    const Cd = parseFloat(dInput.value);

    // --- Calculate Ideal Trajectory (No Drag) ---
    ctx.beginPath();
    ctx.strokeStyle = '#aaa';
    ctx.setLineDash([5, 5]);
    ctx.lineWidth = 2;
    
    let xIdeal = 0;
    let yIdeal = 0;
    let vxIdeal = v0 * Math.cos(theta);
    let vyIdeal = v0 * Math.sin(theta);
    
    ctx.moveTo(0, canvas.height - 20 - yIdeal * scale);
    
    while (yIdeal >= 0) {
      vxIdeal = vxIdeal; // No horizontal acceleration
      vyIdeal -= g * dt;
      
      xIdeal += vxIdeal * dt;
      yIdeal += vyIdeal * dt;
      
      if (yIdeal < 0) break;
      ctx.lineTo(xIdeal * scale, canvas.height - 20 - yIdeal * scale);
    }
    ctx.stroke();
    ctx.setLineDash([]); // Reset dash

    // --- Calculate Real Trajectory (With Drag) ---
    ctx.beginPath();
    ctx.strokeStyle = '#007bff';
    ctx.lineWidth = 3;
    
    let x = 0;
    let y = 0;
    let vx = v0 * Math.cos(theta);
    let vy = v0 * Math.sin(theta);
    const mass = 1; // Mass of the projectile (kg)
    
    ctx.moveTo(0, canvas.height - 20 - y * scale);
    
    while (y >= 0) {
      const v = Math.sqrt(vx*vx + vy*vy);
      
      // Drag force: Fd = (1/2) * rho * v^2 * Cd * A
      // We simplify constants into a single 'dragFactor' for simulation
      const dragFactor = 0.5 * 1.225 * Cd * 0.01; // rho=1.225 (air density), A=0.01 (area)
      const Fd = dragFactor * v * v;
      
      // Drag acts opposite to velocity
      const Fx = -Fd * (vx / v);
      const Fy = -Fd * (vy / v);
      
      // Acceleration: F = ma => a = F/m
      const ax = Fx / mass;
      const ay = -g + (Fy / mass);
      
      vx += ax * dt;
      vy += ay * dt;
      
      x += vx * dt;
      y += vy * dt;
      
      if (y < 0) break;
      
      // Stop drawing if out of bounds
      if (x * scale > canvas.width) break;
      
      ctx.lineTo(x * scale, canvas.height - 20 - y * scale);
    }
    ctx.stroke();

    // Draw projectile at start
    ctx.beginPath();
    ctx.fillStyle = '#333';
    ctx.arc(0, canvas.height - 20, 5, 0, Math.PI * 2);
    ctx.fill();
  }

  // Initial draw
  draw();

  // Re-draw on button click or input change
  fireBtn.addEventListener('click', draw);
  vInput.addEventListener('input', draw);
  aInput.addEventListener('input', draw);
  dInput.addEventListener('input', draw);

</script>
</body>

**Formula Explanation:**

*   **$F_d$**: Drag Force (Newtons). This is the force that acts against the projectile's motion.
*   **$\rho$ (rho)**: Air Density (kg/m³). Denser air, like at sea level, creates more drag than thinner air, like at high altitudes.
*   **$v$**: Velocity of the projectile relative to the air (m/s). Notice that drag is proportional to the *square* of velocity. This means doubling the speed increases drag by four times.
*   **$C_d$**: Drag Coefficient. A dimensionless number that represents the object's shape and surface properties. A streamlined shape has a low $C_d$, while a flat or irregular shape has a high $C_d$.
*   **$A$**: Reference Area (m²). The cross-sectional area of the object facing the airflow. A larger area means more air to push out of the way, leading to more drag.
