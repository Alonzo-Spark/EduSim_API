import { useState, useRef, useEffect } from "react";

const katex = window.katex;

function renderLatex(str: string, display = false): string {
  try {
    return katex.renderToString(str, { throwOnError: false, displayMode: display });
  } catch {
    return str;
  }
}

function InlineMath({ math }: { math: string }) {
  return <span dangerouslySetInnerHTML={{ __html: renderLatex(math, false) }} />;
}

function DisplayMath({ math }: { math: string }) {
  return <div dangerouslySetInnerHTML={{ __html: renderLatex(math, true) }} />;
}

// ─── Physics simulation ────────────────────────────────────────────────────────
function simulateTrajectory(v0: number, angleDeg: number, k: number, g = 9.81, dt = 0.02) {
  const rad = (angleDeg * Math.PI) / 180;
  let vx = v0 * Math.cos(rad);
  let vy = v0 * Math.sin(rad);
  let x = 0, y = 0;
  const pts: [number, number][] = [[0, 0]];
  for (let i = 0; i < 5000; i++) {
    const v = Math.sqrt(vx * vx + vy * vy);
    const ax = -k * v * vx;
    const ay = -g - k * v * vy;
    vx += ax * dt;
    vy += ay * dt;
    x += vx * dt;
    y += vy * dt;
    if (y < 0 && i > 2) break;
    pts.push([x, y]);
  }
  return pts;
}

function drawTrajectories(canvas: HTMLCanvasElement | null, v0: number, angle: number, k: number) {
  if (!canvas) return;
  const ctx = canvas.getContext("2d");
  if (!ctx) return;
  const W = canvas.width, H = canvas.height;
  ctx.clearRect(0, 0, W, H);

  // Grid
  ctx.strokeStyle = "#1e293b";
  ctx.lineWidth = 1;
  for (let x = 0; x < W; x += 40) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke(); }
  for (let y = 0; y < H; y += 40) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke(); }

  const ideal = simulateTrajectory(v0, angle, 0);
  const drag = simulateTrajectory(v0, angle, k);

  const allPts = [...ideal, ...drag];
  const maxX = Math.max(...allPts.map(p => p[0]), 1);
  const maxY = Math.max(...allPts.map(p => p[1]), 1);
  const pad = 30;
  const scaleX = (W - pad * 2) / maxX;
  const scaleY = (H - pad * 2) / maxY;

  function toCanvas(x: number, y: number): [number, number] {
    return [pad + x * scaleX, H - pad - y * scaleY];
  }

  function drawPath(pts: [number, number][], color: string, dash: number[] = []) {
    ctx!.beginPath();
    ctx!.setLineDash(dash);
    ctx!.strokeStyle = color;
    ctx!.lineWidth = 2.5;
    pts.forEach(([x, y], i) => {
      const [cx, cy] = toCanvas(x, y);
      i === 0 ? ctx!.moveTo(cx, cy) : ctx!.lineTo(cx, cy);
    });
    ctx!.stroke();
    ctx!.setLineDash([]);
  }

  drawPath(ideal, "#6366f1", [6, 4]);
  drawPath(drag, "#f97316");

  // Ground line
  ctx.strokeStyle = "#334155";
  ctx.lineWidth = 1.5;
  ctx.beginPath();
  ctx.moveTo(pad, H - pad);
  ctx.lineTo(W - pad, H - pad);
  ctx.stroke();

  // Apex markers
  const idealApex = ideal.reduce((a, b) => b[1] > a[1] ? b : a);
  const dragApex = drag.reduce((a, b) => b[1] > a[1] ? b : a);

  function drawApex(pt: [number, number], color: string) {
    const [cx, cy] = toCanvas(pt[0], pt[1]);
    ctx!.beginPath();
    ctx!.arc(cx, cy, 4, 0, Math.PI * 2);
    ctx!.fillStyle = color;
    ctx!.fill();
  }
  drawApex(idealApex, "#6366f1");
  drawApex(dragApex, "#f97316");

  // Range labels
  const idealRange = ideal[ideal.length - 1][0];
  const dragRange = drag[drag.length - 1][0];
  ctx.font = "11px monospace";
  ctx.fillStyle = "#6366f1";
  ctx.fillText(`R = ${idealRange.toFixed(1)}m`, pad + idealRange * scaleX - 40, H - pad + 18);
  ctx.fillStyle = "#f97316";
  ctx.fillText(`R = ${dragRange.toFixed(1)}m`, pad + dragRange * scaleX - 40, H - pad + 30);
}
