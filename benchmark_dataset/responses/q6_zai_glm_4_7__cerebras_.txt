    <script>
        // --- Interaction Logic ---

        function toggleExplanation(btn) {
            const card = btn.closest('.card');
            const panel = card.querySelector('.explanation-panel');
            const icon = btn.querySelector('span');
            
            if (panel.classList.contains('active')) {
                panel.classList.remove('active');
                icon.innerText = "ℹ️ Explain Formula";
            } else {
                // Close others
                document.querySelectorAll('.explanation-panel').forEach(p => p.classList.remove('active'));
                document.querySelectorAll('.btn-explain span').forEach(s => s.innerText = "ℹ️ Explain Formula");
                
                panel.classList.add('active');
                icon.innerText = "🔽 Hide Explanation";
            }
        }

        // --- Formula Lab / Simulation Logic ---

        const modal = document.getElementById('labModal');
        const canvas = document.getElementById('simCanvas');
        const ctx = canvas.getContext('2d');
        const angleInput = document.getElementById('angleRange');
        const velocityInput = document.getElementById('velocityRange');
        
        // Simulation State
        let animationId = null;
        let projectile = { x: 0, y: 0, vx: 0, vy: 0, active: false };
        let scale = 1; // pixels per meter
        const g = 9.81;
        
        // Resize canvas to fit container
        function resizeCanvas() {
            canvas.width = canvas.parentElement.clientWidth;
            canvas.height = canvas.parentElement.clientHeight;
            drawScene();
        }
        window.addEventListener('resize', resizeCanvas);

        function openLab(focusMode) {
            modal.classList.add('active');
            resizeCanvas();
            
            // Pre-set values based on context if needed, for now just default
            runSimulation(); 
        }

        function closeLab() {
            modal.classList.remove('active');
            if (animationId) cancelAnimationFrame(animationId);
            projectile.active = false;
        }

        // Update slider labels
        angleInput.addEventListener('input', (e) => {
            document.getElementById('angleVal').innerText = e.target.value;
            runSimulation();
        });
        velocityInput.addEventListener('input', (e) => {
            document.getElementById('velocityVal').innerText = e.target.value;
            runSimulation();
        });

        function calculatePhysics(u, thetaDeg) {
            const thetaRad = thetaDeg * (Math.PI / 180);
            const T = (2 * u * Math.sin(thetaRad)) / g;
            const H = (u * u * Math.sin(thetaRad) * Math.sin(thetaRad)) / (2 * g);
            const R = (u * u * Math.sin(2 * thetaRad)) / g;
            return { T, H, R, thetaRad };
        }

        function updateTable(u, thetaDeg) {
            const { T, H, R } = calculatePhysics(u, thetaDeg);
            document.getElementById('resTime').innerText = T.toFixed(2) + " s";
            document.getElementById('resHeight').innerText = H.toFixed(2) + " m";
            document.getElementById('resRange').innerText = R.toFixed(2) + " m";
        }

        function runSimulation() {
            if (animationId) cancelAnimationFrame(animationId);
            
            const u = parseFloat(velocityInput.value);
            const thetaDeg = parseFloat(angleInput.value);
            const thetaRad = thetaDeg * (Math.PI / 180);
            
            // Update Table
            updateTable(u, thetaDeg);

            // Calculate theoretical max to scale the view
            const { H, R } = calculatePhysics(u, thetaDeg);
            
            // Auto-scale logic: fit range and height with padding
            const scaleX = (canvas.width - 40) / R;
            const scaleY = (canvas.height - 40) / H;
            scale = Math.min(scaleX, scaleY, 10); // Limit max zoom
            
            // Initialize Projectile
            projectile.x = 0;
            projectile.y = 0;
            projectile.vx = u * Math.cos(thetaRad);
            projectile.vy = u * Math.sin(thetaRad);
            projectile.active = true;
            projectile.path = [{x:0, y:0}]; // Store path for drawing
            
            const startTime = performance.now();
            const timeScale = 0.05; // Slow down simulation for visual clarity

            function animate(time) {
                if (!projectile.active) return;
                
                const dt = (time - startTime) * 0.001 * 5; // speed multiplier
                
                // Physics update (Euler integration is fine for visualization)
                const t = dt; 
                const nextX = u * Math.cos(thetaRad) * t;
                const nextY = (u * Math.sin(thetaRad) * t) - (0.5 * g * t * t);

                if (nextY < 0) {
                    projectile.active = false;
                    projectile.x = nextX;
                    projectile.y = 0;
                } else {
                    projectile.x = nextX;
                    projectile.y = nextY;
                    projectile.path.push({x: nextX, y: nextY});
                    animationId = requestAnimationFrame(animate);
                }
                
                drawScene();
            }
            
            animationId = requestAnimationFrame(animate);
        }

        function drawScene() {
            // Clear
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // Coordinate system transform (Origin at bottom-left)
            ctx.save();
            ctx.translate(20, canvas.height - 20); 
            
            // Draw Axes
            ctx.beginPath();
            ctx.strokeStyle = '#cbd5e1';
            ctx.lineWidth = 2;
            // X Axis
            ctx.moveTo(0, 0);
            ctx.lineTo(canvas.width - 20, 0);
            // Y Axis
            ctx.moveTo(0, 0);
            ctx.lineTo(0, -(canvas.height - 20));
            ctx.stroke();

            // Draw Parabolic Path (Ghost line for full trajectory)
            const u = parseFloat(velocityInput.value);
            const thetaDeg = parseFloat(angleInput.value);
            const thetaRad = thetaDeg * (Math.PI / 180);
            const { R, H } = calculatePhysics(u, thetaDeg);
            
            ctx.beginPath();
            ctx.strokeStyle = '#e2e8f0';
            ctx.setLineDash([5, 5]);
            ctx.moveTo(0, 0);
            
            for(let x = 0; x <= R; x += R/50) {
                const t = x / (u * Math.cos(thetaRad));
                const y = (u * Math.sin(thetaRad) * t) - (0.5 * g * t * t);
                ctx.lineTo(x * scale, -y * scale);
            }
            ctx.stroke();
            ctx.setLineDash([]);

            // Draw Actual Path Traveled
            if (projectile.path.length > 1) {
                ctx.beginPath();
                ctx.strokeStyle = '#3b82f6';
                ctx.lineWidth = 3;
                ctx.moveTo(0, 0);
                for (let p of projectile.path) {
                    ctx.lineTo(p.x * scale, -p.y * scale);
                }
                ctx.stroke();
            }

            // Draw Projectile Ball
            const drawX = projectile.x * scale;
            const drawY = -projectile.y * scale;
            
            ctx.beginPath();
            ctx.fillStyle = '#ef4444';
            ctx.arc(drawX, drawY, 6, 0, Math.PI * 2);
            ctx.fill();

            // Draw Cannon (Visual only)
            ctx.beginPath();
            ctx.strokeStyle = '#64748b';
            ctx.lineWidth = 4;
            ctx.moveTo(0, 0);
            const barrelLen = 30;
            ctx.lineTo(barrelLen * Math.cos(thetaRad), -barrelLen * Math.sin(thetaRad));
            ctx.stroke();

            ctx.restore();
        }

        // Initialize KaTeX rendering listener
        document.addEventListener("DOMContentLoaded", () => {
            // KaTeX auto-render is handled by the onload attribute in the script tag
            resizeCanvas();
        });
