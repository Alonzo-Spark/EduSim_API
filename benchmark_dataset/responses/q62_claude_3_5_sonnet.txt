const G = 9.81;
const RAD = Math.PI / 180;

function computePendulumPhysics(state: PendulumState): PendulumState {
  const { angle, angularVelocity, length, mass, attachedObjectMass } = state;
  const totalMass = mass + attachedObjectMass;
  const alpha = -(G / length) * Math.sin(angle);
  const omega = angularVelocity + alpha * 0.016;
  const theta = angle + omega * 0.016;
  const v = omega * length;
  const vx = v * Math.cos(theta);
  const vy = v * Math.sin(theta);
  const tension = totalMass * ((v * v) / length + G * Math.cos(theta));
  const px =
    state.pivotPosition.x + length * Math.sin(theta) * 120;
  const py =
    state.pivotPosition.y + length * Math.cos(theta) * 120;
  return {
    ...state,
    angle: theta,
    angularVelocity: omega,
    angularAcceleration: alpha,
    tension,
    velocity: { x: vx, y: vy },
    position: { x: px, y: py },
  };
}

function FormulaLab({ data }: { data: FormulaData }) {
  const [inputs, setInputs] = useState<Record<string, number>>({
    m1: 2.0,
    m2: 0.5,
    v0: 4.2,
    L: 1.5,
  });

  const momentum = inputs.m1 * inputs.v0;
  const vFinal = momentum / (inputs.m1 + inputs.m2);
  const ke_before = 0.5 * inputs.m1 * inputs.v0 ** 2;
  const ke_after = 0.5 * (inputs.m1 + inputs.m2) * vFinal ** 2;
  const keLost = ke_before - ke_after;
  const maxAngle =
    (Math.acos(1 - (vFinal ** 2) / (2 * G * inputs.L)) * 180) / Math.PI;

  return (
    <div className="lab-panel">
      <div className="lab-header">
        <span>⚗</span>
        <span>Formula Lab — Collision Dynamics</span>
      </div>

      <div className="lab-inputs">
        {[
          { key: "m1", label: "Bob Mass (kg)", min: 0.1, max: 10, step: 0.1 },
          {
            key: "m2",
            label: "Object Mass (kg)",
            min: 0.1,
            max: 10,
            step: 0.1,
          },
          {
            key: "v0",
            label: "Impact Velocity (m/s)",
            min: 0.1,
            max: 20,
            step: 0.1,
          },
          { key: "L", label: "String Length (m)", min: 0.5, max: 5, step: 0.1 },
        ].map(({ key, label, min, max, step }) => (
          <div key={key} className="lab-input-row">
            <label>{label}</label>
            <input
              type="range"
              min={min}
              max={max}
              step={step}
              value={inputs[key]}
              onChange={(e) =>
                setInputs((p) => ({ ...p, [key]: parseFloat(e.target.value) }))
              }
            />
            <span className="lab-value">{inputs[key].toFixed(2)}</span>
          </div>
        ))}
      </div>

      <div className="lab-results">
        <div className="lab-result-row">
          <span>Final Velocity</span>

        </div>

function SectionCard({
  block,
  liveState,
}: {
  block: OverlayBlock;
  liveState?: PendulumState;
}) {
  const iconMap: Record<string, string> = {
    definition: "📖",
    formula: "∑",
    table: "📊",
    note: "⚠",
    summary: "✦",
    application: "🔬",
    characteristics: "⚙",
  };

  if (block.type === "formula" && block.formulaData) {
    const liveValue =
      liveState && block.formulaData.title.includes("Momentum")
        ? `p = ${(liveState.mass * Math.hypot(liveState.velocity.x, liveState.velocity.y)).toFixed(2)} kg·m/s`
        : liveState && block.formulaData.title.includes("Tension")
        ? `T = ${liveState.tension.toFixed(2)} N`
        : undefined;

    return <FormulaCard data={block.formulaData} live={liveValue} />;
  }

  if (block.type === "table" && block.tableData) {
    return (
      <TableCard
        title={block.title}
        headers={block.tableData.headers}
        rows={block.tableData.rows}
      />
    );
  }

  return (
    <div className={`section-card ${block.highlight ? "highlighted" : ""} ${block.type}`}>
      <div className="section-card-header">
        <span className="section-icon">{iconMap[block.type] || "•"}</span>
        <span className="section-title">{block.title}</span>
      </div>
      <div className="section-body">
        <p>{block.content}</p>
      </div>
    </div>
  );
}

function PendulumCanvas({ state }: { state: PendulumState }) {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    const W = canvas.width;
    const H = canvas.height;
    const px = W / 2;
    const py = 60;
    const L = state.length * 100;
    const bx = px + L * Math.sin(state.angle);
    const by = py + L * Math.cos(state.angle);

    ctx.clearRect(0, 0, W, H);

    // Background grid
    ctx.strokeStyle = "rgba(255,255,255,0.04)";
    ctx.lineWidth = 1;
    for (let gx = 0; gx < W; gx += 30) {
      ctx.beginPath();
      ctx.moveTo(gx, 0);
      ctx.lineTo(gx, H);
      ctx.stroke();
    }
    for (let gy = 0; gy < H; gy += 30) {
      ctx.beginPath();
      ctx.moveTo(0, gy);
      ctx.lineTo(W, gy);
      ctx.stroke();
    }

    // Pivot
    ctx.beginPath();
    ctx.arc(px, py, 8, 0, Math.PI * 2);
    ctx.fillStyle = "#94a3b8";
    ctx.fill();

    // String
    const grad = ctx.createLinearGradient(px, py, bx, by);
    grad.addColorStop(0, "#60a5fa");
    grad.addColorStop(1, "#a78bfa");
    ctx.beginPath();
    ctx.moveTo(px, py);
    ctx.lineTo(bx, by);
    ctx.strokeStyle = grad;
    ctx.lineWidth = 2.5;
    ctx.stroke();

    // Bob glow
    const glow = ctx.createRadialGradient(bx, by, 0, bx, by, 28);
    glow.addColorStop(0, "rgba(251, 146, 60, 0.4)");
    glow.addColorStop(1, "rgba(251, 146, 60, 0)");
    ctx.beginPath();
    ctx.arc(bx, by, 28, 0, Math.PI * 2);
    ctx.fillStyle = glow;
    ctx.fill();

    // Bob
    ctx.beginPath();
    ctx.arc(bx, by, 14, 0, Math.PI * 2);
    const bobGrad = ctx.createRadialGradient(bx - 4, by - 4, 2, bx, by, 14);
    bobGrad.addColorStop(0, "#fbbf24");
    bobGrad.addColorStop(1, "#f97316");
    ctx.fillStyle = bobGrad;
    ctx.fill();
    ctx.strokeStyle = "#fff3";
    ctx.lineWidth = 1.5;
    ctx.stroke();

    // Attached object
    if (state.attachedObjectMass > 0) {
      const ox = bx + 20;
      const oy = by - 10;
      ctx.beginPath();
      ctx.rect(ox - 10, oy - 10, 20, 20);
      ctx.fillStyle = "#34d399";
      ctx.fill();
      ctx.strokeStyle = "#fff3";
      ctx.lineWidth = 1.5;
      ctx.stroke();

      // Attachment line
      ctx.beginPath();
      ctx.moveTo(bx + 14, by);
      ctx.lineTo(ox - 10, oy);
      ctx.strokeStyle = "#34d39988";
      ctx.lineWidth = 1.5;
      ctx.setLineDash([3, 3]);
      ctx.stroke();
      ctx.setLineDash([]);
    }

    // Velocity vector
    const spd = Math.hypot(state.velocity.x, state.velocity.y);
    if (spd > 0.01) {
      const scale = 12;
      const vx = (state.velocity.x / spd) * scale * spd;
      const vy = (state.velocity.y / spd) * scale * spd;
      ctx.beginPath();
      ctx.moveTo(bx, by);
      ctx.lineTo(bx + vx, by + vy);
      ctx.strokeStyle = "#60a5fa";
      ctx.lineWidth = 2;
      ctx.stroke();

      // Arrow head
      const angle = Math.atan2(vy, vx);
      ctx.beginPath();
      ctx.moveTo(bx + vx, by + vy);
      ctx.lineTo(
        bx + vx - 8 * Math.cos(angle - 0.4),
        by + vy - 8 * Math.sin(angle - 0.4)
      );
      ctx.lineTo(
        bx + vx - 8 * Math.cos(angle + 0.4),
        by + vy - 8 * Math.sin(angle + 0.4)
      );
      ctx.closePath();
      ctx.fillStyle = "#60a5fa";
      ctx.fill();
    }

    // Labels
    ctx.font = "11px monospace";
    ctx.fillStyle = "#94a3b8";
    ctx.fillText(`θ = ${((state.angle * 180) / Math.PI).toFixed(1)}°`, px + 12, py + 16);
    ctx.fillText(`v = ${spd.toFixed(2)} m/s`, bx + 18, by - 18);
  }, [state]);

  return (
    <canvas
      ref={canvasRef}
      width={320}
      height={260}
      className="pendulum-canvas"
    />
  );
}

## What This Builds

### Runtime Architecture

| Layer | Role |
|---|---|
| **Physics Engine** | 60fps pendulum simulation with post-collision state |
| **Event Handler** | `object_attached` triggers overlay mount + telemetry |
| **Block Renderer** | Converts detected sections → typed UI cards |
| **Formula Lab** | Interactive sliders computing live collision outcomes |

### UI Block System

- **`SectionCard`** — Definition, Note, Summary, Application, Characteristics
- **`FormulaCard`** — Every `$$...$$` becomes a dedicated card with Explain + Lab
- **`TableCard`** — Responsive scrollable table inside a card
- **`LiveTelemetry`** — 60fps real-time physics metrics
- **`PendulumCanvas`** — Canvas simulation with velocity vectors + glow

### Formula Lab Features
- Sliders for `m₁`, `m₂`, `v₀`, `L`
- Live computes: final velocity, KE before/after, KE lost, max swing angle
- All derived from the extracted formula context — no re-parsing
