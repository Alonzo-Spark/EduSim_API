import { useState } from "react";
import katex from "katex";

const KTeX = ({ math, display = false }) => {
  const html = katex.renderToString(math, {
    throwOnError: false,
    displayMode: display,
  });
  return (
    <span
      dangerouslySetInnerHTML={{ __html: html }}
      className={display ? "block my-2" : "inline"}
    />
  );
};

const FormulaCard = ({ title, formula, variables, context }) => {
  const [showExplain, setShowExplain] = useState(false);
  const [showLab, setShowLab] = useState(false);
  const [labValues, setLabValues] = useState({});
  const [labResult, setLabResult] = useState(null);

  const computeLab = () => {
    if (title === "Wave Speed") {
      const f = parseFloat(labValues.f);
      const λ = parseFloat(labValues.λ);
      if (!isNaN(f) && !isNaN(λ)) setLabResult((f * λ).toFixed(3));
    } else if (title === "Wave Period") {
      const f = parseFloat(labValues.f);
      if (!isNaN(f) && f !== 0) setLabResult((1 / f).toFixed(4));
    }
  };

  return (
    <div className="rounded-2xl border border-indigo-200 bg-gradient-to-br from-indigo-50 to-white shadow-md overflow-hidden">
      <div className="bg-indigo-600 px-5 py-2 flex items-center gap-2">
        <span className="text-white font-semibold text-sm tracking-wide">⚗️ {title}</span>
      </div>
      <div className="px-6 py-5">
        <div className="bg-white rounded-xl border border-indigo-100 px-4 py-4 text-center shadow-inner mb-4">
          <KTeX math={formula} display={true} />
        </div>
        <div className="flex gap-2 flex-wrap">
          <button
            onClick={() => { setShowExplain(!showExplain); setShowLab(false); }}
            className="px-4 py-1.5 rounded-full bg-indigo-100 hover:bg-indigo-200 text-indigo-700 text-xs font-semibold transition"
          >
            {showExplain ? "▲ Hide Explanation" : "▼ Explain Formula"}
          </button>
          {(title === "Wave Speed" || title === "Wave Period") && (
            <button
              onClick={() => { setShowLab(!showLab); setShowExplain(false); setLabResult(null); }}
              className="px-4 py-1.5 rounded-full bg-emerald-100 hover:bg-emerald-200 text-emerald-700 text-xs font-semibold transition"
            >
              🧪 Open Formula Lab
            </button>
          )}
        </div>

        {showExplain && (
          <div className="mt-4 bg-indigo-50 rounded-xl p-4 border border-indigo-100 text-sm text-slate-700 space-y-1">
            <p className="font-semibold text-indigo-700 mb-2">Variable Guide</p>
            {variables.map((v, i) => (
              <div key={i} className="flex gap-2">
                <span className="font-mono text-indigo-600 w-8 shrink-0"><KTeX math={v.sym} /></span>
                <span className="text-slate-500">—</span>
                <span>{v.desc}</span>
              </div>
            ))}
            {context && <p className="mt-3 text-slate-600 italic border-t border-indigo-100 pt-2">{context}</p>}
          </div>
        )}

        {showLab && (
          <div className="mt-4 bg-emerald-50 rounded-xl p-4 border border-emerald-100 text-sm space-y-3">
            <p className="font-semibold text-emerald-700">🧪 Formula Lab — Try it yourself</p>
            {title === "Wave Speed" && (
              <>
                <div className="flex gap-3 flex-wrap">
                  <label className="flex flex-col gap-1">
                    <span className="text-xs text-slate-500">Frequency <KTeX math="f" /> (Hz)</span>
                    <input type="number" placeholder="e.g. 440" className="border rounded-lg px-3 py-1.5 w-36 text-sm focus:outline-indigo-400"
                      onChange={e => setLabValues(p => ({ ...p, f: e.target.value }))} />
                  </label>
                  <label className="flex flex-col gap-1">
                    <span className="text-xs text-slate-500">Wavelength <KTeX math="\lambda" /> (m)</span>
                    <input type="number" placeholder="e.g. 0.78" className="border rounded-lg px-3 py-1.5 w-36 text-sm focus:outline-indigo-400"
                      onChange={e => setLabValues(p => ({ ...p, λ: e.target.value }))} />
                  </label>
                </div>
                <button onClick={computeLab} className="px-4 py-1.5 bg-emerald-500 hover:bg-emerald-600 text-white text-xs font-bold rounded-full transition">
                  Calculate Wave Speed
                </button>
              </>
            )}
            {title === "Wave Period" && (
              <>
                <label className="flex flex-col gap-1 w-fit">
                  <span className="text-xs text-slate-500">Frequency <KTeX math="f" /> (Hz)</span>
                  <input type="number" placeholder="e.g. 50" className="border rounded-lg px-3 py-1.5 w-36 text-sm focus:outline-indigo-400"
                    onChange={e => setLabValues(p => ({ ...p, f: e.target.value }))} />
                </label>
                <button onClick={computeLab} className="px-4 py-1.5 bg-emerald-500 hover:bg-emerald-600 text-white text-xs font-bold rounded-full transition">
