```python
import markdown
import re

def parse_markdown(markdown_text):
    """Parses markdown and converts it into a structured data format."""

    sections = []
    current_section = None
    lines = markdown_text.splitlines()

    for line in lines:
        if line.startswith("# "):  # Section Heading
            if current_section:
                sections.append(current_section)
            current_section = {
                "title": line[2:],
                "content": "",
                "type": "Section"  # Or "Introduction", "Definition", etc.
            }
        elif current_section:
            current_section["content"] += line + "\n"
        else:
            # Handle content outside of sections (e.g., introductory text)
            pass # or append to a default section

    if current_section:
        sections.append(current_section)

    #Formula Extraction
    formulas = []
    for section in sections:
        latex_matches = re.findall(r'\$\$(.*?)\$\$', section["content"])
        for formula_latex in latex_matches:
            formulas.append({
                "formula": formula_latex,
                "title": section["title"],
                "surroundingContext": section["content"],
                "variables": [] #TODO: Implement variable extraction
            })

    return sections, formulas


# Example Usage
markdown_content = """
# Introduction

This is an introduction to physics.  $$F=ma$$

# Newton's Second Law

Newton's Second Law of Motion describes...

$$v = u + at$$
"""

sections, formulas = parse_markdown(markdown_content)

print("Sections:", sections)
print("Formulas:", formulas)
```
