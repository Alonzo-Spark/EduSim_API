import re

def extract_formulas(markdown_text):
    formulas = []
    lines = markdown_text.splitlines()
    for line in lines:
      if line.startswith("$$"):
        formula = line[2:-2]  # Extract the LaTeX string
        match = re.search(r"([a-zA-Z0-9]+)", formula)  #Extract variables.
        variables = [match.group(1) for match in re.finditer(r"[a-zA-Z0-9]+", formula)] # Find all variables.
        formulas.append({
            "formula": formula,
            "title": "Formula",
            "surroundingContext": line,
            "variables": variables
        })
    return formulas
