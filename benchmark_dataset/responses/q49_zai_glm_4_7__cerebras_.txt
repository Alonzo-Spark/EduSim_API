    <script>
        // Define the content logic
        const physicsData = {
            formula: "V = I \\cdot R",
            originalI: 2,
            originalR: 10,
            originalV: 20
        };

        // Initialize the page once KaTeX is loaded
        document.addEventListener("DOMContentLoaded", () => {
            // Small delay to ensure KaTeX script is parsed if not deferred correctly, 
            // though defer usually handles it.
            setTimeout(renderMath, 100);
            setupLab();
        });

        function renderMath() {
            if (typeof katex === "undefined") {
                console.error("KaTeX not loaded");
                return;
            }

            // 1. Render Main Formula Card
            katex.render(physicsData.formula, document.getElementById('main-formula-display'), {
                throwOnError: false,
                displayMode: true
            });

            // 2. Render Calculation Steps
            katex.render("V = I \\cdot R", document.getElementById('calc-step-1'), { throwOnError: false });
            katex.render("V = 2 \\, \\text{A} \\cdot 10 \\, \\Omega", document.getElementById('calc-step-2'), { throwOnError: false });
            katex.render("V = 20 \\, \\text{V}", document.getElementById('calc-step-3'), { throwOnError: false });
            
            // 3. Render Lab Equation
            updateLabEquation();
        }

        // Interaction: Toggle Explanation
        function toggleExplanation() {
            const box = document.getElementById('formula-explanation');
            box.classList.toggle('active');
        }

        // Interaction: Scroll to Lab
        function scrollToLab() {
            const lab = document.getElementById('lab-section');
            lab.scrollIntoView({ behavior: 'smooth', block: 'center' });
            
            // Visual feedback on the lab card
            lab.style.borderColor = 'var(--primary-color)';
            setTimeout(() => {
                lab.style.borderColor = 'var(--border-color)';
            }, 1000);
        }

        // Formula Lab Logic
        function setupLab() {
            const inputI = document.getElementById('input-current');
            const inputR = document.getElementById('input-resistance');
            const labelI = document.getElementById('label-current');
            const labelR = document.getElementById('label-resistance');
            const outputV = document.getElementById('output-voltage');

            function updateLab() {
                const I = parseFloat(inputI.value);
                const R = parseFloat(inputR.value);
                const V = I * R;

                // Update Labels
                labelI.textContent = `${I} A`;
                labelR.textContent = `${R} Ω`;
                
                // Update Result
                outputV.textContent = V.toFixed(2);

                // Update Live Equation
                updateLabEquation(I, R);
            }

            inputI.addEventListener('input', updateLab);
            inputR.addEventListener('input', updateLab);
            
            // Initial call
            updateLab();
        }

        function updateLabEquation(I, R) {
            // If arguments not provided, use current input values
            if (I === undefined) I = document.getElementById('input-current').value;
            if (R === undefined) R = document.getElementById('input-resistance').value;

            const V = (I * R).toFixed(2);
            const texString = `V = ${I} \\times ${R} = ${V}`;
            
            katex.render(texString, document.getElementById('live-equation'), { throwOnError: false });
        }
    </script>
