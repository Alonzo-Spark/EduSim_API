                    <div class="card-title">Derivation</div>
                </div>
                <div class="card-content">
                    <p>We can derive the formula using the <strong>Work-Energy Theorem</strong>, which states that the net work done on an object equals its change in kinetic energy.</p>
                    
                    <h3>Step 1: Work Done</h3>
                    <p>Consider an object of mass $m$ accelerating from rest to velocity $v$ under constant force $F$ over displacement $s$.</p>
                    $$ W = F \cdot s $$

                    <h3>Step 2: Newton's Second Law</h3>
                    <p>Substitute $F = ma$ into the work equation:</p>
                    $$ W = (ma) \cdot s $$

                    <h3>Step 3: Kinematics</h3>
                    <p>Use the third equation of motion ($v^2 = u^2 + 2as$). Since the object starts from rest, initial velocity $u = 0$:</p>
                    $$ v^2 = 2as \implies s = \frac{v^2}{2a} $$

                    <h3>Step 4: Substitution</h3>
                    <p>Substitute $s$ back into the work equation:</p>
                    $$ W = ma \cdot \left( \frac{v^2}{2a} \right) $$
                    
                    <p>The mass $m$ cancels out partially, and acceleration $a$ cancels out completely:</p>
                    $$ W = \frac{1}{2}mv^2 $$

                    <p>Since this work done equals the change in kinetic energy (from 0 to KE), we conclude:</p>
                    $$ KE = \frac{1}{2}mv^2 $$
                </div>

        // Formula Extraction Logic (Simulated based on DOM content)
        // In a dynamic app, this would parse the markdown. Here we define the structure.

        function toggleExplanation() {
            const box = document.getElementById('formula-explanation');
            box.classList.toggle('active');
        }

        // Formula Lab Logic
        const modal = document.getElementById('lab-modal');
        const inputM = document.getElementById('input-m');
        const inputV = document.getElementById('input-v');
        const labelM = document.getElementById('label-m');
        const labelV = document.getElementById('label-v');
        const resultKE = document.getElementById('result-ke');
        const mathM = document.getElementById('math-m');
        const mathV = document.getElementById('math-v');
        const energyBar = document.getElementById('energy-bar');

        // Max KE for bar visualization (100kg * 50^2 * 0.5)
        const MAX_POSSIBLE_KE = 125000; 

        function openFormulaLab() {
            modal.classList.add('active');
            calculateLab(); // Initial calculation
        }

        function closeFormulaLab() {
            modal.classList.remove('active');
        }

        function calculateLab() {
            const m = parseFloat(inputM.value);
            const v = parseFloat(inputV.value);

            // Update Labels
            labelM.textContent = `${m} kg`;
            labelV.textContent = `${v} m/s`;

            // Update Math Text inside Modal
            mathM.textContent = m;
            mathV.textContent = v;

            // Calculate KE
            const ke = 0.5 * m * Math.pow(v, 2);

            // Update Result Display
            // Format with commas for readability
            resultKE.textContent = ke.toLocaleString(undefined, { maximumFractionDigits: 1 });

            // Update Visual Bar
            const percentage = Math.min((ke / MAX_POSSIBLE_KE) * 100, 100);
            energyBar.style.width = `${percentage}%`;
            
            // Change color based on intensity
            if(percentage > 80) energyBar.style.backgroundColor = '#ef4444'; // Red
            else if(percentage > 40) energyBar.style.backgroundColor = '#f59e0b'; // Orange
            else energyBar.style.backgroundColor = '#2563eb'; // Blue
        }

        // Event Listeners for Sliders
        inputM.addEventListener('input', calculateLab);
        inputV.addEventListener('input', calculateLab);

        // Close modal on outside click
        modal.addEventListener('click', (e) => {
            if (e.target === modal) closeFormulaLab();
        });

    </script>
