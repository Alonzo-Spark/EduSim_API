    <script>
        // Get canvas context
        const canvas = document.getElementById('springCanvas');
        const ctx = canvas.getContext('2d');
        const motionChartCtx = document.getElementById('motionChart').getContext('2d');
        
        // Simulation parameters
        let params = {
            stiffness: 50,       // Spring constant (N/m)
            mass: 2.0,           // Mass (kg)
            damping: 0.2,        // Damping factor
            displacement: 1.5,   // Initial displacement (m)
            gravity: true,       // Gravity enabled
            isRunning: false,
            time: 0
        };
        
        // System state
        let system = {
            anchorX: 100,
            anchorY: 200,
            massX: 400,
            massY: 200,
            velocity: 0,
            equilibriumX: 400,
            acceleration: 0
        };
        
        // Position history for chart
        let positionHistory = [];
        const maxHistory = 100;
        
        // Initialize motion chart
        const motionChart = new Chart(motionChartCtx, {
            type: 'line',
            data: {
                labels: [],
                datasets: [{
                    label: 'Mass Position (m)',
                    data: [],
                    borderColor: '#00c9ff',
                    backgroundColor: 'rgba(0, 201, 255, 0.1)',
                    borderWidth: 2,
                    fill: true,
                    tension: 0.4
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                scales: {
                    x: {
                        title: {
                            display: true,
                            text: 'Time (s)',
                            color: '#aaa'
                        },
                        grid: {
                            color: 'rgba(255, 255, 255, 0.1)'
                        },
                        ticks: {
                            color: '#aaa'
                        }
                    },
                    y: {
                        title: {
                            display: true,
                            text: 'Displacement (m)',
                            color: '#aaa'
                        },
                        grid: {
                            color: 'rgba(255, 255, 255, 0.1)'
                        },
                        ticks: {
                            color: '#aaa'
                        },
                        suggestedMin: -3,
                        suggestedMax: 3
                    }
                },
                plugins: {
                    legend: {
                        labels: {
                            color: '#ddd'
                        }
                    }
                }
            }
        });
        
        // Update slider value displays
        function updateSliderValues() {
            document.getElementById('stiffness-value').textContent = params.stiffness + ' N/m';
            document.getElementById('mass-value').textContent = params.mass.toFixed(1) + ' kg';
            document.getElementById('damping-value').textContent = params.damping.toFixed(2);
            document.getElementById('displacement-value').textContent = params.displacement.toFixed(1) + ' m';
        }
        
        // Initialize system
        function initSystem() {
            system.equilibriumX = canvas.width - 150;
            system.massX = system.equilibriumX + params.displacement * 50;
            system.velocity = 0;
            system.acceleration = 0;
            params.time = 0;
            positionHistory = [];
            updateChart();
            drawSystem();
        }
        
        // Draw the spring-mass system
        function drawSystem() {
            // Clear canvas
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // Draw anchor
            ctx.fillStyle = '#888';
            ctx.fillRect(system.anchorX - 15, system.anchorY - 40, 30, 40);
            ctx.fillStyle = '#555';
            ctx.beginPath();
            ctx.arc(system.anchorX, system.anchorY, 15, 0, Math.PI * 2);
            ctx.fill();
            
            // Draw spring
            drawSpring(system.anchorX, system.anchorY, system.massX, system.massY);
            
            // Draw mass
            const massWidth = 60 + params.mass * 5;
            const massHeight = 60 + params.mass * 5;
            
            ctx.fillStyle = '#00c9ff';
            ctx.fillRect(system.massX - massWidth/2, system.massY - massHeight/2, massWidth, massHeight);
            
            ctx.fillStyle = '#fff';
            ctx.font = 'bold 16px Arial';
            ctx.textAlign = 'center';
            ctx.fillText(params.mass.toFixed(1) + ' kg', system.massX, system.massY + 5);
            
            // Draw equilibrium position
            ctx.strokeStyle = 'rgba(146, 254, 157, 0.5)';
            ctx.setLineDash([5, 5]);
            ctx.beginPath();
            ctx.moveTo(system.equilibriumX, system.massY - 100);
            ctx.lineTo(system.equilibriumX, system.massY + 100);
            ctx.stroke();
            ctx.setLineDash([]);
            
            // Draw info
            ctx.fillStyle = 'rgba(255, 255, 255, 0.7)';
            ctx.font = '14px Arial';
            ctx.textAlign = 'left';
            ctx.fillText(`Displacement: ${((system.massX - system.equilibriumX)/50).toFixed(2)} m`, 20, 30);
            ctx.fillText(`Velocity: ${system.velocity.toFixed(2)} m/s`, 20, 50);
            ctx.fillText(`Acceleration: ${system.acceleration.toFixed(2)} m/s²`, 20, 70);
            
            // Draw legend
            ctx.fillStyle = 'rgba(146, 254, 157, 0.7)';
            ctx.fillText('Equilibrium Position', system.equilibriumX + 10, 30);
        }
        
        // Draw spring as a coil
        function drawSpring(x1, y1, x2, y2) {
            const dx = x2 - x1;
            const dy = y2 - y1;
            const distance = Math.sqrt(dx * dx + dy * dy);
            const coils = 10;
            const coilRadius = 8;
            
            ctx.beginPath();
            ctx.moveTo(x1, y1);
            
            // Draw coils
            for (let i = 0; i < coils; i++) {
                const percent = (i + 0.5) / coils;
                const coilX = x1 + dx * percent;
                const coilY = y1 + dy * percent;
                
                // Alternate coil direction
                const angle = Math.atan2(dy, dx) + Math.PI/2;
                const offsetX = Math.cos(angle) * coilRadius * (i % 2 === 0 ? 1 : -1);
                const offsetY = Math.sin(angle) * coilRadius * (i % 2 === 0 ? 1 : -1);
                
                ctx.lineTo(coilX + offsetX, coilY + offsetY);
            }
            
            ctx.lineTo(x2, y2);
            ctx.strokeStyle = '#ff9d00';
            ctx.lineWidth = 3;
            ctx.stroke();
        }
        
        // Update the system physics
        function updatePhysics(deltaTime) {
            if (!params.isRunning) return;
            
            // Calculate spring force (Hooke's Law)
            const displacement = system.massX - system.equilibriumX;
            const springForce = -params.stiffness * displacement / 50;
            
            // Calculate damping force
            const dampingForce = -params.damping * system.velocity * 10;
            
            // Calculate gravity force
            const gravityForce = params.gravity ? 0.5 * params.mass : 0;
            
            // Total force
            const totalForce = springForce + dampingForce + gravityForce;
            
            // Calculate acceleration (F = ma)
            system.acceleration = totalForce / params.mass;
            
            // Update velocity and position
            system.velocity += system.acceleration * deltaTime;
            system.massX += system.velocity * deltaTime * 50;
            
            // Update time
            params.time += deltaTime;
            
            // Record position for chart
            positionHistory.push({
                time: params.time,
                position: (system.massX - system.equilibriumX) / 50
            });
            
            // Keep only recent history
            if (positionHistory.length > maxHistory) {
                positionHistory.shift();
            }
            
            // Update chart
            updateChart();
        }
        
        // Update the motion chart
        function updateChart() {
            const labels = [];
            const data = [];
            
            for (const point of positionHistory) {
                labels.push(point.time.toFixed(1));
                data.push(point.position);
            }
            
            motionChart.data.labels = labels;
            motionChart.data.datasets[0].data = data;
            motionChart.update();
        }
        
        // Animation loop
        function animate() {
            updatePhysics(0.05);
            drawSystem();
            requestAnimationFrame(animate);
        }
        
        // Event listeners for sliders
        document.getElementById('stiffness').addEventListener('input', function() {
            params.stiffness = parseInt(this.value);
            updateSliderValues();
        });
        
        document.getElementById('mass').addEventListener('input', function() {
            params.mass = parseFloat(this.value);
            updateSliderValues();
        });
        
        document.getElementById('damping').addEventListener('input', function() {
            params.damping = parseFloat(this.value);
            updateSliderValues();
        });
        
        document.getElementById('displacement').addEventListener('input', function() {
            params.displacement = parseFloat(this.value);
            if (!params.isRunning) {
                initSystem();
            }
        });
        
        // Event listeners for buttons
        document.getElementById('start-btn').addEventListener('click', function() {
            params.isRunning = true;
        });
        
        document.getElementById('stop-btn').addEventListener('click', function() {
            params.isRunning = false;
        });
        
        document.getElementById('reset-btn').addEventListener('click', function() {
            initSystem();
        });
        
        document.getElementById('gravity-btn').addEventListener('click', function() {
            params.gravity = !params.gravity;
            this.innerHTML = params.gravity ? 
                '<i class="fas fa-globe"></i> Gravity: ON' : 
                '<i class="fas fa-globe"></i> Gravity: OFF';
        });
        
        // Initialize
        updateSliderValues();
        initSystem();
        animate();
        
        // Make mass draggable
        let isDragging = false;
        
        canvas.addEventListener('mousedown', function(e) {
            const rect = canvas.getBoundingClientRect();
            const mouseX = e.clientX - rect.left;
            const mouseY = e.clientY - rect.top;
            
            const massWidth = 60 + params.mass * 5;
            const massHeight = 60 + params.mass * 5;
            
            if (mouseX > system.massX - massWidth/2 && 
                mouseX < system.massX + massWidth/2 &&
                mouseY > system.massY - massHeight/2 && 
                mouseY < system.massY + massHeight/2) {
                isDragging = true;
                params.isRunning = false;
            }
        });
        
        canvas.addEventListener('mousemove', function(e) {
            if (isDragging) {
                const rect = canvas.getBoundingClientRect();
                system.massX = e.clientX - rect.left;
                drawSystem();
            }
        });
        
        canvas.addEventListener('mouseup', function() {
            isDragging = false;
        });
        
        canvas.addEventListener('mouseleave', function() {
            isDragging = false;
        });
    </script>
</body>

## Features of This Spring-Mass Simulation

This interactive simulation includes:

1. **Physics Visualization**:
   - Spring connecting a fixed anchor to a movable mass
   - Realistic spring animation with coil rendering
   - Equilibrium position indicator

2. **Interactive Controls**:
   - Adjust spring stiffness (k) from 10-200 N/m
   - Set mass from 0.5-10 kg
   - Control damping factor to see overdamped/critically damped/underdamped systems
   - Set initial displacement
   - Toggle gravity on/off

3. **Real-time Physics**:
   - Implements Hooke's Law (F = -k·x)
   - Simulates harmonic motion with damping
   - Calculates acceleration based on net force

4. **Data Visualization**:
   - Real-time chart showing displacement over time
   - Visual indicators for velocity and acceleration

5. **Interactive Elements**:
   - Drag the mass to any position
   - Start/stop oscillation
   - Reset system to initial conditions

The simulation runs entirely in your browser using HTML5 Canvas and JavaScript. You can interact with the mass, adjust parameters, and observe how the system behaves according to the principles of physics.
