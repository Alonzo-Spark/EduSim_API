def detect_sections(markdown_text):
    """Detects sections based on headings."""
    sections = []
    for line in markdown_text.splitlines():
        match = re.match(r"^(Introduction|Definition|Key Concepts|Characteristics|Formula|Applications|Advantages|Disadvantages|Important Notes|Summary|Suggested Questions)", line, re.IGNORECASE)
        if match:

import re

def extract_formulas(markdown_text):
    """Extracts formulas and their context."""
    formulas = []
    for line in markdown_text.splitlines():
        match = re.search(r"$$([^$]+)$$", line)  # Matches LaTeX formulas
        if match:
            formula = match.group(1)
            # Get surrounding context (e.g., previous sentences)
            surrounding_context = ""
            # Iterate backwards to get context
            for i in range(max(0, line.find(match.group(0)) - 50), 0, -1):
                surrounding_context += line[:i].strip() + "\n"
            formulas.append({
                "formula": formula,
                "title": "Main Formula", #Placeholder, could be improved
                "surroundingContext": surrounding_context,
                "variables": [] # To be populated from the formula

import pandas as pd

def render_table(markdown_table_string):
    """Renders a Markdown table as a card."""
    try:
        df = pd.read_csv(StringIO(markdown_table_string), sep="|", engine='python', skipinitialspace=True)
        table_html = df.to_html(index=False, classes='responsive-table')
        return {
            "type": "table",
            "html": table_html
        }
    except Exception as e:
        return {
            "type": "error",
            "message": f"Error rendering table: {e}"
        }
```

def generate_ui_blocks(markdown_text, formulas, sections):
    blocks = []
    current_section = None

    for line in markdown_text.splitlines():
        line = line.strip()

        if line in ["Introduction", "Definition", "Key Concepts", "Characteristics", "Formula", "Applications", "Advantages", "Disadvantages", "Important Notes", "Summary", "Suggested Questions"]:
            current_section = line
            blocks.append({
                "type": "section_header",
                "title": line
            })
        elif current_section:
            if line.startswith("###"):
                blocks.append({
                    "type": "section_subheading",
                    "title": line[3:].strip()
                })
            elif line.startswith("##"):
                blocks.append({
                    "type": "section_subheading",
                    "title": line[2:].strip()
                })
            elif re.match(r"^\d+\.\s", line): #Numbered list
                blocks.append({
                    "type": "list_item",
                    "text": line[3:].strip()
                })
            elif line.startswith("Table"):
                table_content = ""
                # Assuming the table starts with "Table:" and the rest is the table
                table_start = markdown_text.find(line) + len(line)
                table_end = markdown_text.find("\n", table_start)
                if table_end == -1:
                    table_end = len(markdown_text)

                table_content = markdown_text[table_start:table_end]

                table_block = render_table(table_content)
                blocks.append(table_block)
            elif re.match(r"^\s*\n", line) and not blocks: #Empty lines
                pass

            else:
                blocks.append({
                    "type": "paragraph",
                    "text": line
                })
    return blocks
```
