    const labDialog = document.getElementById('formula-lab');

    // Simulation State
    const sim = {
        m: 2.0,
        L: 1.5,
        g: 9.8,
        angle: Math.PI / 4, // Initial angle (45 degrees)
        velocity: 0,
        dragging: false,
        originX: 0,
        originY: 0,
        scale: 100 // pixels per meter
    };

    // Canvas Setup
    const canvas = document.getElementById('simCanvas');
    const ctx = canvas.getContext('2d');
    let animationFrame;

    function resizeCanvas() {
        const rect = canvas.parentElement.getBoundingClientRect();
        canvas.width = rect.width;
        canvas.height = rect.height;
        sim.originX = canvas.width / 2;
        sim.originY = 50;
    }

    window.openFormulaLab = function() {
        labDialog.showModal();
        resizeCanvas();
        if (!animationFrame) animate();
    };

    closeLabBtn.addEventListener('click', () => {
        labDialog.close();
    });

    // Lab Controls
    const inpMass = document.getElementById('inp-mass');
    const inpLen = document.getElementById('inp-length');
    const inpGrav = document.getElementById('inp-gravity');

    function updateParams() {
        sim.m = parseFloat(inpMass.value);
        sim.L = parseFloat(inpLen.value);
        sim.g = parseFloat(inpGrav.value);

        document.getElementById('lbl-mass').textContent = sim.m.toFixed(1) + ' kg';
        document.getElementById('lbl-length').textContent = sim.L.toFixed(1) + ' m';
        document.getElementById('lbl-gravity').textContent = sim.g.toFixed(1) + ' m/s²';
    }

    [inpMass, inpLen, inpGrav].forEach(el => el.addEventListener('input', updateParams));

    // Physics Engine
    function updatePhysics(dt) {
        if (sim.dragging) {
            sim.velocity = 0;
            return;
        }

        // Simple Pendulum Equation: angular acceleration = -(g/L) * sin(theta)
        const acceleration = -(sim.g / sim.L) * Math.sin(sim.angle);
        
        // Semi-implicit Euler integration
        sim.velocity += acceleration * dt;
        sim.angle += sim.velocity * dt;

        // Damping (air resistance approximation for stability)
        sim.velocity *= 0.999; 
    }

    // Drawing Engine
    function draw() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);

        // Calculate bob position
        const bobX = sim.originX + Math.sin(sim.angle) * sim.L * sim.scale;
        const bobY = sim.originY + Math.cos(sim.angle) * sim.L * sim.scale;

        // Draw Pivot
        ctx.beginPath();
        ctx.arc(sim.originX, sim.originY, 5, 0, Math.PI * 2);
        ctx.fillStyle = '#64748b';
        ctx.fill();

        // Draw String
        ctx.beginPath();
        ctx.moveTo(sim.originX, sim.originY);
        ctx.lineTo(bobX, bobY);
        ctx.strokeStyle = '#334155';
        ctx.lineWidth = 2;
        ctx.stroke();

        // Draw Bob
        ctx.beginPath();
        ctx.arc(bobX, bobY, 15 + sim.m * 2, 0, Math.PI * 2); // Size varies slightly with mass
        ctx.fillStyle = '#2563eb';
        ctx.fill();
        ctx.strokeStyle = '#1e40af';
        ctx.lineWidth = 2;
        ctx.stroke();

        // Update UI Stats
        updateStats();
    }

    function updateStats() {
        // Height relative to lowest point (h = L - Lcos(theta))
        const h = sim.L * (1 - Math.cos(sim.angle));
        
        // PE = mgh
        const pe = sim.m * sim.g * h;
        
        // KE = 0.5 * m * (v_linear)^2
        // v_linear = L * angular_velocity
        const v_linear = sim.L * sim.velocity;
        const ke = 0.5 * sim.m * v_linear * v_linear;
        
        const total = pe + ke;

        // Update DOM
        document.getElementById('val-pe').textContent = pe.toFixed(1) + ' J';
        document.getElementById('val-ke').textContent = ke.toFixed(1) + ' J';
        document.getElementById('val-total').textContent = total.toFixed(1) + ' J';

        // Update Bars
        const maxEnergy = total > 0.1 ? total : 1; // Prevent division by zero
        document.getElementById('bar-pe').style.width = (pe / maxEnergy * 100) + '%';
        document.getElementById('bar-ke').style.width = (ke / maxEnergy * 100) + '%';
    }

    let lastTime = 0;
    function animate(timestamp) {
        if (!lastTime) lastTime = timestamp;
        const dt = Math.min((timestamp - lastTime) / 1000, 0.05); // Cap dt for stability
        lastTime = timestamp;

        updatePhysics(dt);
        draw();

        if (labDialog.open) {
            animationFrame = requestAnimationFrame(animate);
        } else {
            animationFrame = null;
        }
    }

    // Mouse Interaction for Lab
    canvas.addEventListener('mousedown', (e) => {
        const rect = canvas.getBoundingClientRect();
        const mouseX = e.clientX - rect.left;
        const mouseY = e.clientY - rect.top;

        const bobX = sim.originX + Math.sin(sim.angle) * sim.L * sim.scale;
        const bobY = sim.originY + Math.cos(sim.angle) * sim.L * sim.scale;

        const dist = Math.hypot(mouseX - bobX, mouseY - bobY);
        if (dist < 30) {
            sim.dragging = true;
            sim.velocity = 0;
        }
    });

    window.addEventListener('mousemove', (e) => {
        if (sim.dragging && labDialog.open) {
            const rect = canvas.getBoundingClientRect();
            const mouseX = e.clientX - rect.left;
            const mouseY = e.clientY - rect.top;

            const dx = mouseX - sim.originX;
            const dy = mouseY - sim.originY;
            sim.angle = Math.atan2(dx, dy);
        }
    });

    window.addEventListener('mouseup', () => {
        sim.dragging = false;
    });

    // Initialize
    renderContent();

</script>
