**I. Core Architecture & Data Structures**

The primary change is moving from a linear Markdown-to-HTML conversion to a structured data model. We'll represent the content as a tree of "blocks" (headings, paragraphs, formulas, tables, sections, cards).

```python
class Block:
    def __init__(self, block_type, content, metadata=None):
        self.block_type = block_type  # e.g., "heading", "paragraph", "formula", "table", "section", "card"
        self.content = content
        self.metadata = metadata or {}  # Store data like formula title, surrounding context, etc.
        self.children = []

    def __repr__(self):
        return f"Block(type={self.block_type}, content='{self.content[:20]}...')"

class FormulaData:
    def __init__(self, formula_string, title, surrounding_context, variables):
        self.formula_string = formula_string
        self.title = title
        self.surrounding_context = surrounding_context
        self.variables = variables

    def __repr__(self):
        return f"FormulaData(title='{self.title[:20]}...')"


class SectionCard:
    def __init__(self, title, content):
        self.title = title
        self.content = content

    def __repr__(self):
        return f"SectionCard(title='{self.title[:20]}...')"


class TableCard:
  def __init__(self, data):
    self.data = data

  def __repr__(self):
    return f"TableCard(data={self.data})"
```

```python
import re

def parse_markdown(markdown_content):
    blocks = []
    lines = markdown_content.splitlines()
    i = 0
    while i < len(lines):
      line = lines[i].strip()
      if line.startswith("# "):
        block_type = "heading"
        content = line[2:]
      elif line.startswith("## "):
        block_type = "heading"
        content = line[3:]
      elif line.startswith("$$"):
        # Extract Formula
        match = re.search(r"$$([^$]+)$$", line)
        if match:
          formula = match.group(1)
          title = "Formula" # Default title
          surrounding_context = ""
          variables = []
          block = Block("formula", formula, metadata={"title": title, "surrounding_context": surrounding_context, "variables": variables})
          blocks.append(block)
          i += 1 #skip the next line
          continue

      elif line.startswith("Table:"):
        # Extract Table Data (basic example – needs more robust parsing)
        table_data = parse_table_from_markdown(lines[i+1:])
        table_card = TableCard(table_data)
        blocks.append(table_card)
        i += len(table_data) + 1

      else:
        block_type = "paragraph"
        content = line
        block = Block(block_type, content)
      blocks.append(block)
      i += 1

    return blocks
```
