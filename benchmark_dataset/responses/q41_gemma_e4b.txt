```python
import re

def extract_formulas(markdown_text):
    formulas = []
    for line in markdown_text.splitlines():
        match = re.search(r"$$([\s\S]*?)$$", line)
        if match:
            formula = match.group(1).strip()
            # Extract surrounding context (lines before and after)
            context = get_surrounding_context(markdown_text, line)
            variables = extract_variables(formula) #Extract variables from formula
            formulas.append({
                "formula": formula,
                "title": "Unnamed Formula", #Placeholder, can be improved
                "surroundingContext": context,
                "variables": variables
            })
    return formulas

def get_surrounding_context(text, line_index):
    context = []
    for i in range(max(0, line_index - 2), min(len(text.splitlines()), line_index + 3)):
        context.append(text.splitlines()[i])
    return "\n".join(context)

def extract_variables(formula):
    #Simple variable extraction - improve with NLP
    variables = []
    for char in formula:
        if char.isalpha():
            variables.append(char)
    return variables
```
