        let animationId;

        const ctx = canvas.getContext('2d');

        // Physics State

        function updatePhysicsParams() {
            physics.mass = parseFloat(massRange.value);
            physics.k = parseFloat(kRange.value);
            physics.damping = parseFloat(dampingRange.value);
            massVal.textContent = physics.mass.toFixed(1);
            kVal.textContent = physics.k.toFixed(1);
        }

        massRange.addEventListener('input', updatePhysicsParams);
        kRange.addEventListener('input', updatePhysicsParams);
        dampingRange.addEventListener('input', updatePhysicsParams);

        function drawSpring(x, y, stretch) {
            ctx.beginPath();
            ctx.strokeStyle = '#64748b';
            ctx.lineWidth = 2;
            ctx.lineJoin = 'round';
            ctx.lineCap = 'round';

            const startX = 0; // Wall at left
            const endX = x - 20; // Connect to block left side
            const segments = 12;
            const width = 20;
            
            ctx.moveTo(startX, y);
            
            for (let i = 0; i <= segments; i++) {
                const px = startX + (endX - startX) * (i / segments);
                // Zigzag offset
                const py = y + (i % 2 === 0 ? 0 : (i % 4 === 1 ? width : -width));
                // Flatten ends
                const offset = (i === 0 || i === segments) ? 0 : (i % 2 === 0 ? 0 : (i % 4 === 1 ? width : -width));
                
                // Actually let's do a simple coil visual
                const coilY = y + (i === 0 || i === segments ? 0 : (i % 2 === 1 ? 15 : -15));
                
                ctx.lineTo(px, coilY);
            }
            ctx.stroke();
        }

        function drawBlock(x, y) {
            ctx.fillStyle = '#3b82f6';
            ctx.shadowColor = 'rgba(0,0,0,0.2)';
            ctx.shadowBlur = 10;
            ctx.shadowOffsetY = 4;
            ctx.fillRect(x, y - 20, 40, 40); // 40x40 block
            
            // Reset shadow
            ctx.shadowColor = 'transparent';
            
            // Mass label
            ctx.fillStyle = 'white';
            ctx.font = '12px Inter';
            ctx.textAlign = 'center';
            ctx.fillText('m', x + 20, y + 5);
        }

        function drawEnvironment() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // Draw Floor
            ctx.beginPath();
            ctx.strokeStyle = '#e2e8f0';
            ctx.lineWidth = 2;
            ctx.moveTo(0, physics.anchorY + 21);
            ctx.lineTo(canvas.width, physics.anchorY + 21);
            ctx.stroke();

            // Draw Wall
            ctx.fillStyle = '#cbd5e1';
            ctx.fillRect(0, physics.anchorY - 50, 10, 100);
            
            // Draw Equilibrium Line
            ctx.beginPath();
            ctx.strokeStyle = '#ef4444'; // Red line for equilibrium
            ctx.setLineDash([5, 5]);
            ctx.moveTo(physics.anchorX, physics.anchorY - 60);
            ctx.lineTo(physics.anchorX, physics.anchorY + 60);
            ctx.stroke();
            ctx.setLineDash([]);
            
            ctx.fillStyle = '#ef4444';
            ctx.font = '10px Inter';
            ctx.fillText('x = 0', physics.anchorX, physics.anchorY - 65);
        }

        function loop() {
            if (!isDragging) {
                // Physics logic: F = -kx - cv
                const force = -physics.k * physics.positionX;
                const dampingForce = -physics.damping * physics.velocity;
                const acceleration = (force + dampingForce) / physics.mass;

                physics.velocity += acceleration;
                physics.positionX += physics.velocity;
            }

            drawEnvironment();
            
            const currentX = physics.anchorX + physics.positionX;
            drawSpring(currentX, physics.anchorY, physics.positionX);
            drawBlock(currentX, physics.anchorY);

            // Update text
            liveData.innerHTML = `Displacement ($x$): ${(physics.positionX/100).toFixed(2)} m | Velocity ($v$): ${(physics.velocity).toFixed(2)} m/s`;

            animationId = requestAnimationFrame(loop);
        }

        // Mouse Interaction for Lab
        function getMousePos(evt) {
            const rect = canvas.getBoundingClientRect();
            return {
                x: evt.clientX - rect.left,
                y: evt.clientY - rect.top
            };
        }

        canvas.addEventListener('mousedown', (e) => {
            const pos = getMousePos(e);
            const blockX = physics.anchorX + physics.positionX;
            // Check if clicking inside block
            if (pos.x >= blockX && pos.x <= blockX + 40 && pos.y >= physics.anchorY - 20 && pos.y <= physics.anchorY + 20) {
                isDragging = true;
                physics.velocity = 0; // Stop movement when grabbing
            }
        });

        window.addEventListener('mousemove', (e) => {
            if (isDragging) {
                const pos = getMousePos(e);
                // Update position based on mouse
                physics.positionX = pos.x - physics.anchorX - 20; // Center offset
            }
        });

        window.addEventListener('mouseup', () => {
            isDragging = false;
        });

        function resetSimulation() {
            physics.positionX = 100;
            physics.velocity = 0;
            physics.damping = parseFloat(dampingRange.value);
        }

        // Modal Logic
        const modalOverlay = document.getElementById('labModal');

        function openLab() {
            modalOverlay.classList.add('active');
            // Start loop if not running
            if (!animationId) loop();
        }

        function closeLab() {
            modalOverlay.classList.remove('active');
        }

        // Close modal on outside click
        modalOverlay.addEventListener('click', (e) => {
            if (e.target === modalOverlay) closeLab();
        });
