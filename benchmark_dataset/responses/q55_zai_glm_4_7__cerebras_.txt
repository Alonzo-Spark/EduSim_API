    const courseContent = [
        {
            type: 'card',
            title: 'Quick Definition',
            content: `

        },
        {
            type: 'card',
            title: 'Key Concepts',
            content: `
                <ul>
                    <li><strong>Coefficient of Friction (μ):</strong> A dimensionless scalar value. Ice on ice has μ ≈ 0.05, while Rubber on concrete has μ ≈ 1.0.</li>
                    <li><strong>Normal Force (N):</strong> The support force exerted upon an object that is in contact with another stable object. Usually equal to the object's weight on a flat surface.</li>
                    <li><strong>Zero Friction Scenario:</strong> If μ scales to zero, $f_k$ becomes 0. According to Newton's First Law, the object will slide forever at constant velocity unless acted upon by another force.</li>
                </ul>
            `
        },
        {
            type: 'table',
            title: 'Material Properties',
            headers: ['Material Pair', 'Coeff. of Friction (μ)', 'Behavior'],

            ]
        },
        {
            type: 'card',
            title: 'Summary',

                <p>As the coefficient of friction ($\\mu_k$) approaches zero, the friction force ($f_k$) vanishes. In the simulation, you will observe the block maintaining its velocity for much longer distances, eventually stopping only due to numerical precision or air resistance (not modeled here).</p>

    const root = document.getElementById('content-root');

    function renderContent() {
        courseContent.forEach(section => {
            if (section.type === 'card') {
                root.appendChild(createCard(section));
            } else if (section.type === 'formula') {
                root.appendChild(createFormulaCard(section));
            } else if (section.type === 'simulation') {
                root.appendChild(createSimulationCard(section));
            } else if (section.type === 'table') {
                root.appendChild(createTableCard(section));
            }
        });
        
        // Render Math
        renderMathInElement(root, {
            delimiters: [
                {left: "$$", right: "$$", display: true},
                {left: "$", right: "$", display: false}
            ],
            throwOnError : false
        });
    }

    function createCard(data) {
        const el = document.createElement('section');
        el.className = 'card';

                <span class="badge">Textbook</span>
            </div>
            <div class="card-body">${data.content}</div>
        `;
        return el;
    }

    // Formula Lab Logic
    let labFormulaLatex = '';
    
    function openFormulaLab(latex) {
        labFormulaLatex = latex;
        const modal = document.getElementById('formula-lab-modal');
        modal.classList.add('active');
        
        // Render formula in modal
        const display = document.getElementById('lab-formula-display');

    // Lab Inputs

    function updateLabCalc() {
        const mu = parseFloat(document.getElementById('lab-mu').value);
        const N = parseFloat(document.getElementById('lab-n').value);
        const fk = mu * N;
        
        document.getElementById('lab-result').textContent = fk.toFixed(2) + ' N';
    }

    function showToast(msg) {
        const t = document.getElementById('toast');
        t.textContent = msg;
        t.classList.add('show');
        setTimeout(() => t.classList.remove('show'), 3000);
    }

    // ==========================================
    // PHYSICS ENGINE
    // ==========================================
    
    let simState = {
        running: false,
        x: 50, // pixels
        v: 0,  // m/s (virtual)
        mu: 0.3,
        v0: 15,
        mass: 5, // kg
        scale: 10, // pixels per meter
        dt: 1/60, // seconds per frame
        animId: null
    };

    function initSimulation() {
        const block = document.getElementById('sim-block');
        const sliderMu = document.getElementById('mu-slider');
        const sliderV0 = document.getElementById('v0-slider');
        const btnStart = document.getElementById('btn-start');
        const btnReset = document.getElementById('btn-reset');
        const velVector = document.getElementById('vel-vector');

        // Update UI from sliders
        sliderMu.addEventListener('input', (e) => {
            simState.mu = parseFloat(e.target.value);
            document.getElementById('mu-display').textContent = simState.mu.toFixed(2);
            if (!simState.running) updateStatus(`Friction set to $\\mu = ${simState.mu}$`);
        });

        sliderV0.addEventListener('input', (e) => {
            simState.v0 = parseFloat(e.target.value);
            document.getElementById('v0-display').textContent = simState.v0 + ' m/s';
            if (!simState.running) updateStatus(`Initial velocity set to $v_0 = ${simState.v0}$ m/s`);
        });

        btnStart.addEventListener('click', () => {
            if (simState.running) return;
            startSim();
        });

        btnReset.addEventListener('click', () => {
            stopSim();
            resetSim();
        });

        resetSim();

        function resetSim() {
            simState.x = 50;
            simState.v = 0;
            simState.running = false;
            block.style.transform = `translateX(0px)`;
            velVector.style.width = '0px';
            updateStatus("Ready. Adjust parameters and push block.");
            renderMathInElement(document.getElementById('sim-status'));
        }

        function startSim() {
            simState.running = true;
            simState.v = simState.v0;
            simState.x = 50; // reset pos
            updateStatus("Simulating... Block is decelerating.");
            renderMathInElement(document.getElementById('sim-status'));
            showToast("Simulation Started");
            loop();
        }

        function stopSim() {
            simState.running = false;
            cancelAnimationFrame(simState.animId);
        }

        function loop() {
            if (!simState.running) return;

            // Physics: a = -mu * g
            // g = 9.8
            const g = 9.8;
            const a = -simState.mu * g;

            // Euler integration
            simState.v += a * simState.dt;
            simState.x += simState.v * simState.scale * simState.dt;

            // Stop condition
            if (simState.v <= 0) {
                simState.v = 0;
                simState.running = false;
                updateStatus("Stopped. Velocity reached zero.");
                renderMathInElement(document.getElementById('sim-status'));
                showToast("Block Stopped");
            }

            // Render
            block.style.transform = `translateX(${simState.x - 50}px)`;
            
            // Visualizing Velocity Vector
            // Scale vector length for visibility
            const vecLen = Math.max(0, simState.v * 5); 
            velVector.style.width = `${vecLen}px`;
            velVector.style.transform = `translateX(${simState.x}px)`;
            
            // Loop bounds (wrap around for infinite feel if mu is very low)
            if (simState.x > 800) {
                simState.x = 50;
            }

            simState.animId = requestAnimationFrame(loop);
        }
    }

    function updateStatus(text) {
        document.getElementById('sim-status').innerHTML = `<span>${text}</span>`;
    }

    // Initialize App
    document.addEventListener('DOMContentLoaded', renderContent);

</script>
