# Simple Pendulum Simulation: Mass vs. Period

I'll create an interactive simulation to demonstrate how mass affects the period of a simple pendulum. The simulation will show that mass doesn't affect the pendulum's period - a key physics concept.

## The Physics Behind Pendulums

The period of a simple pendulum is given by:

$$T = 2\pi\sqrt{\frac{L}{g}}$$

Where:
- $T$ = period (time for one complete swing)
- $L$ = length of the pendulum
- $g$ = acceleration due to gravity (9.8 m/s² on Earth)

Notice that mass ($m$) is not in the equation! Let's verify this through simulation.

        // DOM elements
        const canvas = document.getElementById('pendulumCanvas');
        const ctx = canvas.getContext('2d');
        const massSlider = document.getElementById('massSlider');
        const lengthSlider = document.getElementById('lengthSlider');
        const gravitySlider = document.getElementById('gravitySlider');
        const angleSlider = document.getElementById('angleSlider');
        const massValue = document.getElementById('massValue');
        const lengthValue = document.getElementById('lengthValue');
        const gravityValue = document.getElementById('gravityValue');
        const angleSliderValue = document.getElementById('angleSliderValue');
        const periodValue = document.getElementById('periodValue');
        const angleValue = document.getElementById('angleValue');
        const velocityValue = document.getElementById('velocityValue');
        const startBtn = document.getElementById('startBtn');
        const resetBtn = document.getElementById('resetBtn');
        const pauseBtn = document.getElementById('pauseBtn');
        const runExperimentBtn = document.getElementById('runExperiment');
        const dataTableBody = document.querySelector('#dataTable tbody');

        // Pendulum parameters
        let pendulum = {
            angle: 30 * Math.PI / 180, // Initial angle in radians
            angularVelocity: 0,
            length: 1.0, // meters
            mass: 5, // kg
            gravity: 9.8, // m/s²
            pivotX: 250,
            pivotY: 100,
            bobRadius: 20,
            isRunning: false,
            lastTime: 0,
            periodMeasurements: [],
            currentPeriod: 0,
            lastZeroCrossing: 0
        };

        // Update displays
        function updateDisplays() {
            massValue.textContent = pendulum.mass;
            lengthValue.textContent = pendulum.length.toFixed(1);
            gravityValue.textContent = pendulum.gravity.toFixed(1);
            angleSliderValue.textContent = (pendulum.angle * 180 / Math.PI).toFixed(0);
            
            // Update bob size based on mass (visual only)
            pendulum.bobRadius = 15 + pendulum.mass / 2;
        }

        // Calculate theoretical period
        function calculateTheoreticalPeriod() {
            return 2 * Math.PI * Math.sqrt(pendulum.length / pendulum.gravity);
        }

        // Draw the pendulum
        function drawPendulum() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // Draw pivot
            ctx.beginPath();
            ctx.arc(pendulum.pivotX, pendulum.pivotY, 5, 0, Math.PI * 2);
            ctx.fillStyle = '#333';
            ctx.fill();
            
            // Calculate bob position
            const bobX = pendulum.pivotX + pendulum.length * 100 * Math.sin(pendulum.angle);
            const bobY = pendulum.pivotY + pendulum.length * 100 * Math.cos(pendulum.angle);
            
            // Draw string
            ctx.beginPath();
            ctx.moveTo(pendulum.pivotX, pendulum.pivotY);
            ctx.lineTo(bobX, bobY);
            ctx.strokeStyle = '#666';
            ctx.lineWidth = 2;
            ctx.stroke();
            
            // Draw bob with color based on mass
            const hue = 200 - (pendulum.mass * 5);
            ctx.beginPath();
            ctx.arc(bobX, bobY, pendulum.bobRadius, 0, Math.PI * 2);
            ctx.fillStyle = `hsl(${hue}, 70%, 50%)`;
            ctx.fill();
            ctx.strokeStyle = '#333';
            ctx.lineWidth = 2;
            ctx.stroke();
            
            // Draw mass text
            ctx.fillStyle = 'white';
            ctx.font = 'bold 14px Arial';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText(`${pendulum.mass}kg`, bobX, bobY);
            
            // Draw scale
            ctx.beginPath();
            ctx.moveTo(pendulum.pivotX - 150, pendulum.pivotY);
            ctx.lineTo(pendulum.pivotX + 150, pendulum.pivotY);
            ctx.strokeStyle = '#999';
            ctx.setLineDash([5, 3]);
            ctx.stroke();
            ctx.setLineDash([]);
            
            // Update data displays
            angleValue.textContent = (pendulum.angle * 180 / Math.PI).toFixed(2);
            velocityValue.textContent = pendulum.angularVelocity.toFixed(2);
            periodValue.textContent = pendulum.currentPeriod.toFixed(2);
        }

        // Update pendulum physics
        function updatePendulum(timestamp) {
            if (!pendulum.isRunning) return;
            
            if (!pendulum.lastTime) pendulum.lastTime = timestamp;
            const deltaTime = (timestamp - pendulum.lastTime) / 1000; // Convert to seconds
            pendulum.lastTime = timestamp;
            
            // Calculate angular acceleration (damping included)
            const damping = 0.995; // To prevent perpetual motion
            const angularAcceleration = -pendulum.gravity / pendulum.length * Math.sin(pendulum.angle);
            
            // Update angular velocity and angle
            pendulum.angularVelocity += angularAcceleration * deltaTime;
            pendulum.angularVelocity *= damping;
            pendulum.angle += pendulum.angularVelocity * deltaTime;
            
            // Period measurement
            measurePeriod();
            
            drawPendulum();
            requestAnimationFrame(updatePendulum);
        }

        // Measure the period by tracking zero crossings
        function measurePeriod() {
            const now = Date.now();
            
            // Check for crossing equilibrium (angle = 0) with positive velocity
            if (Math.sign(pendulum.angle) !== Math.sign(pendulum.angle - pendulum.angularVelocity) && 
                pendulum.angularVelocity > 0) {
                
                if (pendulum.lastZeroCrossing) {
                    const period = (now - pendulum.lastZeroCrossing) / 1000; // Convert to seconds
                    pendulum.currentPeriod = period;
                }
                pendulum.lastZeroCrossing = now;
            }
        }

        // Reset the simulation
        function resetSimulation() {
            pendulum.angle = parseFloat(angleSlider.value) * Math.PI / 180;
            pendulum.angularVelocity = 0;
            pendulum.lastZeroCrossing = 0;
            pendulum.currentPeriod = 0;
            pendulum.lastTime = 0;
            drawPendulum();
        }

        // Run experiment with different masses
        function runMassExperiment() {
            const originalMass = pendulum.mass;
            const originalAngle = pendulum.angle;
            const originalRunning = pendulum.isRunning;
            
            // Clear table
            dataTableBody.innerHTML = '';
            
            // Pause simulation if running
            if (pendulum.isRunning) {
                pendulum.isRunning = false;
                startBtn.textContent = 'Start';
            }
            
            // Run tests for masses from 1kg to 20kg
            for (let mass = 1; mass <= 20; mass += 2) {
                pendulum.mass = mass;
                resetSimulation();
                
                // Run simulation briefly to measure period
                pendulum.isRunning = true;
                const startTime = Date.now();
                let periodMeasurements = [];
                
                // Run for 5 seconds to get stable measurement
                const interval = setInterval(() => {
                    if (pendulum.currentPeriod > 0) {
                        periodMeasurements.push(pendulum.currentPeriod);
                    }
                    
                    if (Date.now() - startTime > 5000) {
                        clearInterval(interval);
                        pendulum.isRunning = false;
                        
                        // Calculate average period
                        const avgPeriod = periodMeasurements.length > 0 ? 
                            periodMeasurements.reduce((a, b) => a + b, 0) / periodMeasurements.length : 0;
                        
                        // Calculate theoretical period
                        const theoreticalPeriod = calculateTheoreticalPeriod();
                        
                        // Add row to table
                        const row = document.createElement('tr');
                        row.innerHTML = `
                            <td>${mass}</td>
                            <td>${avgPeriod.toFixed(3)}</td>
                            <td>${theoreticalPeriod.toFixed(3)}</td>
                            <td>${Math.abs(avgPeriod - theoreticalPeriod).toFixed(3)}</td>
                        `;
                        dataTableBody.appendChild(row);
                        
                        // Restore original state after last test
                        if (mass >= 19) {
                            pendulum.mass = originalMass;
                            pendulum.angle = originalAngle;
                            pendulum.isRunning = originalRunning;
                            updateDisplays();
                            drawPendulum();
                        }
                    }
                }, 50);
            }
        }

        // Event listeners
        massSlider.addEventListener('input', () => {
            pendulum.mass = parseFloat(massSlider.value);
            updateDisplays();
            drawPendulum();
        });

        lengthSlider.addEventListener('input', () => {
            pendulum.length = parseFloat(lengthSlider.value);
            updateDisplays();
            drawPendulum();
        });

        gravitySlider.addEventListener('input', () => {
            pendulum.gravity = parseFloat(gravitySlider.value);
            updateDisplays();
            drawPendulum();
        });

        angleSlider.addEventListener('input', () => {
            pendulum.angle = parseFloat(angleSlider.value) * Math.PI / 180;
            updateDisplays();
            drawPendulum();
        });

        startBtn.addEventListener('click', () => {
            if (!pendulum.isRunning) {
                pendulum.isRunning = true;
                pendulum.lastTime = 0;
                startBtn.textContent = 'Running...';
                requestAnimationFrame(updatePendulum);
            }
        });

        pauseBtn.addEventListener('click', () => {
            pendulum.isRunning = false;
            startBtn.textContent = 'Start';
        });

        resetBtn.addEventListener('click', resetSimulation);
        runExperimentBtn.addEventListener('click', runMassExperiment);

        // Initialize
        updateDisplays();
        drawPendulum();
        
        // Add initial data to table
        const initialRow = document.createElement('tr');
        initialRow.innerHTML = `
            <td colspan="4" style="text-align: center; font-style: italic;">
                Run the experiment to see how mass affects the period
            </td>
        `;
        dataTableBody.appendChild(initialRow);
    </script>
</body>
</html>
