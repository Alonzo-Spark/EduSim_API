# Gravity Scaling and Inclined Plane Range Simulation

I'll create an interactive simulation to demonstrate how gravity scaling affects the range of a block sliding down an inclined plane and becoming a projectile.

## Physics Insight
The range of the block is surprisingly **independent of gravity scaling**! Here's why:
- Initial potential energy: $mgh$
- Speed at ramp bottom: $v = \sqrt{2gh}$
- Horizontal velocity: $v_x = v \cos\theta$
- Vertical velocity: $v_y = v \sin\theta$
- Time of flight: $t = \frac{2v_y}{g}$
- **Range**: $R = v_x t = \frac{2v_x v_y}{g} = \frac{2(\sqrt{2gh}\cos\theta)(\sqrt{2gh}\sin\theta)}{g} = 4h\cos\theta\sin\theta$

The $g$ terms cancel out! The reduced acceleration on low-gravity planets is offset by longer flight times.

    <script>
        // Physics simulation using Matter.js
        const Engine = Matter.Engine,
            Render = Matter.Render,
            Runner = Matter.Runner,
            Bodies = Matter.Bodies,
            Body = Matter.Body,
            Composite = Matter.Composite,
            Events = Matter.Events;

        // Create engine
        const engine = Engine.create();
        engine.gravity.y = 0.0098; // Earth gravity (scaled for simulation)

        // Create renderer
        const render = Render.create({
            canvas: document.getElementById('physicsCanvas'),
            engine: engine,
            options: {
                width: document.querySelector('.simulation-container').clientWidth,
                height: document.querySelector('.simulation-container').clientHeight,
                wireframes: false,
                background: '#0f1b30'
            }
        });

        // Simulation parameters
        const params = {
            angle: 30,
            height: 4,
            gravity: 9.8,
            planet: 'earth',
            blockSize: 30
        };

        // Physics objects
        let ground, ramp, block;
        let blockReleased = false;
        let range = 0;
        let flightTime = 0;
        let launchTime = 0;
        let launchX = 0;
        let resultsCalculated = false;

        // Initialize simulation
        function initSimulation() {
            // Clear existing objects
            Composite.clear(engine.world);
            blockReleased = false;
            resultsCalculated = false;
            range = 0;
            flightTime = 0;
            launchTime = 0;
            
            // Calculate dimensions based on parameters
            const canvasHeight = render.options.height;
            const canvasWidth = render.options.width;
            const rampHeightPx = params.height * 40;
            const rampLengthPx = rampHeightPx / Math.sin(params.angle * Math.PI / 180);
            const rampBaseX = canvasWidth * 0.1;
            const rampBaseY = canvasHeight * 0.7;
            
            // Create ground
            ground = Bodies.rectangle(canvasWidth / 2, canvasHeight - 20, canvasWidth, 60, {
                isStatic: true,
                render: { fillStyle: '#3d5a80' }
            });
            
            // Create ramp
            const rampCenterX = rampBaseX + (rampLengthPx * Math.cos(params.angle * Math.PI / 180)) / 2;
            const rampCenterY = rampBaseY - (rampLengthPx * Math.sin(params.angle * Math.PI / 180)) / 2;
            
            ramp = Bodies.rectangle(rampCenterX, rampCenterY, rampLengthPx, 20, {
                isStatic: true,
                angle: -params.angle * Math.PI / 180,
                render: { fillStyle: '#8a3b12' }
            });
            
            // Create block at top of ramp
            const blockX = rampBaseX + params.blockSize/2;
            const blockY = rampBaseY - rampHeightPx + params.blockSize/2;
            
            block = Bodies.rectangle(blockX, blockY, params.blockSize, params.blockSize, {
                render: { fillStyle: '#f6b950' },
                friction: 0.05,
                restitution: 0.2
            });
            
            // Add objects to world
            Composite.add(engine.world, [ground, ramp, block]);
            
            // Store launch position
            launchX = rampBaseX + rampLengthPx * Math.cos(params.angle * Math.PI / 180);
            
            // Update display
            updateDisplay();
        }

        // Event when block leaves the ramp
        Events.on(engine, 'beforeUpdate', function() {
            if (!blockReleased && block.position.x > launchX - params.blockSize) {
                blockReleased = true;
                launchTime = engine.timing.timestamp;
            }
            
            // Record when block hits the ground
            if (blockReleased && !resultsCalculated && block.position.y > render.options.height - 50) {
                const currentTime = engine.timing.timestamp;
                flightTime = (currentTime - launchTime) / 1000; // Convert to seconds
                range = (block.position.x - launchX) / 50; // Convert to meters
                resultsCalculated = true;
                updateDisplay();
            }
        });

        // Update data display
        function updateDisplay() {
            document.getElementById('gravityValue').textContent = params.gravity.toFixed(1) + ' m/s²';
            document.getElementById('rangeValue').textContent = range.toFixed(2) + ' m';
            document.getElementById('timeValue').textContent = flightTime.toFixed(2) + ' s';
        }

        // Set gravity for different planets
        function setGravity(planet) {
            params.planet = planet;
            document.querySelectorAll('.controls button').forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');
            
            switch(planet) {
                case 'earth':
                    params.gravity = 9.8;
                    engine.gravity.y = 0.0098;
                    break;
                case 'moon':
                    params.gravity = 1.6;
                    engine.gravity.y = 0.0016;
                    break;
                case 'mars':
                    params.gravity = 3.7;
                    engine.gravity.y = 0.0037;
                    break;
                case 'jupiter':
                    params.gravity = 24.8;
                    engine.gravity.y = 0.0248;
                    break;
            }
            
            initSimulation();
            updateDisplay();
        }

        // Event listeners
        document.getElementById('earth').addEventListener('click', () => setGravity('earth'));
        document.getElementById('moon').addEventListener('click', () => setGravity('moon'));
        document.getElementById('mars').addEventListener('click', () => setGravity('mars'));
        document.getElementById('jupiter').addEventListener('click', () => setGravity('jupiter'));
        document.getElementById('resetBtn').addEventListener('click', initSimulation);
        
        document.getElementById('angle').addEventListener('input', function() {
            params.angle = parseInt(this.value);
            document.getElementById('angleValue').textContent = params.angle + '°';
            initSimulation();
        });
        
        document.getElementById('height').addEventListener('input', function() {
            params.height = parseFloat(this.value);
            document.getElementById('heightValue').textContent = params.height + 'm';
            initSimulation();
        });

        // Start simulation
        initSimulation();
        Render.run(render);
        const runner = Runner.create();
        Runner.run(runner, engine);
    </script>
</body>
</html>

5. **Physics Explanation**:

The simulation clearly demonstrates that while gravity scaling affects the block's acceleration and flight time, the horizontal range remains constant regardless of the gravitational environment.
