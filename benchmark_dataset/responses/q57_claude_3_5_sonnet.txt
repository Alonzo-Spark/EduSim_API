const AnimatedWave = ({ omega, omega0, gamma, playing }) => {
  const canvasRef = useRef(null);
  const animRef = useRef(null);
  const tRef = useRef(0);

  const A =
    1 /
    Math.sqrt(
      Math.pow(omega0 * omega0 - omega * omega, 2) +
        Math.pow(gamma * omega, 2)
    );
  const Anorm = Math.min(A * omega0 * omega0, 3.5);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    const W = canvas.width;
    const H = canvas.height;

    const draw = () => {
      ctx.clearRect(0, 0, W, H);

      // Grid
      ctx.strokeStyle = "rgba(99,102,241,0.08)";
      ctx.lineWidth = 1;
      for (let x = 0; x < W; x += 40) {
        ctx.beginPath();
        ctx.moveTo(x, 0);
        ctx.lineTo(x, H);
        ctx.stroke();
      }
      for (let y = 0; y < H; y += 40) {
        ctx.beginPath();
        ctx.moveTo(0, y);
        ctx.lineTo(W, y);
        ctx.stroke();
      }

      // Axis
      ctx.strokeStyle = "rgba(148,163,184,0.3)";
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(0, H / 2);
      ctx.lineTo(W, H / 2);
      ctx.stroke();

      // Wave
      const grad = ctx.createLinearGradient(0, 0, W, 0);
      const ratio = omega / omega0;
      if (ratio < 0.85) {
        grad.addColorStop(0, "#6366f1");
        grad.addColorStop(1, "#818cf8");
      } else if (ratio < 1.15) {
        grad.addColorStop(0, "#f59e0b");
        grad.addColorStop(1, "#fbbf24");
      } else {
        grad.addColorStop(0, "#10b981");
        grad.addColorStop(1, "#34d399");
      }

      ctx.beginPath();
      ctx.strokeStyle = grad;
      ctx.lineWidth = 2.5;
      ctx.shadowColor =
        ratio >= 0.85 && ratio <= 1.15 ? "#f59e0b" : "#6366f1";
      ctx.shadowBlur = Anorm > 2 ? 18 : 6;

      for (let px = 0; px < W; px++) {
        const t = tRef.current;
        const xPos = px / W;
        const y =
          H / 2 -
          (Anorm / 3.5) * (H / 2 - 20) * Math.sin(2 * Math.PI * xPos * 3 - omega * t);
        if (px === 0) ctx.moveTo(px, y);
        else ctx.lineTo(px, y);
      }
      ctx.stroke();
      ctx.shadowBlur = 0;

      if (playing) {
        tRef.current += 0.03;
        animRef.current = requestAnimationFrame(draw);
      }
    };

    draw();
    return () => cancelAnimationFrame(animRef.current);
  }, [omega, omega0, gamma, playing, Anorm]);

  return (
    <canvas
      ref={canvasRef}
      width={560}
      height={160}
      style={{ width: "100%", borderRadius: 10 }}
    />
  );
};

const AmplitudeCurve = ({ omega, omega0, gamma }) => {
  const canvasRef = useRef(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    const W = canvas.width;
    const H = canvas.height;

    ctx.clearRect(0, 0, W, H);

    // Grid
    ctx.strokeStyle = "rgba(99,102,241,0.08)";
    ctx.lineWidth = 1;
    for (let x = 0; x < W; x += 40) {
      ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,H); ctx.stroke();
    }
    for (let y = 0; y < H; y += 30) {
      ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(W,y); ctx.stroke();
    }

    // Axes
    ctx.strokeStyle = "rgba(148,163,184,0.4)";
    ctx.lineWidth = 1;
    ctx.beginPath(); ctx.moveTo(30, H-20); ctx.lineTo(W-10, H-20); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(30, 10); ctx.lineTo(30, H-20); ctx.stroke();

    // Labels
    ctx.fillStyle = "#64748b";
    ctx.font = "11px sans-serif";
    ctx.fillText("ω", W - 18, H - 8);
    ctx.fillText("A", 8, 18);

    // Resonance line
    const rx = 30 + (omega0 / (2 * omega0)) * (W - 40);
    ctx.strokeStyle = "rgba(245,158,11,0.25)";
    ctx.setLineDash([4, 4]);
    ctx.lineWidth = 1;
    ctx.beginPath(); ctx.moveTo(rx, 10); ctx.lineTo(rx, H-20); ctx.stroke();
    ctx.setLineDash([]);
    ctx.fillStyle = "#f59e0b55";
    ctx.font = "10px sans-serif";
    ctx.fillText("ω₀", rx - 8, H - 6);

    // Curve
    const maxA = 1 / (gamma * omega0);
    ctx.beginPath();
    ctx.strokeStyle = "#6366f1";
    ctx.lineWidth = 2;
    let first = true;
    for (let px = 31; px < W - 10; px++) {
      const w = ((px - 30) / (W - 40)) * 2 * omega0;
      const denom = Math.sqrt(
        Math.pow(omega0 * omega0 - w * w, 2) + Math.pow(gamma * w, 2)
      );
      const a = denom > 0 ? (1 / denom) / maxA : 0;
      const py = H - 20 - Math.min(a, 1) * (H - 36);
      if (first) { ctx.moveTo(px, py); first = false; }
      else ctx.lineTo(px, py);
    }
    ctx.stroke();

    // Current point
    const curX = 30 + (omega / (2 * omega0)) * (W - 40);
    const denom = Math.sqrt(
      Math.pow(omega0 * omega0 - omega * omega, 2) + Math.pow(gamma * omega, 2)
    );
    const curA = denom > 0 ? (1 / denom) / maxA : 0;
    const curY = H - 20 - Math.min(curA, 1) * (H - 36);

    ctx.beginPath();
    ctx.arc(curX, curY, 6, 0, Math.PI * 2);
    ctx.fillStyle = "#f59e0b";
    ctx.fill();
    ctx.strokeStyle = "#fff";
    ctx.lineWidth = 2;
    ctx.stroke();
  }, [omega, omega0, gamma]);

  return (
    <canvas
      ref={canvasRef}
      width={560}
      height={160}
      style={{ width: "100%", borderRadius: 10 }}
    />
  );
};
