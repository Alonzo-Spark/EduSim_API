            <div class="derivation-step">
                <strong>1. Setup:</strong>
                <p>Consider three resistors <span class="math">R₁, R₂, R₃</span> connected in series to a battery of voltage <span class="math">V</span>. Let the current flowing through the circuit be <span class="math">I</span>.</p>
            </div>

            <div class="derivation-step">
                <strong>2. Apply Ohm's Law:</strong>
                <p>According to Ohm's Law (<span class="math">V = IR</span>), the potential difference across each resistor is:</p>
                <div style="background: #f3f4f6; padding: 10px; border-radius: 8px; margin-top: 5px; font-family: var(--math-font);">
                    V₁ = IR₁, &nbsp; V₂ = IR₂, &nbsp; V₃ = IR₃
                </div>
            </div>

            <div class="derivation-step">
                <strong>3. Total Voltage:</strong>
                <p>In a series circuit, the total voltage is the sum of the individual voltage drops:</p>
                <div style="background: #f3f4f6; padding: 10px; border-radius: 8px; margin-top: 5px; font-family: var(--math-font);">
                    V = V₁ + V₂ + V₃
                </div>
            </div>

            <div class="derivation-step">
                <strong>4. Substitute and Simplify:</strong>
                <p>Substitute the Ohm's law expressions into the voltage equation:</p>
                <div style="background: #f3f4f6; padding: 10px; border-radius: 8px; margin-top: 5px; font-family: var(--math-font);">
                    V = IR₁ + IR₂ + IR₃
                </div>
                <p>Factor out the current <span class="math">I</span>:</p>
                <div style="background: #f3f4f6; padding: 10px; border-radius: 8px; margin-top: 5px; font-family: var(--math-font);">
                    V = I(R₁ + R₂ + R₃)
                </div>
            </div>

            <div class="derivation-step">
                <strong>5. Equivalent Resistance:</strong>
                <p>If we replace the three resistors with a single equivalent resistor <span class="math">Rₑq</span>, Ohm's Law for the whole circuit is <span class="math">V = IRₑq</span>.</p>
                <p>Comparing the two equations for <span class="math">V</span>:</p>
                <div style="background: #f3f4f6; padding: 10px; border-radius: 8px; margin-top: 5px; font-family: var(--math-font);">
                    IRₑq = I(R₁ + R₂ + R₃)
                </div>
                <p>Dividing both sides by <span class="math">I</span>:</p>
                <div style="background: #ecfdf5; padding: 10px; border-radius: 8px; margin-top: 5px; font-weight: bold; font-family: var(--math-font);">
                    Rₑq = R₁ + R₂ + R₃
                </div>
            </div>

    <script>
        /* =========================================
           LOGIC: CIRCUIT SIMULATOR
           ========================================= */
        const r1Input = document.getElementById('input-r1');
        const r2Input = document.getElementById('input-r2');
        const r3Input = document.getElementById('input-r3');
        
        const r1Val = document.getElementById('val-r1');
        const r2Val = document.getElementById('val-r2');
        const r3Val = document.getElementById('val-r3');
        
        const r1Label = document.getElementById('label-r1');
        const r2Label = document.getElementById('label-r2');
        const r3Label = document.getElementById('label-r3');
        
        const reqDisplay = document.getElementById('display-req');

        function updateCircuit() {
            const r1 = parseInt(r1Input.value);
            const r2 = parseInt(r2Input.value);
            const r3 = parseInt(r3Input.value);

            // Update Text Labels
            r1Val.textContent = r1 + ' Ω';
            r2Val.textContent = r2 + ' Ω';
            r3Val.textContent = r3 + ' Ω';

            r1Label.textContent = r1 + 'Ω';
            r2Label.textContent = r2 + 'Ω';
            r3Label.textContent = r3 + 'Ω';

            // Calculate Equivalent Resistance
            const req = r1 + r2 + r3;
            reqDisplay.textContent = req + ' Ω';

            // Update Animation Speed based on resistance (Higher R = Slower Current)
            // Base speed 4s. Max resistance 300.
            const speed = 1 + (req / 20); 
            const circle = document.querySelector('.electron-flow animateMotion');
            circle.setAttribute('dur', speed + 's');
        }

        // Attach Listeners
        [r1Input, r2Input, r3Input].forEach(input => {
            input.addEventListener('input', updateCircuit);
        });

        /* =========================================
           LOGIC: FORMULA LAB
           ========================================= */
        let labResistors = [10, 20, 30];
        const labVisuals = document.getElementById('lab-visuals');
        const labCalcString = document.getElementById('lab-calc-string');
        const labTotal = document.getElementById('lab-total');

        function renderLab() {
            labVisuals.innerHTML = '';
            let total = 0;
            let calcStr = [];

            labResistors.forEach((r, index) => {
                total += r;
                calcStr.push(r);

                // Create visual bar
                const bar = document.createElement('div');
                bar.style.width = '30px';
                // Scale height: assume max 100 for visualization
                bar.style.height = Math.min(r, 100) + '%'; 
                bar.style.backgroundColor = '#3b82f6';
                bar.style.borderRadius = '4px 4px 0 0';
                bar.style.position = 'relative';
                
                // Tooltip/Label
                const label = document.createElement('span');
                label.textContent = r;
                label.style.position = 'absolute';
                label.style.bottom = '-20px';
                label.style.left = '50%';
                label.style.transform = 'translateX(-50%)';
                label.style.fontSize = '10px';
                label.style.color = '#64748b';
                
                bar.appendChild(label);
                labVisuals.appendChild(bar);
            });

            labCalcString.textContent = calcStr.join(' + ');
            labTotal.textContent = total;
        }

        function addResistor() {
            // Add a random resistor between 5 and 50
            const newR = Math.floor(Math.random() * 45) + 5;
            labResistors.push(newR);
            renderLab();
        }

        function removeResistor() {
            if (labResistors.length > 1) {
                labResistors.pop();
                renderLab();
            } else {
                showToast("You need at least one resistor!");
            }
        }

        /* =========================================
           LOGIC: UI UTILITIES
           ========================================= */
        function toggleFormulaLab() {
            const lab = document.getElementById('formula-lab');
            lab.classList.toggle('active');
        }

        function showToast(message) {
            const toast = document.getElementById('toast');
            toast.textContent = message;
            toast.classList.add('show');
            setTimeout(() => {
                toast.classList.remove('show');
            }, 3000);
        }

        // Initialize
        updateCircuit();
        renderLab();

    </script>
