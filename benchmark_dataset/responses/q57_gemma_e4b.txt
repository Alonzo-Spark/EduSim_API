```python
import re

def extract_formulas(markdown_text):
    """Extracts LaTeX formulas from the markdown text."""
    formula_blocks = re.findall(r"$$([^$]+)$$", markdown_text)
    formulas = []
    for formula in formula_blocks:
        formulas.append({
            "formula": formula,
            "title": "Formula", #Default title
            "surroundingContext": "",
            "variables": []
        })
    return formulas
```
