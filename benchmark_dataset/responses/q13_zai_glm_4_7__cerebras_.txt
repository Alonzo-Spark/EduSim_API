        // 1. Render Math with KaTeX
        document.addEventListener("DOMContentLoaded", function() {
            renderMathInElement(document.body, {

                throwOnError: false

        const formulaData = {
            formula: "F = ma",
            title: "Newton's Second Law",
            variables: {
                F: { name: "Force", unit: "N" },
                m: { name: "Mass", unit: "kg" },
                a: { name: "Acceleration", unit: "m/s²" }
            }
        };

        // 3. Interaction Logic
        
        // Formula Lab Logic
        const labModal = document.getElementById('labModal');
        const explainModal = document.getElementById('explainModal');
        const forceInput = document.getElementById('forceInput');
        const massInput = document.getElementById('massInput');
        const forceVal = document.getElementById('forceVal');
        const massVal = document.getElementById('massVal');
        const accelResult = document.getElementById('accelResult');

        function openFormulaLab() {
            labModal.classList.add('active');
            calculateLab(); // Initial calculation
        }

        function closeFormulaLab() {
            labModal.classList.remove('active');
        }

        function explainFormula() {
            explainModal.classList.add('active');
        }

        function closeExplain() {
            explainModal.classList.remove('active');
        }

        // Close modals on outside click
        window.onclick = function(event) {
            if (event.target == labModal) {
                closeFormulaLab();
            }
            if (event.target == explainModal) {
                closeExplain();
            }
        }

        // Lab Calculation Logic
        function calculateLab() {
            const F = parseFloat(forceInput.value);
            const m = parseFloat(massInput.value);
            
            // Update Labels
            forceVal.textContent = F + " N";
            massVal.textContent = m + " kg";

            // Calculate Acceleration
            let a = 0;
            if (m > 0) {
                a = F / m;
            }

            // Update Result with animation effect
            accelResult.textContent = a.toFixed(2);
        }

        // Event Listeners for Sliders
        forceInput.addEventListener('input', calculateLab);
        massInput.addEventListener('input', calculateLab);

    </script>
