    <script>
        // Toggle Explanation Panels
        function toggleExplanation(id, btn) {
            const panel = document.getElementById(id);
            const isActive = panel.classList.contains('active');
            
            // Close all other panels first (optional, keeps it clean)
            document.querySelectorAll('.interactive-panel').forEach(p => p.classList.remove('active'));
            
            if (!isActive) {
                panel.classList.add('active');
                btn.innerHTML = '<span>🔼 Hide Explanation</span>';
            } else {
                btn.innerHTML = '<span>ℹ️ Explain Formula</span>';
            }
        }

        // Toggle Lab Panels
        function toggleLab(id, btn) {
            const panel = document.getElementById(id);
            const isActive = panel.classList.contains('active');
            
            if (!isActive) {
                panel.classList.add('active');
                btn.innerHTML = '<span>🔼 Close Formula Lab</span>';
            } else {
                panel.classList.remove('active');
                btn.innerHTML = '<span>🧪 Open Formula Lab</span>';
            }
        }

        // Formula Lab Logic for a_c = v^2 / r
        function calculateAC() {
            const v = parseFloat(document.getElementById('input-v').value);
            const r = parseFloat(document.getElementById('input-r').value);
            
            // Update labels
            document.getElementById('val-v').textContent = v;
            document.getElementById('val-r').textContent = r;
            
            // Calculate
            const ac = (Math.pow(v, 2) / r).toFixed(2);
            
            // Update Result
            document.getElementById('result-ac').textContent = ac;
        }
